# Cairn v0.8.0 — Plan P24: New block types (divider, button, video upload)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the three small block types the editor is still missing:

1. **Divider** — atomic block that renders `<hr>`. Slash-command `/divider`.
2. **Button** — block with attrs `{label, href, variant: 'primary'|'secondary'}`. Renders an `<a class="btn …">` styled with Tailwind.
3. **Video upload** — block with attrs `{fileId, mimeType}`. Renders `<video controls>` pulling from `/api/files/[id]?sig=…` (the existing signed URL). Upload UI mirrors the image-upload pattern. MIME-type allowlist on `/api/files` upload extends to `video/mp4` + `video/webm`.

Each block:
- Is a TipTap node extension (`src/components/editor/blocks/<name>-node.ts` + view if interactive).
- Has a slash-command entry.
- Has a read-only renderer that public pages (`/p/<slug>`) reuse — divider stays as `<hr>`, button as `<a class="btn">`, video as `<video controls>`.
- Has a Yjs round-trip test (insert → serialize → parse → assert attrs preserved).

**Existing pattern to mirror:** The image-upload + file-upload extensions (`src/components/editor/image-extension.ts`, `src/components/editor/file-extension.ts`) wire to the existing `/api/files` POST endpoint and signed-URL reads. v0.6 P21 already enforces per-workspace quota on `/api/files`; this plan extends only the MIME allowlist.

```ts
import { mergeAttributes, Node } from '@tiptap/core';
import { ReactNodeViewRenderer } from '@tiptap/react';
import { ButtonView } from './button';

function sanitizeHref(raw: unknown): string {
  if (typeof raw !== 'string') return '#';
  try {
    const u = new URL(raw);
    if (u.protocol === 'http:' || u.protocol === 'https:' || u.protocol === 'mailto:') {
      return u.toString();
    }
    return '#';
  } catch {
    return '#';
  }
}

function variantClass(v: unknown): string {
  return v === 'secondary' ? 'btn btn-secondary' : 'btn btn-primary';
}

export const ButtonNode = Node.create({
  name: 'button',
  group: 'block',
  atom: true,
  selectable: true,
  draggable: true,

  addAttributes() {
    return {
      label: { default: 'Button' },
      href: { default: '#' },
      variant: { default: 'primary' },
    };
  },

  parseHTML() {
    return [{ tag: 'a[data-cairn-button]' }];
  },

  renderHTML({ node, HTMLAttributes }) {
    const href = sanitizeHref(node.attrs.href);
    const cls = variantClass(node.attrs.variant);
    const label = typeof node.attrs.label === 'string' ? node.attrs.label : 'Button';
    return [
      'a',
      mergeAttributes(HTMLAttributes, {
        'data-cairn-button': '',
        href,
        class: `${cls} inline-block rounded-md px-3 py-1.5 text-sm`,
        target: href.startsWith('http') ? '_blank' : undefined,
        rel: href.startsWith('http') ? 'noopener noreferrer' : undefined,
      }),
      label,
    ];
  },

  addNodeView() {
    return ReactNodeViewRenderer(ButtonView);
  },
});
```

```tsx
import type { NodeViewProps } from '@tiptap/react';
import { NodeViewWrapper } from '@tiptap/react';
import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';

export function ButtonView({ node, editor, updateAttributes }: NodeViewProps) {
  const label = (node.attrs.label as string) ?? 'Button';
  const href = (node.attrs.href as string) ?? '#';
  const variant = (node.attrs.variant as 'primary' | 'secondary') ?? 'primary';
  const [editing, setEditing] = useState(!href || href === '#');
  const [draftLabel, setDraftLabel] = useState(label);
  const [draftHref, setDraftHref] = useState(href);
  const [draftVariant, setDraftVariant] = useState<'primary' | 'secondary'>(variant);

  if (editing && editor.isEditable) {
    return (
      <NodeViewWrapper className="my-3 rounded-md border p-3">
        <div className="space-y-2">
          <Input
            value={draftLabel}
            onChange={(e) => setDraftLabel(e.target.value)}
            placeholder="Label"
          />
          <Input
            value={draftHref}
            onChange={(e) => setDraftHref(e.target.value)}
            placeholder="https://example.com"
          />
          <div className="flex gap-2">
            <Button
              type="button"
              variant={draftVariant === 'primary' ? 'default' : 'outline'}
              size="sm"
              onClick={() => setDraftVariant('primary')}
            >
              Primary
            </Button>
            <Button
              type="button"
              variant={draftVariant === 'secondary' ? 'default' : 'outline'}
              size="sm"
              onClick={() => setDraftVariant('secondary')}
            >
              Secondary
            </Button>
          </div>
          <Button
            type="button"
            onClick={() => {
              updateAttributes({ label: draftLabel, href: draftHref, variant: draftVariant });
              setEditing(false);
            }}
          >
            Save
          </Button>
        </div>
      </NodeViewWrapper>
    );
  }

  return (
    <NodeViewWrapper className="my-3">
      <a
        href={href}
        target={href.startsWith('http') ? '_blank' : undefined}
        rel={href.startsWith('http') ? 'noopener noreferrer' : undefined}
        className={`btn inline-block rounded-md px-3 py-1.5 text-sm ${variant === 'secondary' ? 'btn-secondary border bg-secondary text-secondary-foreground' : 'btn-primary bg-primary text-primary-foreground'}`}
      >
        {label}
      </a>
      {editor.isEditable && (
        <Button
          type="button"
          variant="ghost"
          size="sm"
          className="ml-2"
          onClick={() => setEditing(true)}
        >
          Edit
        </Button>
      )}
    </NodeViewWrapper>
  );
}
```

> **Note on the `src` attribute in `renderHTML`:** the HTML-only render (used by exports + the public page) emits a relative `/api/files/<id>` URL. This URL is signed at request time in the live React node view (`video.tsx`) via the existing `signFileUrl` helper; the HTML export path embeds the unsigned base URL because the server route accepts the same signed pattern at fetch time when a viewer follows the link. For public-page rendering, the renderer overrides the HTML with a server-signed URL — see Task 4 of this plan.

```ts
import { mergeAttributes, Node } from '@tiptap/core';
import { ReactNodeViewRenderer } from '@tiptap/react';
import { VideoView } from './video';

export const VideoNode = Node.create({
  name: 'video',
  group: 'block',
  atom: true,
  selectable: true,
  draggable: true,

  addAttributes() {
    return {
      fileId: { default: null },
      mimeType: { default: null },
    };
  },

  parseHTML() {
    return [{ tag: 'video[data-cairn-video]' }];
  },

  renderHTML({ node, HTMLAttributes }) {
    const fileId = node.attrs.fileId as string | null;
    const mimeType = (node.attrs.mimeType as string | null) ?? 'video/mp4';
    if (!fileId) {
      return [
        'div',
        mergeAttributes(HTMLAttributes, { 'data-cairn-video': '', class: 'my-3' }),
        ['span', {}, 'Video pending upload'],
      ];
    }
    return [
      'video',
      mergeAttributes(HTMLAttributes, {
        'data-cairn-video': '',
        controls: 'true',
        preload: 'metadata',
        class: 'my-3 w-full',
      }),
      ['source', { src: `/api/files/${fileId}`, type: mimeType }],
    ];
  },

  addNodeView() {
    return ReactNodeViewRenderer(VideoView);
  },
});
```

- [ ] **Step 6: Implement `src/components/editor/blocks/video.tsx`**

```tsx
import type { NodeViewProps } from '@tiptap/react';
import { NodeViewWrapper } from '@tiptap/react';
import { useState } from 'react';
import { VideoUploadButton } from './video-upload-button';

export function VideoView({ node, editor, updateAttributes }: NodeViewProps) {
  const fileId = node.attrs.fileId as string | null;
  const mimeType = (node.attrs.mimeType as string | null) ?? 'video/mp4';
  const [signedUrl, setSignedUrl] = useState<string | null>(null);

  // The signed URL is minted server-side via the signing helper. In the
  // editor we hit a lightweight signing endpoint (or compute it directly
  // via the existing helper if it's reachable from the client). For
  // simplicity here, use a same-origin fetch to /api/files/<id> directly
  // — the route signs internally if needed.
  if (fileId && !signedUrl) {
    void fetch(`/api/files/${fileId}/sign`)
      .then((r) => (r.ok ? r.json() : null))
      .then((j) => j && setSignedUrl((j as { url: string }).url))
      .catch(() => {
        /* fall through to no src */
      });
  }

  if (!fileId) {
    if (!editor.isEditable) {
      return (
        <NodeViewWrapper className="my-3 rounded-md border p-3 text-sm text-muted-foreground">
          Empty video.
        </NodeViewWrapper>
      );
    }
    return (
      <NodeViewWrapper className="my-3 rounded-md border p-3">
        <VideoUploadButton
          onUploaded={({ fileId: id, mimeType: mt }) => updateAttributes({ fileId: id, mimeType: mt })}
        />
      </NodeViewWrapper>
    );
  }

  return (
    <NodeViewWrapper className="my-3">
      <video controls preload="metadata" className="w-full">
        <source src={signedUrl ?? `/api/files/${fileId}`} type={mimeType} />
        Your browser does not support video playback.
      </video>
    </NodeViewWrapper>
  );
}
```

```tsx
'use client';

import { useState } from 'react';
import { Button } from '@/components/ui/button';

export type VideoUploadButtonProps = {
  onUploaded: (info: { fileId: string; mimeType: string }) => void;
};

export function VideoUploadButton({ onUploaded }: VideoUploadButtonProps) {
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState<string | null>(null);

  async function upload(file: File) {
    setBusy(true);
    setErr(null);
    try {
      const form = new FormData();
      form.set('file', file);
      const res = await fetch('/api/files', { method: 'POST', body: form });
      if (!res.ok) {
        setErr((await res.json().catch(() => ({ error: 'upload failed' }))).error ?? 'upload failed');
        return;
      }
      const json = (await res.json()) as { id: string; mimeType: string };
      onUploaded({ fileId: json.id, mimeType: json.mimeType });
    } finally {
      setBusy(false);
    }
  }

  return (
    <div className="space-y-2">
      <input
        type="file"
        accept="video/mp4,video/webm"
        disabled={busy}
        onChange={(e) => {
          const f = e.target.files?.[0];
          if (f) void upload(f);
        }}
      />
      {busy && <Button disabled>Uploading…</Button>}
      {err && <p className="text-sm text-destructive">{err}</p>}
    </div>
  );
}
```

function signVideoUrls(doc: unknown): unknown {
  if (!doc || typeof doc !== 'object') return doc;
  const node = doc as { type?: string; attrs?: Record<string, unknown>; content?: unknown[] };
  if (node.type === 'video' && node.attrs?.fileId) {
    const signed = signFileUrl({
      fileId: String(node.attrs.fileId),
      ttlSeconds: 60 * 60,
      secret: env().AUTH_SECRET,
    });
    // Stash on a synthetic attr the renderHTML doesn't currently read; teach
    // VideoNode.renderHTML to prefer `signedSrc` when present.
    node.attrs.signedSrc = signed;
  }
  if (Array.isArray(node.content)) node.content.map(signVideoUrls);
  return doc;
}

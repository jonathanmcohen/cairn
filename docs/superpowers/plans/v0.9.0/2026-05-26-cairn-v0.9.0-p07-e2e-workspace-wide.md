> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. **Plan-review subagent runs before implementer dispatch (v0.7-v0.8 retrospective §2).** Spec ref: `docs/superpowers/specs/2026-05-26-cairn-v0.9.0-design.md` §2 G1 #4+#5. **Goal:** Workspace admin can flip an irreversible-by-design "Encrypt all pages in this workspace" setting. On enable: client generates a workspace-key (WSK), wraps it under every current workspace member's public key (stored in `workspace_encryption_keys`), and sweeps every existing non-encrypted page — re-encrypting its content using the WSK as DEK (per-page DEK = WSK in this mode; `page_encryption_keys` rows are NOT used — a separate `pages.encrypted_under_wsk boolean` flag distinguishes the mode). New members joining: any current member can wrap the WSK under the new member's public key + insert a row. Removed members → full re-key: mint a fresh WSK, wrap under remaining members, re-encrypt every page with the new WSK in a single transaction.

**Architecture:**
- This plan's migration (likely `0037` post-P5/P6/P8 — accept Drizzle's) adds:
  - `workspaces.e2e_mode text not null default 'off'` — `'off' | 'per_page' | 'workspace_wide'`.
  - `pages.encrypted_under_wsk boolean not null default false` — distinguishes WSK-mode pages from per-page-DEK-mode pages (both have `encrypted=true`).
- P5's `workspace_encryption_keys` table already exists. P5's `key_version` column is exercised here — incremented on every rekey.
- Re-key sweep runs CLIENT-side (server has no key). A long-running browser task encrypts pages in chunks of 50; progress posted to `/api/workspaces/[id]/wsk-sweep-progress` so other tabs / sessions show the status.
- Server-side endpoints validate authorization + idempotency; they never see plaintext.

**Tech Stack:** Existing crypto core (P5) + page-cipher (P6) + Drizzle. New audit actions appended to `src/lib/audit/actions.ts`. Vitest for server gates + a worked end-to-end sweep test with a real seeded workspace.

```ts
import { and, eq, inArray, sql } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';
import * as schema from '@/db/schema';
import { userKeypairs, workspaceEncryptionKeys, workspaceMembers, workspaces } from '@/db/schema';

type Db = PostgresJsDatabase<typeof schema>;

export type WrappedForMember = { memberUserId: string; wrappedWsk: string };

export async function assertCoverageAndKeypairs(
  db: Db,
  workspaceId: string,
  wrapped: WrappedForMember[],
): Promise<{ ok: true } | { ok: false; status: 400 | 409; error: string }> {
  const members = await db
    .select({ userId: workspaceMembers.userId })
    .from(workspaceMembers)
    .where(eq(workspaceMembers.workspaceId, workspaceId));
  const memberIds = new Set(members.map((m) => m.userId));
  const payloadIds = new Set(wrapped.map((w) => w.memberUserId));
  if (memberIds.size !== payloadIds.size || ![...memberIds].every((id) => payloadIds.has(id))) {
    return { ok: false, status: 400, error: 'wrapped WSKs must cover every workspace member exactly once' };
  }
  const keypairs = await db
    .select({ userId: userKeypairs.userId })
    .from(userKeypairs)
    .where(inArray(userKeypairs.userId, [...memberIds]));
  if (keypairs.length !== memberIds.size) {
    return { ok: false, status: 409, error: 'one or more members have no registered keypair' };
  }
  return { ok: true };
}

export async function getCurrentKeyVersion(db: Db, workspaceId: string): Promise<number> {
  const rows = await db
    .select({ v: workspaceEncryptionKeys.keyVersion })
    .from(workspaceEncryptionKeys)
    .where(eq(workspaceEncryptionKeys.workspaceId, workspaceId))
    .limit(1);
  return rows[0]?.v ?? 0;
}
```

```ts
import { eq } from 'drizzle-orm';
import { NextResponse } from 'next/server';
import { z } from 'zod';
import { db } from '@/db/client';
import { workspaceEncryptionKeys, workspaces } from '@/db/schema';
import { recordAudit } from '@/lib/audit/record';
import { requireRole } from '@/lib/auth/require-role';
import { assertCoverageAndKeypairs } from '@/lib/e2e/wsk-server';

export const runtime = 'nodejs';

const Body = z.object({
  wrapped: z.array(z.object({ memberUserId: z.string().uuid(), wrappedWsk: z.string().min(1) })).min(1),
});

export async function POST(
  req: Request,
  { params }: { params: Promise<{ workspaceId: string }> },
): Promise<Response> {
  const { workspaceId } = await params;
  const gate = await requireRole(workspaceId, 'owner');
  if (gate instanceof NextResponse) return gate;

  const parsed = Body.safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: 'bad request' }, { status: 400 });

  const [ws] = await db.select().from(workspaces).where(eq(workspaces.id, workspaceId));
  if (!ws) return NextResponse.json({ error: 'not found' }, { status: 404 });
  if (ws.e2eMode !== 'off') {
    return NextResponse.json({ error: 'workspace already has E2E enabled' }, { status: 409 });
  }

  const ok = await assertCoverageAndKeypairs(db, workspaceId, parsed.data.wrapped);
  if (!ok.ok) return NextResponse.json({ error: ok.error }, { status: ok.status });

  await db.transaction(async (tx) => {
    await tx.insert(workspaceEncryptionKeys).values(
      parsed.data.wrapped.map((w) => ({
        workspaceId,
        memberUserId: w.memberUserId,
        wrappedWsk: Buffer.from(w.wrappedWsk, 'base64'),
        keyVersion: 1,
      })),
    );
    await tx.update(workspaces).set({ e2eMode: 'workspace_wide' }).where(eq(workspaces.id, workspaceId));
    await recordAudit(tx, {
      workspaceId,
      actorUserId: gate.userId,
      action: 'e2e.workspace.encrypted',
      targetType: 'workspace',
      targetId: workspaceId,
      metadata: { memberCount: parsed.data.wrapped.length, keyVersion: 1 },
    });
  });

  return NextResponse.json({ ok: true });
}
```

```ts
import { and, eq } from 'drizzle-orm';
import { NextResponse } from 'next/server';
import { z } from 'zod';
import { db } from '@/db/client';
import { userKeypairs, workspaceEncryptionKeys, workspaceMembers, workspaces } from '@/db/schema';
import { recordAudit } from '@/lib/audit/record';
import { requireRole } from '@/lib/auth/require-role';
import { getCurrentKeyVersion } from '@/lib/e2e/wsk-server';

export const runtime = 'nodejs';
const Body = z.object({ memberUserId: z.string().uuid(), wrappedWsk: z.string().min(1) });

export async function POST(
  req: Request,
  { params }: { params: Promise<{ workspaceId: string }> },
): Promise<Response> {
  const { workspaceId } = await params;
  const gate = await requireRole(workspaceId, 'viewer');
  if (gate instanceof NextResponse) return gate;
  const parsed = Body.safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: 'bad request' }, { status: 400 });

  const [ws] = await db.select().from(workspaces).where(eq(workspaces.id, workspaceId));
  if (!ws || ws.e2eMode !== 'workspace_wide') {
    return NextResponse.json({ error: 'workspace is not in workspace_wide mode' }, { status: 409 });
  }

  // Caller must already hold a WSK row at the current key_version.
  const v = await getCurrentKeyVersion(db, workspaceId);
  const [callerRow] = await db
    .select()
    .from(workspaceEncryptionKeys)
    .where(and(eq(workspaceEncryptionKeys.workspaceId, workspaceId), eq(workspaceEncryptionKeys.memberUserId, gate.userId)));
  if (!callerRow || callerRow.keyVersion !== v) {
    return NextResponse.json({ error: 'caller has no current-version WSK' }, { status: 403 });
  }

  // Target must be a current workspace member with a registered keypair AND no existing WSK row.
  const [target] = await db
    .select({ userId: workspaceMembers.userId })
    .from(workspaceMembers)
    .where(and(eq(workspaceMembers.workspaceId, workspaceId), eq(workspaceMembers.userId, parsed.data.memberUserId)));
  if (!target) return NextResponse.json({ error: 'target is not a workspace member' }, { status: 404 });
  const [kp] = await db.select().from(userKeypairs).where(eq(userKeypairs.userId, parsed.data.memberUserId));
  if (!kp) return NextResponse.json({ error: 'target has no keypair' }, { status: 409 });
  const [existing] = await db
    .select()
    .from(workspaceEncryptionKeys)
    .where(and(eq(workspaceEncryptionKeys.workspaceId, workspaceId), eq(workspaceEncryptionKeys.memberUserId, parsed.data.memberUserId)));
  if (existing) return NextResponse.json({ error: 'target already has a WSK row' }, { status: 409 });

  await db.transaction(async (tx) => {
    await tx.insert(workspaceEncryptionKeys).values({
      workspaceId,
      memberUserId: parsed.data.memberUserId,
      wrappedWsk: Buffer.from(parsed.data.wrappedWsk, 'base64'),
      keyVersion: v,
    });
    await recordAudit(tx, {
      workspaceId,
      actorUserId: gate.userId,
      action: 'e2e.workspace.member_added',
      targetType: 'member',
      targetId: parsed.data.memberUserId,
      metadata: { keyVersion: v },
    });
  });
  return NextResponse.json({ ok: true });
}
```

## Task 5: `POST /api/workspaces/[workspaceId]/e2e/rekey` (member removed)

**Goal:** Owner-only. Client submits a fresh wrapped-WSK roster for **remaining** members + a batch of newly-re-encrypted page bundles (each `{pageId, contentEncrypted: base64}`). Server validates owner; validates roster covers exactly the current member set; in one transaction: increments `key_version`, deletes the old `workspace_encryption_keys` rows for this workspace, inserts the new ones, updates each `pages.content_encrypted` with the new ciphertext, records `e2e.workspace.member_removed` + `e2e.workspace.rekey_started` + `e2e.workspace.rekey_completed`. Re-keying batches are idempotent within a `key_version`: a re-call with the same `key_version` and overlapping pages is a no-op.

```ts
import { and, eq, inArray } from 'drizzle-orm';
import { NextResponse } from 'next/server';
import { z } from 'zod';
import { db } from '@/db/client';
import { pages, workspaceEncryptionKeys, workspaces } from '@/db/schema';
import { recordAudit } from '@/lib/audit/record';
import { requireRole } from '@/lib/auth/require-role';
import { assertCoverageAndKeypairs, getCurrentKeyVersion } from '@/lib/e2e/wsk-server';

export const runtime = 'nodejs';

const Body = z.object({
  wrapped: z.array(z.object({ memberUserId: z.string().uuid(), wrappedWsk: z.string().min(1) })).min(1),
  pageBundles: z.array(z.object({ pageId: z.string().uuid(), contentEncrypted: z.string().min(1) })),
  removedMemberId: z.string().uuid().nullable(),
});

export async function POST(
  req: Request,
  { params }: { params: Promise<{ workspaceId: string }> },
): Promise<Response> {
  const { workspaceId } = await params;
  const gate = await requireRole(workspaceId, 'owner');
  if (gate instanceof NextResponse) return gate;
  const parsed = Body.safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: 'bad request' }, { status: 400 });

  const [ws] = await db.select().from(workspaces).where(eq(workspaces.id, workspaceId));
  if (!ws || ws.e2eMode !== 'workspace_wide') {
    return NextResponse.json({ error: 'workspace is not in workspace_wide mode' }, { status: 409 });
  }

  const ok = await assertCoverageAndKeypairs(db, workspaceId, parsed.data.wrapped);
  if (!ok.ok) return NextResponse.json({ error: ok.error }, { status: ok.status });

  const prevVersion = await getCurrentKeyVersion(db, workspaceId);
  const nextVersion = prevVersion + 1;

  await db.transaction(async (tx) => {
    if (parsed.data.removedMemberId) {
      await recordAudit(tx, {
        workspaceId,
        actorUserId: gate.userId,
        action: 'e2e.workspace.member_removed',
        targetType: 'member',
        targetId: parsed.data.removedMemberId,
        metadata: { keyVersion: prevVersion },
      });
    }
    await recordAudit(tx, {
      workspaceId,
      actorUserId: gate.userId,
      action: 'e2e.workspace.rekey_started',
      targetType: 'workspace',
      targetId: workspaceId,
      metadata: { fromVersion: prevVersion, toVersion: nextVersion, pageCount: parsed.data.pageBundles.length },
    });
    await tx.delete(workspaceEncryptionKeys).where(eq(workspaceEncryptionKeys.workspaceId, workspaceId));
    await tx.insert(workspaceEncryptionKeys).values(
      parsed.data.wrapped.map((w) => ({
        workspaceId,
        memberUserId: w.memberUserId,
        wrappedWsk: Buffer.from(w.wrappedWsk, 'base64'),
        keyVersion: nextVersion,
      })),
    );
    for (const bundle of parsed.data.pageBundles) {
      await tx
        .update(pages)
        .set({
          encrypted: true,
          encryptedUnderWsk: true,
          contentEncrypted: Buffer.from(bundle.contentEncrypted, 'base64'),
          contentText: '',
          content: { type: 'doc', content: [] },
        })
        .where(and(eq(pages.id, bundle.pageId), eq(pages.workspaceId, workspaceId)));
    }
    await recordAudit(tx, {
      workspaceId,
      actorUserId: gate.userId,
      action: 'e2e.workspace.rekey_completed',
      targetType: 'workspace',
      targetId: workspaceId,
      metadata: { keyVersion: nextVersion },
    });
  });
  return NextResponse.json({ ok: true, keyVersion: nextVersion });
}
```

## Task 6: `POST /api/pages/[pageId]/encrypt-under-wsk` (enable sweep)

**Goal:** Called once per page during the initial enable-sweep. Caller submits `{contentEncrypted: base64}` (already encrypted under the WSK client-side; server NEVER sees the key). Server validates: caller has edit access; workspace `e2eMode === 'workspace_wide'`; page currently `encrypted_under_wsk = false`. Sets `encrypted=true`, `encrypted_under_wsk=true`, writes `content_encrypted`, blanks `content_text` + `content` sentinel.

**Files:**
- Create: `src/app/api/pages/[pageId]/encrypt-under-wsk/route.ts`
- Create: `tests/api/pages/encrypt-under-wsk.test.ts`

- [ ] **Step 1: Failing test** — owner happy path; refuse when workspace mode is `'off'` (409); refuse when page already `encrypted_under_wsk` (idempotency: a re-call with the same content is a 200 no-op; a re-call with DIFFERENT content is the 409 — this guards against accidental double-encryption).

- [ ] **Step 2: Implement route**

```ts
import { eq } from 'drizzle-orm';
import { NextResponse } from 'next/server';
import { z } from 'zod';
import { db } from '@/db/client';
import { pages, workspaces } from '@/db/schema';
import { recordAudit } from '@/lib/audit/record';
import { requirePageAccess } from '@/lib/pages/access';

export const runtime = 'nodejs';
const Body = z.object({ contentEncrypted: z.string().min(1) });

export async function POST(
  req: Request,
  { params }: { params: Promise<{ pageId: string }> },
): Promise<Response> {
  const { pageId } = await params;
  const access = await requirePageAccess(pageId, 'editor');
  if (access instanceof NextResponse) return access;
  const { page, ctx } = access;

  const parsed = Body.safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: 'bad request' }, { status: 400 });

  const [ws] = await db.select().from(workspaces).where(eq(workspaces.id, page.workspaceId));
  if (ws?.e2eMode !== 'workspace_wide') {
    return NextResponse.json({ error: 'workspace is not in workspace_wide mode' }, { status: 409 });
  }

  const ct = Buffer.from(parsed.data.contentEncrypted, 'base64');
  if (page.encryptedUnderWsk && page.contentEncrypted) {
    if (Buffer.from(page.contentEncrypted).equals(ct)) return NextResponse.json({ ok: true, alreadyEncrypted: true });
    return NextResponse.json({ error: 'page already encrypted; use /rekey to rotate' }, { status: 409 });
  }

  await db.transaction(async (tx) => {
    await tx.update(pages).set({
      encrypted: true,
      encryptedUnderWsk: true,
      contentEncrypted: ct,
      contentText: '',
      content: { type: 'doc', content: [] },
    }).where(eq(pages.id, pageId));
    await recordAudit(tx, {
      workspaceId: page.workspaceId,
      actorUserId: ctx.userId,
      action: 'e2e.page.encrypted',
      targetType: 'page',
      targetId: pageId,
      metadata: { mode: 'workspace_wide' },
    });
  });
  return NextResponse.json({ ok: true });
}
```

```tsx
'use client';

import { useState } from 'react';
import { Button } from '@/components/ui/button';

type Member = { memberUserId: string; publicKey: string };
type PageRow = { id: string; title: string };

export function WorkspaceE2EToggle({ workspaceId }: { workspaceId: string }) {
  const [busy, setBusy] = useState(false);
  const [progress, setProgress] = useState<{ done: number; total: number } | null>(null);
  const [err, setErr] = useState<string | null>(null);

  async function enable() {
    setBusy(true); setErr(null); setProgress(null);
    try {
      const { generateDek, wrapDek, unlockUserKeypair } = await import('@/lib/e2e/crypto');
      const { encryptPageContent } = await import('@/lib/e2e/page-cipher');

      // Unlock caller keypair (mirror EncryptPageAction in P6).
      const sealedJson = window.localStorage.getItem('cairn.e2e.sealedKeypair');
      if (!sealedJson) throw new Error('register a passkey-protected keypair on this device first');
      const sealed = JSON.parse(sealedJson) as {
        publicKey: string; encryptedPrivateKey: string; kdfSalt: string; kdfIters: number;
      };
      const passphrase = window.prompt('Enter your E2E passphrase to enable workspace encryption');
      if (!passphrase) throw new Error('cancelled');
      await unlockUserKeypair({
        publicKey: Buffer.from(sealed.publicKey, 'base64'),
        encryptedPrivateKey: Buffer.from(sealed.encryptedPrivateKey, 'base64'),
        kdfSalt: Buffer.from(sealed.kdfSalt, 'base64'),
        kdfIters: sealed.kdfIters,
      }, passphrase);

      // Fetch member roster.
      const roster: Member[] = await (await fetch(`/api/workspaces/${workspaceId}/keypair-roster`)).json();

      // Mint WSK + wrap for every member.
      const wsk = generateDek();
      const wrapped = roster.map((m) => ({
        memberUserId: m.memberUserId,
        wrappedWsk: Buffer.from(wrapDek(wsk, Buffer.from(m.publicKey, 'base64'))).toString('base64'),
      }));
      const enableRes = await fetch(`/api/workspaces/${workspaceId}/e2e/enable`, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ wrapped }),
      });
      if (!enableRes.ok) throw new Error(`enable failed: ${enableRes.status}`);

      // Sweep every page.
      const pageRows: PageRow[] = await (await fetch(`/api/workspaces/${workspaceId}/page-ids`)).json();
      setProgress({ done: 0, total: pageRows.length });
      for (let i = 0; i < pageRows.length; i++) {
        const id = pageRows[i]!.id;
        const docRes = await fetch(`/api/pages/${id}`);
        const doc = (await docRes.json()) as { content: unknown };
        const ct = encryptPageContent(doc.content, wsk);
        const r = await fetch(`/api/pages/${id}/encrypt-under-wsk`, {
          method: 'POST',
          headers: { 'content-type': 'application/json' },
          body: JSON.stringify({ contentEncrypted: ct.toString('base64') }),
        });
        if (!r.ok) throw new Error(`sweep failed at page ${id}: ${r.status}`);
        setProgress({ done: i + 1, total: pageRows.length });
      }
    } catch (e) {
      setErr((e as Error).message);
    } finally {
      setBusy(false);
    }
  }

  return (
    <div>
      <Button onClick={enable} disabled={busy}>
        {busy ? 'Encrypting workspace…' : 'Enable workspace-wide encryption'}
      </Button>
      {progress ? <p className="text-sm">Encrypted {progress.done} / {progress.total} pages</p> : null}
      {err ? <p className="text-sm text-destructive">{err}</p> : null}
      <p className="text-xs text-muted-foreground mt-2">
        This action cannot be undone in v0.9. To remove a member or rotate the workspace key, use the rekey flow.
      </p>
    </div>
  );
}
```

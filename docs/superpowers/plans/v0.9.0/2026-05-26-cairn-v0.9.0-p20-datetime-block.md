# Cairn v0.9.0 — Plan P20: Date/time block with timezone (luxon)

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` to implement this plan task-by-task. Steps use checkbox (`- [x]`) syntax for tracking. Mark `- [x]` → `- [x]` as work lands.

**Goal:** A first-class TipTap `DateTime` block carrying an ISO timestamp, its original time zone, and a display format. Renders in the viewer's local zone with a small badge showing the original tz. Inline picker for date / time / tz. Slash command `/datetime`. Markdown export emits semantic `<time datetime="…">…</time>` HTML. Search-index extractor parses the ISO so date filters added by P29 can match.

**Architecture:** New runtime dep `luxon` (handles every cross-tz / DST edge). New TipTap node `DateTime` at `src/components/editor/blocks/datetime-node.ts` carrying `iso` (UTC instant string), `tz` (IANA zone), `display_format` (Luxon token string, default `'yyyy-LL-dd HH:mm'`). Renderer `datetime.tsx` parses the ISO into a Luxon `DateTime`, converts to the viewer's local zone, and renders the formatted string + an `<aria-label>` showing the original tz. Markdown export visits `datetime` nodes and emits the `<time>` element. Search indexer extracts the ISO into a side-channel field stored in `pages.search_metadata` jsonb (already exists) so P29 can range-filter on it.

**Tech Stack:** `luxon@^3` (new dep), TipTap 3, Next.js 16, Vitest 4 + jsdom, existing search indexer at `src/lib/pages/search-indexer.ts`, existing markdown exporter at `src/lib/pages/export-markdown.ts`.

- [x] **Step 5: Implement `src/lib/datetime/format.ts`**

```ts
import { DateTime } from 'luxon';

export const DEFAULT_DISPLAY_FORMAT = 'yyyy-LL-dd HH:mm';

/**
 * Render an ISO instant in the viewer's local tz using the supplied Luxon
 * format token string. `originalTz` is included only for context (it appears
 * in the badge — not in this function's output). Falls back to UTC if the
 * viewer's tz string is unparseable.
 */
export function formatForViewer(
  iso: string,
  _originalTz: string,
  displayFormat: string,
  viewerTz: string,
): string {
  let dt = DateTime.fromISO(iso, { zone: 'utc' });
  if (!dt.isValid) return iso;
  let zoned = dt.setZone(viewerTz);
  if (!zoned.isValid) zoned = dt.setZone('utc');
  return zoned.toFormat(displayFormat);
}

/**
 * Build a UTC-instant ISO string from a date+time-in-tz triple. Luxon resolves
 * DST anomalies as follows:
 *   - non-existent (spring-forward gap): the wall-clock time is shifted forward
 *     into the post-jump offset.
 *   - ambiguous (fall-back overlap): the earlier instant is chosen.
 */
export function parseInput(input: { date: string; time: string; tz: string }): string {
  const dt = DateTime.fromISO(`${input.date}T${input.time}`, { zone: input.tz });
  if (!dt.isValid) {
    throw new Error(`invalid datetime input: ${input.date}T${input.time}@${input.tz}: ${dt.invalidReason}`);
  }
  return dt.toUTC().toISO({ suppressMilliseconds: false }) ?? `${input.date}T${input.time}:00.000Z`;
}
```

import { Node, mergeAttributes } from '@tiptap/core';
import { DEFAULT_DISPLAY_FORMAT } from '@/lib/datetime/format';

export const DateTimeNode = Node.create({
  name: 'datetime',
  group: 'inline',
  inline: true,
  atom: true,
  selectable: true,

  addAttributes() {
    return {
      iso: { default: '' },
      tz: { default: 'UTC' },
      display_format: { default: DEFAULT_DISPLAY_FORMAT },
    };
  },

  parseHTML() {
    return [{ tag: 'time[datetime]' }];

  renderHTML({ HTMLAttributes }) {
    return [
      'time',
      mergeAttributes(HTMLAttributes, {
        datetime: HTMLAttributes.iso ?? '',
        'data-tz': HTMLAttributes.tz ?? 'UTC',
        'data-format': HTMLAttributes.display_format ?? DEFAULT_DISPLAY_FORMAT,
        class: 'cairn-datetime',
      }),
    ];
  },
});

```tsx
'use client';

import { ReactNodeViewRenderer } from '@tiptap/react';
import { Info } from 'luxon';
import { useMemo, useState } from 'react';
import { Popover, PopoverContent, PopoverTrigger } from '@/components/ui/popover';
import { DateTimeNode } from '@/components/editor/blocks/datetime-node';
import { formatForViewer, parseInput, DEFAULT_DISPLAY_FORMAT } from '@/lib/datetime/format';

export function DateTimeView({
  node,
  updateAttributes,
  viewerTz,
}: {
  node: { attrs: { iso: string; tz: string; display_format: string } };
  updateAttributes?: (attrs: Partial<{ iso: string; tz: string; display_format: string }>) => void;
  viewerTz?: string;
}): JSX.Element {
  const tz = viewerTz ?? Intl.DateTimeFormat().resolvedOptions().timeZone ?? 'UTC';
  const formatted = useMemo(
    () => formatForViewer(node.attrs.iso, node.attrs.tz, node.attrs.display_format || DEFAULT_DISPLAY_FORMAT, tz),
    [node.attrs.iso, node.attrs.tz, node.attrs.display_format, tz],
  );
  const [date, setDate] = useState(node.attrs.iso.slice(0, 10));
  const [time, setTime] = useState(node.attrs.iso.slice(11, 16));
  const [zone, setZone] = useState(node.attrs.tz);
  const zones = useMemo(() => Info.normalizeZone('UTC').isValid ? Intl.supportedValuesOf('timeZone') : ['UTC'], []);

  function commit(): void {
    if (!updateAttributes) return;
    try {
      const iso = parseInput({ date, time, tz: zone });
      updateAttributes({ iso, tz: zone });
    } catch {
      // Invalid input — leave the node unchanged.
    }
  }

  return (
    <Popover>
      <PopoverTrigger asChild>
        <time
          dateTime={node.attrs.iso}
          className="inline-flex items-center gap-1 cursor-pointer underline decoration-dotted"
          aria-label={`${formatted} (original timezone ${node.attrs.tz})`}
        >
          <span>{formatted}</span>
          <span className="text-xs text-muted-foreground">({node.attrs.tz})</span>
        </time>
      </PopoverTrigger>
      <PopoverContent className="w-72 space-y-2">
        <label className="block text-xs">
          Date
          <input type="date" value={date} onChange={(e) => setDate(e.target.value)} className="w-full border rounded px-2 py-1" />
        </label>
        <label className="block text-xs">
          Time
          <input type="time" value={time} onChange={(e) => setTime(e.target.value)} className="w-full border rounded px-2 py-1" />
        </label>
        <label className="block text-xs">
          Timezone
          <select value={zone} onChange={(e) => setZone(e.target.value)} className="w-full border rounded px-2 py-1">
            {zones.map((z) => (
              <option key={z} value={z}>
                {z}
              </option>
            ))}
          </select>
        </label>
        <button type="button" onClick={commit} className="w-full px-2 py-1 bg-primary text-primary-foreground rounded">
          Apply
        </button>
      </PopoverContent>
    </Popover>
  );
}

export const DateTimeExtension = DateTimeNode.extend({
  addNodeView() {
    return ReactNodeViewRenderer((props) => <DateTimeView {...props} />);
  },
});
```

- [x] **Step 3: Implement `src/lib/datetime/extract-from-doc.ts`**

```ts
import { DateTime } from 'luxon';

type PMNode = {
  type?: string;
  content?: PMNode[];
  attrs?: Record<string, unknown>;
};

export type ExtractedDateTime = { iso: string; tz: string; epochMs: number };

export function extractDateTimesFromDoc(doc: PMNode): ExtractedDateTime[] {
  const out: ExtractedDateTime[] = [];
  function walk(n: PMNode): void {
    if (n.type === 'datetime') {
      const iso = String(n.attrs?.iso ?? '');
      const tz = String(n.attrs?.tz ?? 'UTC');
      const dt = DateTime.fromISO(iso, { zone: 'utc' });
      if (dt.isValid) {
        out.push({ iso, tz, epochMs: dt.toMillis() });
      }
    }
    for (const c of n.content ?? []) walk(c);
  }
  walk(doc);
  return out;
}
```

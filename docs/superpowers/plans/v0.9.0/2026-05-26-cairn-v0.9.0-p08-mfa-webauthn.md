# Cairn v0.9.0 — Plan P8: MFA migration 0036 + WebAuthn passkeys + step-up + admin-enforce

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. **Plan-review subagent runs before implementer dispatch (v0.7-v0.8 retrospective §2).** Spec ref: `docs/superpowers/specs/2026-05-26-cairn-v0.9.0-design.md` §2 G1 #20. Branch: `release/v0.9.0` — every commit lands on the branch, NOT `main`.

**Goal:** Add WebAuthn passkeys alongside the existing v0.6 P19 TOTP. Passkeys are a complement, not a replacement — users can enroll either, both, or neither (subject to admin policy). Three orthogonal feature surfaces ship together:

1. **Enrollment** — user-facing flow at `/settings/security/passkeys` to register and revoke FIDO2 credentials.
2. **Step-up** — middleware that, for sensitive ops (workspace delete, admin role grant — extensible list), requires a WebAuthn assertion within the last 5 minutes. Missing/stale → 403 `stepup-required`; the client triggers an inline assertion and retries.
3. **Admin enforce** — workspace-admin-configurable policy: if `workspace_mfa_policies.require_mfa = true`, sign-in for members without any enrolled MFA method (TOTP or WebAuthn) returns 403 `mfa-enrollment-required` with a link to the enrollment page.

**Architecture:**
- Migration `0036_mfa_webauthn.sql` adds two tables:
  - `user_webauthn_credentials` — one row per registered passkey. Stores `credential_id` (unique), `public_key`, `sign_count`, `transports text[]`, `aaguid uuid`, `nickname`, timestamps.
  - `workspace_mfa_policies` — one row per workspace. Columns: `require_mfa boolean`, `methods text[]` (any of `'totp'`, `'webauthn'`).
- `@simplewebauthn/server` (server-side ceremony helpers) + `@simplewebauthn/browser` (client-side `startRegistration` / `startAuthentication`).
- New env vars: `CAIRN_RP_ID`, `CAIRN_RP_NAME` (default `Cairn`), `CAIRN_RP_ORIGIN`. Documented in `.env.example` + `docs/operations.md`. **Startup-time sanity check** in `src/server/entrypoint.ts` verifies `CAIRN_RP_ORIGIN` matches `NEXTAUTH_URL`'s origin; logs a loud warning on mismatch (mismatch causes permanent credential invalidation — retrospective §3 risk #5).
- Step-up: a recent-assertion record lives in the Auth.js JWT (`stepUpAt: number`). Middleware checks `Date.now() - stepUpAt < 5 * 60_000`.
- Audit actions: `mfa.passkey_added`, `mfa.passkey_removed`, `mfa.passkey_used`, `mfa.stepup_required`, `mfa.policy_changed`.

**Tech Stack:** `@simplewebauthn/server@^11` + `@simplewebauthn/browser@^11` (current major as of 2026-05; verify on install). Auth.js v5 JWT extension for `stepUpAt`. Drizzle migration `0036` (number aspirational — accept emission).

```ts
  dataType() { return 'bytea'; },
});

export const userWebauthnCredentials = pgTable(
  'user_webauthn_credentials',
  {

    userId: uuid('user_id').notNull().references(() => users.id, { onDelete: 'cascade' }),

    publicKey: bytea('public_key').notNull(),
    // Anti-cloning monotonic counter; we update on every successful assertion.
    signCount: bigint('sign_count', { mode: 'number' }).notNull().default(0),
    transports: text('transports').array(),
    aaguid: uuid('aaguid'),
    nickname: text('nickname'),
    createdAt: timestamp('created_at', { withTimezone: true }).notNull().defaultNow(),
    lastUsedAt: timestamp('last_used_at', { withTimezone: true }),
  },
  (t) => ({
    byUser: index('user_webauthn_credentials_user_idx').on(t.userId),

);
```

```ts
import {
  generateAuthenticationOptions,
  generateRegistrationOptions,
  verifyAuthenticationResponse,
  verifyRegistrationResponse,
} from '@simplewebauthn/server';
import type {
  AuthenticationResponseJSON,
  PublicKeyCredentialCreationOptionsJSON,
  PublicKeyCredentialRequestOptionsJSON,
  RegistrationResponseJSON,
} from '@simplewebauthn/server';
import { eq } from 'drizzle-orm';
import { db } from '@/db/client';
import { userWebauthnCredentials } from '@/db/schema';
import { env } from '@/lib/env';

export type BeginRegistrationInput = {
  userId: string;
  userName: string;
  userDisplayName: string;
};

export type BeginRegistrationOutput = {
  options: PublicKeyCredentialCreationOptionsJSON;
  expectedChallenge: string;
};

export async function beginRegistration(
  input: BeginRegistrationInput,
): Promise<BeginRegistrationOutput> {
  const e = env();
  // Exclude already-registered credentials to prevent dupes for this user.
  const existing = await db
    .select({ credentialId: userWebauthnCredentials.credentialId, transports: userWebauthnCredentials.transports })
    .from(userWebauthnCredentials)
    .where(eq(userWebauthnCredentials.userId, input.userId));
  const options = await generateRegistrationOptions({
    rpID: e.CAIRN_RP_ID,
    rpName: e.CAIRN_RP_NAME,
    userID: new TextEncoder().encode(input.userId),
    userName: input.userName,
    userDisplayName: input.userDisplayName,
    excludeCredentials: existing.map((c) => ({
      id: c.credentialId,
      transports: (c.transports ?? undefined) as AuthenticatorTransport[] | undefined,
    })),
    authenticatorSelection: { residentKey: 'preferred', userVerification: 'preferred' },
    attestationType: 'none',
  });
  return { options, expectedChallenge: options.challenge };
}

export async function finishRegistration(input: {
  userId: string;
  response: RegistrationResponseJSON;
  expectedChallenge: string;
  nickname: string | null;
}): Promise<{ ok: true } | { ok: false; error: string }> {
  const e = env();
  const verification = await verifyRegistrationResponse({
    response: input.response,
    expectedChallenge: input.expectedChallenge,
    expectedOrigin: e.CAIRN_RP_ORIGIN,
    expectedRPID: e.CAIRN_RP_ID,
    requireUserVerification: false,
  });
  if (!verification.verified || !verification.registrationInfo) {
    return { ok: false, error: 'verification failed' };
  }
  const info = verification.registrationInfo;
  await db.insert(userWebauthnCredentials).values({
    userId: input.userId,
    credentialId: info.credential.id,
    publicKey: Buffer.from(info.credential.publicKey),
    signCount: info.credential.counter,
    transports: info.credential.transports ?? null,
    aaguid: info.aaguid && info.aaguid !== '00000000-0000-0000-0000-000000000000' ? info.aaguid : null,
    nickname: input.nickname,
  });
  return { ok: true };
}

export type BeginAssertionInput = { userId: string };
export type BeginAssertionOutput = {
  options: PublicKeyCredentialRequestOptionsJSON;
  expectedChallenge: string;
};

export async function beginAssertion(input: BeginAssertionInput): Promise<BeginAssertionOutput> {
  const e = env();
  const allowList = await db
    .select({ credentialId: userWebauthnCredentials.credentialId, transports: userWebauthnCredentials.transports })
    .from(userWebauthnCredentials)
    .where(eq(userWebauthnCredentials.userId, input.userId));
  const options = await generateAuthenticationOptions({
    rpID: e.CAIRN_RP_ID,
    allowCredentials: allowList.map((c) => ({
      id: c.credentialId,
      transports: (c.transports ?? undefined) as AuthenticatorTransport[] | undefined,
    })),
    userVerification: 'preferred',
  });
  return { options, expectedChallenge: options.challenge };
}

export async function finishAssertion(input: {
  userId: string;
  response: AuthenticationResponseJSON;
  expectedChallenge: string;
}): Promise<{ ok: true; credentialId: string } | { ok: false; error: string }> {
  const e = env();
  const [cred] = await db
    .select()
    .from(userWebauthnCredentials)
    .where(eq(userWebauthnCredentials.credentialId, input.response.id));
  if (!cred || cred.userId !== input.userId) return { ok: false, error: 'unknown credential' };

  const verification = await verifyAuthenticationResponse({
    response: input.response,
    expectedChallenge: input.expectedChallenge,
    expectedOrigin: e.CAIRN_RP_ORIGIN,
    expectedRPID: e.CAIRN_RP_ID,
    credential: {
      id: cred.credentialId,
      publicKey: cred.publicKey,
      counter: Number(cred.signCount),
      transports: (cred.transports ?? undefined) as AuthenticatorTransport[] | undefined,
    },
    requireUserVerification: false,
  });
  if (!verification.verified) return { ok: false, error: 'verification failed' };

  await db
    .update(userWebauthnCredentials)
    .set({
      signCount: verification.authenticationInfo.newCounter,
      lastUsedAt: new Date(),
    })
    .where(eq(userWebauthnCredentials.credentialId, cred.credentialId));
  return { ok: true, credentialId: cred.credentialId };
}
```

```ts
const Body = z.object({ response: z.any(), nickname: z.string().max(120).nullable().optional() });

export async function POST(req: Request): Promise<Response> {

    targetId: null,
    metadata: { nickname: parsed.data.nickname ?? null },
  });

  res.cookies.delete('cairn_wac');
```

```ts
import { z } from 'zod';
import { db } from '@/db/client';
import { workspaceMfaPolicies } from '@/db/schema';
import { recordAudit } from '@/lib/audit/record';
import { requireRole } from '@/lib/auth/require-role';

const Body = z.object({
  requireMfa: z.boolean(),
  methods: z.array(z.enum(['totp', 'webauthn'])).min(1),
});

export async function PUT(
  req: Request,
  { params }: { params: Promise<{ workspaceId: string }> },
): Promise<Response> {
  const { workspaceId } = await params;
  const gate = await requireRole(workspaceId, 'admin');
  if (gate instanceof NextResponse) return gate;

    await tx
      .insert(workspaceMfaPolicies)
      .values({
        workspaceId,
        requireMfa: parsed.data.requireMfa,
        methods: parsed.data.methods,
      })
      .onConflictDoUpdate({
        target: workspaceMfaPolicies.workspaceId,
        set: {
          requireMfa: parsed.data.requireMfa,
          methods: parsed.data.methods,
          updatedAt: sql`now()`,
        },

      workspaceId,
      actorUserId: gate.userId,
      action: 'mfa.policy_changed',
      targetType: 'mfa_policy',
      targetId: workspaceId,
```

```ts
export const STEPUP_TTL_MS = 5 * 60 * 1000;

export type StepUpResult =
  | { ok: true }
  | { ok: false; status: 403; code: 'stepup-required'; message: string };

export function requireStepUp(input: { stepUpAt: number | undefined | null }): StepUpResult {
  if (!input.stepUpAt) {
    return { ok: false, status: 403, code: 'stepup-required', message: 'WebAuthn step-up required' };
  }
  if (Date.now() - input.stepUpAt > STEPUP_TTL_MS) {
    return { ok: false, status: 403, code: 'stepup-required', message: 'WebAuthn step-up expired' };
  }
  return { ok: true };
}
```

```ts
import { and, eq, inArray } from 'drizzle-orm';
import { db } from '@/db/client';
import { userTotp, userWebauthnCredentials, workspaceMembers, workspaceMfaPolicies } from '@/db/schema';

export type MfaCheckResult =
  | { ok: true }
  | { ok: false; code: 'mfa-enrollment-required'; status: 403; workspaceIds: string[] };

export async function checkMfaEnrollmentForSignIn(input: { userId: string }): Promise<MfaCheckResult> {
  // Workspaces this user belongs to.
  const memberWorkspaces = await db
    .select({ workspaceId: workspaceMembers.workspaceId })
    .from(workspaceMembers)
    .where(eq(workspaceMembers.userId, input.userId));
  if (memberWorkspaces.length === 0) return { ok: true };

  const ids = memberWorkspaces.map((r) => r.workspaceId);
  const policies = await db
    .select()
    .from(workspaceMfaPolicies)
    .where(and(inArray(workspaceMfaPolicies.workspaceId, ids), eq(workspaceMfaPolicies.requireMfa, true)));
  if (policies.length === 0) return { ok: true };

  // Enrolled methods.
  const [totpRow] = await db.select().from(userTotp).where(eq(userTotp.userId, input.userId));
  const enrolledTotp = totpRow?.confirmedAt != null;
  const [webauthnRow] = await db
    .select()
    .from(userWebauthnCredentials)
    .where(eq(userWebauthnCredentials.userId, input.userId))
    .limit(1);
  const enrolledWebauthn = !!webauthnRow;

  // The user satisfies a policy if they have at least one of its allowed methods enrolled.
  const failing = policies.filter((p) => {
    const allowed = new Set(p.methods);
    return !(allowed.has('totp') && enrolledTotp) && !(allowed.has('webauthn') && enrolledWebauthn);
  });
  if (failing.length === 0) return { ok: true };
  return {
    ok: false,
    code: 'mfa-enrollment-required',
    status: 403,
    workspaceIds: failing.map((p) => p.workspaceId),
  };
}
```

```tsx
export function StepUpModal({ onComplete, onCancel }: { onComplete: () => void; onCancel: () => void }) {
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState<string | null>(null);

  async function run() {
    setBusy(true); setErr(null);
    try {
      const optsRes = await fetch('/api/webauthn/assert-options', { method: 'POST' });
      if (!optsRes.ok) throw new Error('options fetch failed');
      const { options } = await optsRes.json();
      const response = await startAuthentication({ optionsJSON: options });
      const r = await fetch('/api/webauthn/assert', {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ response }),
      });
      if (!r.ok) throw new Error(`assert failed: ${r.status}`);
      onComplete();
    } catch (e) {
      setErr((e as Error).message);
    } finally {
      setBusy(false);
    }
  }

  return (
    <div role="dialog" aria-modal="true" className="fixed inset-0 bg-black/50 flex items-center justify-center">
      <div className="bg-background p-6 rounded shadow-md w-96 space-y-3">
        <h2 className="text-lg font-semibold">Confirm with your passkey</h2>
        <p className="text-sm text-muted-foreground">
          This action requires a recent passkey assertion (within 5 minutes).
        </p>
        <div className="flex gap-2 justify-end">
          <Button variant="ghost" onClick={onCancel} disabled={busy}>Cancel</Button>
          <Button onClick={run} disabled={busy}>{busy ? 'Verifying…' : 'Use passkey'}</Button>
        </div>
        {err ? <p className="text-sm text-destructive">{err}</p> : null}
      </div>
    </div>
  );
}
```

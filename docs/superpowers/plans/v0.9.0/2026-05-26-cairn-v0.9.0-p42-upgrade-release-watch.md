# Cairn v0.9.0 — Plan P42: Release-watch daemon + admin upgrade UI

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. **Every commit lands on `release/v0.9.0` — never on `main`** (retrospective §1).

**Goal:** Surface new Cairn releases to admins without forcing them to poll GitHub. An inline cron tick (uses v0.7 P14's `cron_schedules` infrastructure) polls a release feed on a configurable interval, semver-compares the latest tag against the running version, and fans out a notification (uses v0.8 P15's bell + drawer + `/notifications` inbox) to every admin user across every workspace. A new admin page at `/admin/upgrade` renders current vs available version + release-notes link + "Apply upgrade now" button. The button POSTs to `/api/admin/upgrade/apply` which invokes P41's `applyUpgrade` from an internal worker; the route streams the worker's stdout via SSE so the admin can watch the upgrade in real time. **Auto-apply remains OFF by default** — the button is the only path that triggers an actual upgrade.

**Architecture:** Cron tick lives in `src/lib/upgrade/release-watch.ts` and registers with v0.7 P14's scheduler at `src/lib/cron/schedules.ts` (the same place backup/digest/reminder ticks register). On each tick: fetch `CAIRN_RELEASE_FEED_URL` (default `https://api.github.com/repos/jonathanmcohen/cairn/releases`), pull the highest semver tag, compare to `require('../../package.json').version` baked into the runtime image at build time. If `latest > current`, insert a notification row per admin user not already notified for that version. UI lives at `src/app/admin/upgrade/page.tsx` (server component) + `<UpgradeApplyButton>` (client component). SSE stream is a Next 16 route handler returning a `ReadableStream` of `data: {...}\n\n` events as P41's `applyUpgrade` runs (the function gets re-shaped to accept an `onProgress` callback in Task 4).

**Tech Stack:** Node 22, Next.js 16 RSC + Client Components, `semver` (already a dep via Auth.js v5; verify in Task 1), v0.7 P14 cron scheduler, v0.8 P15 notification infrastructure (no schema change — notifications.kind union extended), Vitest + Testcontainers + `nock` for HTTP mocks.

**Spec:** `docs/superpowers/specs/2026-05-26-cairn-v0.9.0-design.md` §3 G8 P42 + §5.2 new env vars (`CAIRN_RELEASE_FEED_URL` default `https://api.github.com/repos/jonathanmcohen/cairn/releases`; `CAIRN_RELEASE_WATCH_INTERVAL` default `24h`; `CAIRN_RELEASE_WATCH_ENABLED` default `true`) + extends the notification.kind union with `'upgrade_available'`.

```ts
import semver from 'semver';

export type ReleaseFeedResult =
  | { ok: true; latestTag: string; releaseNotesUrl: string }
  | { ok: false; reason: string };

export type GitHubRelease = {
  tag_name: string;
  html_url: string;
  draft: boolean;
  prerelease: boolean;
};

export async function fetchReleaseFeed(input: {
  url: string;
  fetchImpl?: typeof fetch;
}): Promise<ReleaseFeedResult> {
  const f = input.fetchImpl ?? fetch;
  let res: Response;
  try {
    res = await f(input.url, {
      headers: { Accept: 'application/vnd.github+json', 'User-Agent': 'cairn-release-watch' },
    });
  } catch (err) {
    return { ok: false, reason: `network: ${(err as Error).message}` };
  }
  if (res.status === 403 && res.headers.get('X-RateLimit-Remaining') === '0') {
    return { ok: false, reason: 'GitHub API rate limit exceeded' };
  }
  if (!res.ok) return { ok: false, reason: `HTTP ${res.status}` };

  let body: GitHubRelease[];
  try {
    body = (await res.json()) as GitHubRelease[];
  } catch {
    return { ok: false, reason: 'feed body not JSON' };
  }
  if (!Array.isArray(body)) return { ok: false, reason: 'feed body not an array' };

  const tags = body
    .filter((r) => !r.draft && !r.prerelease)
    .map((r) => ({ tag: stripV(r.tag_name), url: r.html_url }))
    .filter((r) => semver.valid(r.tag));

  if (tags.length === 0) return { ok: false, reason: 'no stable tags in feed' };

  const sorted = tags.sort((a, b) => semver.rcompare(a.tag, b.tag));
  return { ok: true, latestTag: sorted[0].tag, releaseNotesUrl: sorted[0].url };
}

export function compareVersions(input: { current: string; latest: string }): { isNewer: boolean } {
  const c = stripV(input.current);
  const l = stripV(input.latest);
  if (!semver.valid(c) || !semver.valid(l)) return { isNewer: false };
  return { isNewer: semver.gt(l, c) };
}

function stripV(t: string): string {
  return t.startsWith('v') ? t.slice(1) : t;
}
```

```ts
import { sql, and, eq } from 'drizzle-orm';
import { getDbFromEnv } from '@/db/client';
import { notifications, workspaceMembers } from '@/db/schema';
import { compareVersions } from './feed';

export type ReleaseFeedFetch = () => Promise<
  | { ok: true; latestTag: string; releaseNotesUrl: string }
  | { ok: false; reason: string }
>;

export type ReleaseWatchTickInput = {
  currentVersion: string;
  fetchFeed: ReleaseFeedFetch;
  db?: ReturnType<typeof getDbFromEnv>;
};

export type ReleaseWatchTickResult = {
  notificationsCreated: number;
  latestTag?: string;
  feedError?: string;
};

export async function runReleaseWatchTick(input: ReleaseWatchTickInput): Promise<ReleaseWatchTickResult> {
  const feed = await input.fetchFeed();
  if (!feed.ok) return { notificationsCreated: 0, feedError: feed.reason };

  const cmp = compareVersions({ current: input.currentVersion, latest: feed.latestTag });
  if (!cmp.isNewer) return { notificationsCreated: 0, latestTag: feed.latestTag };

  const db = input.db ?? getDbFromEnv();

  // Find every admin/owner user across every workspace who has not yet been
  // notified for this version. Use a single UPSERT-friendly insert with an
  // anti-join.
  const result = (await db.execute(sql`
    INSERT INTO notifications (recipient_user_id, kind, data, created_at)
    SELECT DISTINCT wm.user_id,
           'upgrade_available',
           ${sql.raw(`'${JSON.stringify({
             version: feed.latestTag,
             releaseNotesUrl: feed.releaseNotesUrl,
           }).replace(/'/g, "''")}'::jsonb`)},
           now()
    FROM workspace_members wm
    WHERE wm.role IN ('owner', 'admin')
      AND NOT EXISTS (
        SELECT 1 FROM notifications n
        WHERE n.recipient_user_id = wm.user_id
          AND n.kind = 'upgrade_available'
          AND n.data->>'version' = ${feed.latestTag}
      )
    RETURNING id
  `)) as unknown as Array<{ id: string }>;

  return { notificationsCreated: result.length, latestTag: feed.latestTag };
}
```

```ts
import { auth } from '@/lib/auth/config';
import { applyUpgrade, type ProgressEvent } from '@/lib/upgrade/apply';
import { sql } from 'drizzle-orm';
import { getDbFromEnv } from '@/db/client';
import { env } from '@/lib/env';
import pkg from '../../../../../../package.json' assert { type: 'json' };

export async function POST(_req: Request): Promise<Response> {
  const session = await auth();
  if (!session?.user?.id) {
    return new Response(JSON.stringify({ error: 'unauthenticated' }), { status: 401 });
  }

  const db = getDbFromEnv();
  const adminRows = (await db.execute(sql`
    SELECT 1 FROM workspace_members
    WHERE user_id = ${session.user.id} AND role IN ('owner', 'admin')
    LIMIT 1
  `)) as unknown as Array<unknown>;
  if (adminRows.length === 0) {
    return new Response(JSON.stringify({ error: 'forbidden' }), { status: 403 });
  }

  const encoder = new TextEncoder();
  const stream = new ReadableStream<Uint8Array>({
    async start(controller) {
      const push = (e: ProgressEvent) => {
        controller.enqueue(encoder.encode(`data: ${JSON.stringify(e)}\n\n`));
      };
      const result = await applyUpgrade({
        databaseUrl: env().DATABASE_URL,
        backupDir: env().CAIRN_BACKUP_DIR ?? '/data/backups',
        fromVersion: pkg.version,
        toVersion: pkg.version,
        onProgress: push,
      }).catch((err: unknown) => {
        push({ stage: 'failed', message: (err as Error).message });
        return { ok: false };
      });
      push({ stage: result.ok ? 'done' : 'failed' });
      controller.close();
    },
  });

  return new Response(stream, {
    status: 200,
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      Connection: 'keep-alive',
    },
  });
}
```

```tsx
'use client';

import { useState } from 'react';

export function UpgradeApplyButton({ disabled }: { disabled: boolean }) {
  const [running, setRunning] = useState(false);
  const [log, setLog] = useState<string[]>([]);

  async function onClick() {
    setRunning(true);
    setLog([]);
    const res = await fetch('/api/admin/upgrade/apply', { method: 'POST' });
    if (!res.ok || !res.body) {
      setLog((l) => [...l, `error: HTTP ${res.status}`]);
      setRunning(false);
      return;
    }
    const reader = res.body.getReader();
    const decoder = new TextDecoder();
    let buf = '';
    for (;;) {
      const { value, done } = await reader.read();
      if (done) break;
      buf += decoder.decode(value, { stream: true });
      const events = buf.split('\n\n');
      buf = events.pop() ?? '';
      for (const ev of events) {
        const line = ev.replace(/^data:\s*/, '');
        if (line) setLog((l) => [...l, line]);
      }
    }
    setRunning(false);
  }

  return (
    <div>
      <button
        type="button"
        onClick={onClick}
        disabled={disabled || running}
        className="rounded bg-primary px-4 py-2 text-primary-foreground disabled:opacity-50"
      >
        {running ? 'Applying...' : 'Apply upgrade now'}
      </button>
      {log.length > 0 && (
        <pre className="mt-4 max-h-96 overflow-auto rounded bg-muted p-3 text-xs">{log.join('\n')}</pre>
      )}
    </div>
  );
}
```

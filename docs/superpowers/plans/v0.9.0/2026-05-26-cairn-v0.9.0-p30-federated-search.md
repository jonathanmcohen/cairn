# Cairn v0.9.0 — Plan P30: Federated multi-workspace + cross-instance search

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Three layered search scopes on top of the v0.6 P22 / v0.7 P13 search pipeline.

1. **User-membership scope.** Default. Search across every workspace the signed-in user is a member of. Single UNION query.
2. **Admin cross-workspace scope.** When the caller has the admin role on the *current* workspace AND `include_all_workspaces=true` is passed, also search workspaces the admin is NOT a member of. Emits `search.cross_workspace_admin` audit on every such call.
3. **Cross-instance scope.** A `peer_instances` table holds remote Cairn deployments. When at least one peer is `enabled`, the route fans out an HMAC-signed `POST /api/search/federated/peer` request to each peer in parallel, merges results, dedupes by `(peer_name, page_id)`, and returns the unified set. Each peer call is rate-limited 10/min (per peer).

**Architecture:** Local search keeps reusing `searchPages` (v0.6 P22 + v0.7 P13). Federation is a thin wrapper that maps over the workspace scopes, then fans out to peers. Outbound peer requests are signed with `CAIRN_FEDERATION_SHARED_SECRET` (per-peer entry stores the SAME secret hashed — that's how the receiving side knows which peer it is). Encrypted pages (`pages.encrypted = true` once P5 lands) are skipped at the federation layer — they're never returned in federated results regardless of scope.

**Tech Stack:** Drizzle migration `0046_peer_instances.sql`, `node:crypto` for HMAC SHA-256, existing `requireRole`/`recordAudit`, in-memory token-bucket for per-peer rate-limit (reuses any existing rate-limit helper — confirm during dispatch).

## Task 2: HMAC envelope (sign + verify)

**Goal:** `signEnvelope({ q, workspaceScope, ts, nonce }, secret)` returns `{ headers: Record<string, string>, body: string }`. `verifyEnvelope(req, secrets[])` re-derives the HMAC across the same fields, compares with timing-safe compare, and rejects (`{ kind: 'unknown_peer' | 'bad_signature' | 'stale' | 'replay' }`) on any mismatch. Replay window: 5 minutes (`ts` must be within ±300s of wall-clock). Nonces are tracked in an in-memory LRU (per-process, 1024 entries) — peer requests at scale should funnel through one Cairn instance per workspace.

```ts
import { createHmac, timingSafeEqual } from 'node:crypto';

export type EnvelopeInput = {
  q: string;
  workspaceScope: 'all' | string; // 'all' or a specific workspace id
  ts: number;
  nonce: string;
};

export type SignedEnvelope = {
  headers: Record<string, string>;
  body: string;
};

export type VerifyResult =
  | { ok: true; peerName: string; payload: { q: string; workspaceScope: string } }
  | { ok: false; kind: 'unknown_peer' | 'bad_signature' | 'stale' | 'replay' | 'malformed' };

const SKEW_MS = 300_000; // ±5 min
const NONCE_LRU = new Map<string, number>(); // nonce → expiresAt
const NONCE_LRU_CAP = 1024;

function rememberNonce(nonce: string, ts: number): boolean {
  const now = Date.now();
  // Evict expired.
  for (const [k, exp] of NONCE_LRU) {
    if (exp < now) NONCE_LRU.delete(k);
  }
  if (NONCE_LRU.has(nonce)) return false;
  if (NONCE_LRU.size >= NONCE_LRU_CAP) {
    const first = NONCE_LRU.keys().next().value;
    if (first !== undefined) NONCE_LRU.delete(first);
  }
  NONCE_LRU.set(nonce, ts + SKEW_MS);
  return true;
}

/** Test-only: reset the nonce LRU between tests. */
export function __resetNonceLruForTests(): void {
  NONCE_LRU.clear();
}

function canonical(input: EnvelopeInput): string {
  // Concatenation order is part of the protocol; do not reorder.
  return `${input.ts}\n${input.nonce}\n${input.workspaceScope}\n${input.q}`;
}

function hmacHex(secret: string, message: string): string {
  return createHmac('sha256', secret).update(message).digest('hex');
}

export function signEnvelope(input: EnvelopeInput, secret: string): SignedEnvelope {
  const body = JSON.stringify({ q: input.q, workspaceScope: input.workspaceScope });
  const sig = hmacHex(secret, canonical(input));
  return {
    headers: {
      'content-type': 'application/json',
      'x-cairn-peer-ts': String(input.ts),
      'x-cairn-peer-nonce': input.nonce,
      'x-cairn-peer-sig': sig,
    },
    body,
  };
}

function safeEqHex(a: string, b: string): boolean {
  if (a.length !== b.length) return false;
  return timingSafeEqual(Buffer.from(a, 'hex'), Buffer.from(b, 'hex'));
}

/**
 * Verify an inbound peer request. `peers` is the known-peer list pulled from
 * the peer_instances table (decrypted + active). Returns `{ ok: true,
 * peerName }` on success — the caller then knows which peer to scope the
 * downstream search to.
 *
 * The implementation deliberately re-reads the body and recomputes the canonical
 * message. Synchronous-only: caller wraps the verify in await for the body
 * string before passing the Request in.
 */
export function verifyEnvelope(
  req: Request,
  peers: ReadonlyArray<{ name: string; secret: string }>,
): VerifyResult {
  const ts = Number(req.headers.get('x-cairn-peer-ts'));
  const nonce = req.headers.get('x-cairn-peer-nonce');
  const sig = req.headers.get('x-cairn-peer-sig');
  if (!Number.isFinite(ts) || !nonce || !sig) return { ok: false, kind: 'malformed' };

  if (Math.abs(Date.now() - ts) > SKEW_MS) return { ok: false, kind: 'stale' };

  // NOTE: the caller must provide a buffered Request (body already read into a
  // string). We re-derive by reading from the cloned request. For simplicity
  // we expose a separate `verifyEnvelopeWithBody` helper for callers that
  // already have the raw body — see export below.
  throw new Error('use verifyEnvelopeWithBody — Request bodies must be buffered first');
}

/** Async-friendly verify that takes the already-read body string. */
export function verifyEnvelopeWithBody(
  headers: Headers,
  rawBody: string,
  peers: ReadonlyArray<{ name: string; secret: string }>,
): VerifyResult {
  const ts = Number(headers.get('x-cairn-peer-ts'));
  const nonce = headers.get('x-cairn-peer-nonce');
  const sig = headers.get('x-cairn-peer-sig');
  if (!Number.isFinite(ts) || !nonce || !sig) return { ok: false, kind: 'malformed' };
  if (Math.abs(Date.now() - ts) > SKEW_MS) return { ok: false, kind: 'stale' };

  let parsed: { q: string; workspaceScope: string };
  try {
    parsed = JSON.parse(rawBody);
  } catch {
    return { ok: false, kind: 'malformed' };
  }
  if (typeof parsed.q !== 'string' || typeof parsed.workspaceScope !== 'string') {
    return { ok: false, kind: 'malformed' };
  }

  const message = canonical({ q: parsed.q, workspaceScope: parsed.workspaceScope, ts, nonce });

  // Try each peer's secret; success on the first match.
  for (const peer of peers) {
    const expected = hmacHex(peer.secret, message);
    if (expected.length === sig.length && safeEqHex(expected, sig)) {
      // Replay check happens AFTER signature match so an attacker can't burn
      // nonces by spamming bad signatures with valid nonces.
      if (!rememberNonce(nonce, ts)) return { ok: false, kind: 'replay' };
      return { ok: true, peerName: peer.name, payload: parsed };
    }
  }
  return { ok: false, kind: peers.length === 0 ? 'unknown_peer' : 'bad_signature' };
}
```

import * as schema from '@/db/schema';
import { recordAudit } from '@/lib/audit/record';
import type { Role } from '@/lib/auth/require-role';
import { type SearchFilters, searchPages, type SearchResult } from '@/lib/pages/search';

export type FederatedInput = {
  userId: string;
  workspaceId: string;
  role: Role;
  query: string;
  filters: SearchFilters;
  includeAllWorkspaces: boolean;
};

export type FederatedResult = {
  local: Array<SearchResult & { workspaceId: string }>;
  /** Reserved — populated by peer-fanout (Task 4). */
  peer: Array<SearchResult & { peerName: string; workspaceId: string }>;
};

/** List workspace ids the user is a member of. */
async function memberWorkspaceIds(
  db: PostgresJsDatabase<typeof schema>,
  userId: string,
): Promise<string[]> {
  const rows = await db
    .select({ id: schema.workspaceMembers.workspaceId })
    .from(schema.workspaceMembers)
    .where(eq(schema.workspaceMembers.userId, userId));
  return rows.map((r) => r.id);
}

async function nonMemberWorkspaceIds(
  db: PostgresJsDatabase<typeof schema>,
  userId: string,
): Promise<string[]> {
  const memberIds = await memberWorkspaceIds(db, userId);
  if (memberIds.length === 0) {
    const all = await db.select({ id: schema.workspaces.id }).from(schema.workspaces);
    return all.map((r) => r.id);
  }
  const rows = await db
    .select({ id: schema.workspaces.id })
    .from(schema.workspaces)
    .where(notInArray(schema.workspaces.id, memberIds));
  return rows.map((r) => r.id);
}

async function searchOneWorkspace(
  db: PostgresJsDatabase<typeof schema>,
  workspaceId: string,
  query: string,
  filters: SearchFilters,
): Promise<Array<SearchResult & { workspaceId: string }>> {
  const results = await searchPages(db, {
    workspaceId,
    query,
    limit: 20,
    filters,
    mode: 'fts',
  });
  // Filter encrypted pages at the federation boundary. Use the runtime guard
  // because `pages.encrypted` only exists post-P5; pre-P5 builds simply pass
  // through with the property undefined.
  const filtered = results.filter((r) => !('encrypted' in r) || (r as { encrypted?: boolean }).encrypted !== true);
  return filtered.map((r) => ({ ...r, workspaceId }));
}

export async function federatedSearch(
  db: PostgresJsDatabase<typeof schema>,
  input: FederatedInput,
): Promise<FederatedResult> {
  const memberIds = await memberWorkspaceIds(db, input.userId);

  // Scope 1: member workspaces.
  const memberHits = (
    await Promise.all(memberIds.map((wid) => searchOneWorkspace(db, wid, input.query, input.filters)))
  ).flat();

  // Scope 2: admin cross-workspace.
  let adminHits: Array<SearchResult & { workspaceId: string }> = [];
  if (input.includeAllWorkspaces && input.role === 'admin') {
    const others = await nonMemberWorkspaceIds(db, input.userId);
    adminHits = (
      await Promise.all(others.map((wid) => searchOneWorkspace(db, wid, input.query, input.filters)))
    ).flat();
    // Audit fires on EVERY cross-workspace admin search call, even if it
    // returned zero hits — the privacy concern is the QUERY, not the results.
    await recordAudit(db, {
      action: 'search.cross_workspace_admin',
      actorId: input.userId,
      workspaceId: input.workspaceId,
      targetType: 'workspace',
      targetId: input.workspaceId,
      details: { query: input.query, scope: 'admin_cross_workspace' },
    });
  }

  return {
    local: [...memberHits, ...adminHits],
    peer: [],
  };
}

```ts
import { randomUUID } from 'node:crypto';
import { eq } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';
import * as schema from '@/db/schema';
import type { SearchResult } from '@/lib/pages/search';
import { signEnvelope } from './peer-hmac';

const RATE_LIMIT = 10; // calls per peer per window
const RATE_WINDOW_MS = 60_000;

type RateBucket = { count: number; windowStart: number };
const buckets = new Map<string, RateBucket>(); // peerId → bucket

/** Test-only: clear rate-limit state between tests. */
export function __resetRateLimitForTests(): void {
  buckets.clear();
}

function allow(peerId: string): boolean {
  const now = Date.now();
  const b = buckets.get(peerId);
  if (!b || now - b.windowStart > RATE_WINDOW_MS) {
    buckets.set(peerId, { count: 1, windowStart: now });
    return true;
  }
  if (b.count >= RATE_LIMIT) return false;
  b.count += 1;
  return true;
}

export type PeerHit = SearchResult & { peerName: string; workspaceId: string };

export type FanOutInput = {
  workspaceId: string;
  query: string;
  sharedSecret: string;
  fetchImpl?: typeof fetch;
};

/**
 * Fan-out to every enabled peer for the workspace. Returns the merged +
 * deduped result set. Peer errors are recorded in `peer_instances.last_error`
 * but do not abort the fan-out for other peers.
 *
 * Dedup key: `(peerName, pageId)`. Within one peer's response, the first
 * occurrence wins.
 */
export async function fanOutToPeers(
  db: PostgresJsDatabase<typeof schema>,
  input: FanOutInput,
): Promise<PeerHit[]> {
  if (!input.sharedSecret) return [];
  const fetchImpl = input.fetchImpl ?? fetch;
  const peers = await db
    .select()
    .from(schema.peerInstances)
    .where(eq(schema.peerInstances.workspaceId, input.workspaceId));
  const enabled = peers.filter((p) => p.enabled);

  const callOne = async (peer: typeof enabled[number]): Promise<PeerHit[]> => {
    if (!allow(peer.id)) return [];
    const ts = Date.now();
    const nonce = randomUUID();
    const signed = signEnvelope(
      { q: input.query, workspaceScope: 'all', ts, nonce },
      input.sharedSecret,
    );
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 5_000);
    try {
      const res = await fetchImpl(`${peer.baseUrl.replace(/\/$/, '')}/api/search/federated/peer`, {
        method: 'POST',
        headers: signed.headers,
        body: signed.body,
        signal: controller.signal,
      });
      if (!res.ok) {
        await db
          .update(schema.peerInstances)
          .set({ lastError: `HTTP ${res.status}` })
          .where(eq(schema.peerInstances.id, peer.id));
        return [];
      }
      const json = (await res.json()) as { results?: SearchResult[] };
      const seen = new Set<string>();
      const out: PeerHit[] = [];
      for (const r of json.results ?? []) {
        if (seen.has(r.id)) continue;
        seen.add(r.id);
        out.push({ ...r, peerName: peer.name, workspaceId: input.workspaceId });
      }
      await db
        .update(schema.peerInstances)
        .set({ lastSyncedAt: new Date(), lastError: null })
        .where(eq(schema.peerInstances.id, peer.id));
      return out;
    } catch (err) {
      const message = err instanceof Error ? err.message : 'fan-out error';
      await db
        .update(schema.peerInstances)
        .set({ lastError: message })
        .where(eq(schema.peerInstances.id, peer.id));
      return [];
    } finally {
      clearTimeout(timeout);
    }
  };

  const all = await Promise.all(enabled.map(callOne));
  return all.flat();
}
```

import { NextResponse } from 'next/server';
import { getDb } from '@/db/client';
import * as schema from '@/db/schema';
import { searchPages } from '@/lib/pages/search';
import { verifyEnvelopeWithBody } from '@/lib/search/peer-hmac';

/**
 * Inbound federation endpoint. NOT user-authenticated — peer-authenticated.
 * The HMAC verify step pulls the secret list from peer_instances; matching
 * peer = caller. Encrypted pages are filtered out at the search layer (see
 * federatedSearch); the same filter applies here.
 */
export async function POST(req: Request): Promise<Response> {
  const rawBody = await req.text();
  const db = getDb();
  // We don't know the workspace at this point — try every enabled peer's
  // secret across every workspace. In a multi-tenant deployment with many
  // peers this is at most N HMACs; in practice N ≤ 10 per workspace.
  const peers = await db
    .select()
    .from(schema.peerInstances);
  const peerList = peers
    .filter((p) => p.enabled)
    .map((p) => ({
      name: p.name,
      secret: p.sharedSecretHash, // stored as raw secret in MVP — see plan note
      workspaceId: p.workspaceId,
    }));
  const verify = verifyEnvelopeWithBody(req.headers, rawBody, peerList);
  if (!verify.ok) return NextResponse.json({ error: verify.kind }, { status: 401 });

  // Identify which workspace this peer belongs to (matched by name).
  const matched = peerList.find((p) => p.name === verify.peerName);
  if (!matched) return NextResponse.json({ error: 'unknown_peer' }, { status: 401 });

  // Run local search scoped to the peer's workspace.
  const results = await searchPages(db, {
    workspaceId: matched.workspaceId,
    query: verify.payload.q,
    limit: 20,
    filters: {},
    mode: 'fts',
  });
  // Strip encrypted pages.
  const filtered = results.filter(
    (r) => !('encrypted' in r) || (r as { encrypted?: boolean }).encrypted !== true,
  );
  return NextResponse.json({ results: filtered });
}

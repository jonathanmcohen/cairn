# Cairn v0.9.0 — Plan P38: OpenAPI generator + `/openapi.json` + Swagger UI at `/api-docs`

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up a self-documenting API surface. Build a generator that walks every route handler under `src/app/api/**`, extracts the Zod schemas they use for request/response validation, and emits OpenAPI 3.1 JSON via `zod-to-openapi`. Serve the document at `/openapi.json` (workspace-member-gated, 1h cache) and render Swagger UI at `/api-docs` (also workspace-member-gated). No public exposure.

**Architecture:** Cairn route handlers parse inputs with `.parse(z.object({...}))` inline today. Migrating every handler to export its schemas is a refactor cost we don't want — instead, the generator works in two passes:

1. **Static manifest:** new `src/lib/openapi/manifest.ts` enumerates every route + its method + a reference to the schema module(s) it uses. The manifest is hand-curated for v0.9 (50–70 entries) but maintained centrally so adding a new route is a one-line append.
2. **Runtime build:** the generator imports each referenced schema module, calls `extendZodWithOpenApi` from `@asteasolutions/zod-to-openapi`, registers each schema in a `OpenAPIRegistry`, then emits the document via `OpenApiGeneratorV31`.

Schema modules add an `.openapi({example, description})` decorator on key fields where it improves the rendered doc — not exhaustive, but enough that Swagger UI is useful out of the box.

**Tech Stack:** `@asteasolutions/zod-to-openapi` (new dep), `swagger-ui-dist` (new dep — vendored static assets, no CDN to keep CSP tight), `@apidevtools/swagger-parser` (dev dep, validates the emitted spec in CI).

**Depends on:** v0.5 P1 + v0.7 PATs (auth shape on existing routes), v0.5 + v0.7 + v0.8 + v0.9 route inventory.

```ts
import { readdir, readFile } from 'node:fs/promises';
import { join } from 'node:path';
import { describe, expect, it } from 'vitest';
import { manifest } from '@/lib/openapi/manifest';

async function walk(dir: string): Promise<string[]> {
  const out: string[] = [];
  for (const entry of await readdir(dir, { withFileTypes: true })) {
    const full = join(dir, entry.name);
    if (entry.isDirectory()) out.push(...(await walk(full)));
    else if (entry.name === 'route.ts') out.push(full);
  }
  return out;
}

describe('openapi manifest coverage', () => {
  it('lists every route handler under src/app/api/**', async () => {
    const files = await walk('src/app/api');
    const found = new Set<string>();
    for (const file of files) {
      const src = await readFile(file, 'utf8');
      const path = `/${file.replace(/^src\/app\//, '').replace(/\/route\.ts$/, '')}`;
      for (const m of ['GET', 'POST', 'PUT', 'DELETE', 'PATCH'] as const) {
        if (new RegExp(`export\\s+(?:async\\s+)?function\\s+${m}\\b`).test(src)) {
          found.add(`${m} ${path}`);
        }
      }
    }
    const documented = new Set(manifest.map((e) => `${e.method} ${e.path}`));
    const missing = [...found].filter((f) => !documented.has(f));
    expect(missing, `${missing.length} routes missing from manifest`).toEqual([]);
  });
});
```

- [ ] **Step 2: Implement `src/lib/openapi/registry.ts`**

```ts
import { OpenAPIRegistry } from '@asteasolutions/zod-to-openapi';
import { manifest, type ManifestEntry } from './manifest';
import { z } from './decorators';

export function buildRegistry(): OpenAPIRegistry {
  const registry = new OpenAPIRegistry();

  registry.registerComponent('securitySchemes', 'session', {
    type: 'apiKey',
    in: 'cookie',
    name: 'next-auth.session-token',
  });
  registry.registerComponent('securitySchemes', 'pat', {
    type: 'http',
    scheme: 'bearer',
    bearerFormat: 'PAT',
  });

  for (const entry of manifest) {
    const parameters: Array<Record<string, unknown>> = [];
    for (const p of entry.pathParams ?? []) {
      parameters.push({
        name: p,
        in: 'path',
        required: true,
        schema: { type: 'string' },
      });
    }
    if (entry.querySchema) {
      parameters.push({
        name: 'query',
        in: 'query',
        required: false,
        schema: registry.register(`${entry.method}_${entry.path}_Query`, entry.querySchema),
      });
    }

    const requestBody = entry.requestSchema
      ? {
          content: {
            'application/json': {
              schema: registry.register(`${entry.method}_${entry.path}_Request`, entry.requestSchema),
            },
          },
        }
      : undefined;

    const responses: Record<string, unknown> = {
      '200': {
        description: 'OK',
        content: entry.responseSchema
          ? {
              'application/json': {
                schema: registry.register(`${entry.method}_${entry.path}_Response`, entry.responseSchema),
              },
            }
          : undefined,
      },
      '401': { description: 'Unauthorized' },
      '403': { description: 'Forbidden' },
      '404': { description: 'Not Found' },
    };

    registry.registerPath({
      method: entry.method.toLowerCase() as Lowercase<ManifestEntry['method']>,
      path: entry.path,
      summary: entry.summary,
      tags: entry.tags,
      security: (entry.security ?? ['session']).map((s) => ({ [s]: [] })),
      parameters,
      ...(requestBody ? { request: { body: requestBody } } : {}),
      responses,
    });
  }

  return registry;
}
```

```ts
import { NextResponse } from 'next/server';
import { getAuthContext } from '@/lib/auth/require-role';
import { generateOpenApiDocument } from '@/lib/openapi/generate';

let cached: { json: string; expires: number } | null = null;
const TTL_MS = 60 * 60 * 1000;

export async function GET(): Promise<NextResponse> {
  const ctx = await getAuthContext();
  if (!ctx) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
  if (!ctx.workspaceId) return NextResponse.json({ error: 'no workspace' }, { status: 403 });

  const now = Date.now();
  if (!cached || cached.expires < now) {
    const doc = generateOpenApiDocument();
    cached = { json: JSON.stringify(doc), expires: now + TTL_MS };
  }
  return new NextResponse(cached.json, {
    headers: {
      'content-type': 'application/json; charset=utf-8',
      'cache-control': 'private, max-age=3600',
    },
  });
}
```

```tsx
'use client';
import 'swagger-ui-dist/swagger-ui.css';
import { useEffect, useRef } from 'react';

export default function SwaggerUiClient({ specUrl }: { specUrl: string }): JSX.Element {
  const ref = useRef<HTMLDivElement | null>(null);

  useEffect(() => {
    let cancelled = false;
    (async () => {
      const { default: SwaggerUIBundle } = await import('swagger-ui-dist/swagger-ui-es-bundle.js');
      if (cancelled || !ref.current) return;
      // @ts-expect-error swagger-ui-dist's bundle has no TS typings
      SwaggerUIBundle({
        url: specUrl,
        domNode: ref.current,
        deepLinking: true,
        docExpansion: 'list',
      });
    })();
    return () => {
      cancelled = true;
    };
  }, [specUrl]);

  return <div ref={ref} role="region" aria-label="Swagger UI" />;
}
```

# Cairn v0.9.0 — Plan P14: Page lock migration `0041` + audit + auto-unlock cron

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add per-page locks so a page can be frozen for editing — content, title, icon, cover, comments — until the locker (or an admin) unlocks it, or until an optional `locked_until` expiry passes. Auto-unlock cron sweeps expired locks every 5 minutes. Every lock action is audited.

**Architecture:**
- Migration `0041_page_lock.sql` adds three nullable columns to `pages`: `locked_at timestamptz`, `locked_by uuid REFERENCES users(id) ON DELETE SET NULL`, `locked_until timestamptz`.
- A page is "locked" iff `locked_at IS NOT NULL`. `locked_until` may be NULL (manual unlock only) or a future timestamp (auto-unlock).
- Library `src/lib/pages/lock.ts` exposes `lockPage`, `unlockPage`, `isLocked`. Every mutation pathway for `pages` (content writes, title/icon/cover, comments, page-acl changes if the policy is "lock-followers-only") calls `isLocked(page)` first and refuses with `403 PageLocked` for non-admins and non-locker-self-unlock requests.
- Admins can override the lock; doing so writes a `page.unlock_overridden_by_admin` audit row.
- Cron entry registered at app startup (NOT per workspace — single global row) runs every 5 minutes via the v0.7 P14 scheduler. The CLI subcommand `pages:auto-unlock` executes a single `UPDATE` for rows with `locked_until < now()` and emits `page.auto_unlocked` audit rows.

**Existing surface (read before editing):**
- `src/lib/auth/require-role.ts` — `requireRole`, `hasMinRole`, `getAuthContext`. Use `hasMinRole('admin', ctx.role)` to gate the admin-override path.
- `src/lib/pages/access.ts` — `requirePageAccess(pageId, role)`. P14 does NOT modify this helper; instead, every concrete mutation helper gains an explicit `isLocked` gate.
- `src/lib/pages/update.ts`, `src/lib/pages/content.ts`, `src/lib/pages/comments.ts` (or equivalent) — these are the write paths that need the gate.
- v0.7 P3 `page_acls` table + `requirePageAcl` — separate from this; we only block writes when `locked_at IS NOT NULL`, independent of ACL grants.
- `src/server/scheduler.ts` (v0.7 P14) — singleton scheduler that spawns `node dist/server/cli.js ...` for due rows.

**Spec:** Section 3 G2 P14. Status row: `G2 P14 — Page lock + auto-unlock`.

**Tech Stack:** Drizzle migration + ORM, Vitest + Testcontainers, Next.js 16 server components for the lock banner, shadcn `<Badge>` + `<Tooltip>` + `<Button>`, `cron-parser` for next-run-at math.

```ts
import { eq, sql } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';
import * as schema from '@/db/schema';

export type LockState =
  | { locked: false; lockedBy: null; lockedAt: null; lockedUntil: null }
  | { locked: true; lockedBy: string; lockedAt: Date; lockedUntil: Date | null };

export async function isLocked(
  db: PostgresJsDatabase<typeof schema>,
  pageId: string,
): Promise<LockState> {
  const row = (
    await db
      .select({
        lockedAt: schema.pages.lockedAt,
        lockedBy: schema.pages.lockedBy,
        lockedUntil: schema.pages.lockedUntil,
      })
      .from(schema.pages)
      .where(eq(schema.pages.id, pageId))
      .limit(1)
  )[0];
  if (!row?.lockedAt || !row.lockedBy) {
    return { locked: false, lockedBy: null, lockedAt: null, lockedUntil: null };
  }
  return {
    locked: true,
    lockedBy: row.lockedBy,
    lockedAt: row.lockedAt,
    lockedUntil: row.lockedUntil,
  };
}

export class PageLockedError extends Error {
  readonly code = 'PageLocked';
  readonly status = 403;
  constructor(public state: LockState) {
    super('Page is locked');
  }
}

/**
 * Guard used by every mutation helper. Throws `PageLockedError` unless the
 * caller is the locker OR is performing an admin override.
 */
export async function requireUnlocked(
  db: PostgresJsDatabase<typeof schema>,
  input: { pageId: string; byUserId: string; adminOverride: boolean },
): Promise<void> {
  const state = await isLocked(db, input.pageId);
  if (!state.locked) return;
  if (input.adminOverride) return;
  if (state.lockedBy === input.byUserId) return;
  throw new PageLockedError(state);
}

export async function lockPage(
  db: PostgresJsDatabase<typeof schema>,
  input: { pageId: string; byUserId: string; workspaceId: string; lockedUntil?: Date | null },
): Promise<void> {
  const now = new Date();
  await db
    .update(schema.pages)
    .set({
      lockedAt: now,
      lockedBy: input.byUserId,
      lockedUntil: input.lockedUntil ?? null,
    })
    .where(eq(schema.pages.id, input.pageId));
  await db.execute(sql`
    INSERT INTO audit_log (id, workspace_id, user_id, action, payload, created_at)
    VALUES (
      gen_random_uuid(),
      ${input.workspaceId},
      ${input.byUserId},
      'page.locked',
      ${JSON.stringify({ pageId: input.pageId, lockedUntil: input.lockedUntil ?? null })}::jsonb,
      now()
    )
  `);
}

export async function unlockPage(
  db: PostgresJsDatabase<typeof schema>,
  input: {
    pageId: string;
    byUserId: string;
    workspaceId: string;
    adminOverride: boolean;
  },
): Promise<void> {
  const state = await isLocked(db, input.pageId);
  if (!state.locked) return;
  if (state.lockedBy !== input.byUserId && !input.adminOverride) {
    throw new PageLockedError(state);
  }
  await db
    .update(schema.pages)
    .set({ lockedAt: null, lockedBy: null, lockedUntil: null })
    .where(eq(schema.pages.id, input.pageId));

  const action = input.adminOverride && state.lockedBy !== input.byUserId
    ? 'page.unlock_overridden_by_admin'
    : 'page.unlocked';
  await db.execute(sql`
    INSERT INTO audit_log (id, workspace_id, user_id, action, payload, created_at)
    VALUES (
      gen_random_uuid(),
      ${input.workspaceId},
      ${input.byUserId},
      ${action},
      ${JSON.stringify({ pageId: input.pageId, originalLockerId: state.lockedBy })}::jsonb,
      now()
    )
  `);
}
```

```ts
import { sql } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';
import * as schema from '@/db/schema';

export async function runAutoUnlockSweep(
  db: PostgresJsDatabase<typeof schema>,
): Promise<{ unlockedCount: number }> {
  // Capture the rows about to be unlocked so we can audit each.
  const expired = (await db.execute(sql`
    SELECT id AS page_id, workspace_id, locked_by
    FROM pages
    WHERE locked_at IS NOT NULL
      AND locked_until IS NOT NULL
      AND locked_until < now()
  `)) as unknown as Array<{ page_id: string; workspace_id: string; locked_by: string | null }>;

  if (expired.length === 0) return { unlockedCount: 0 };

  await db.execute(sql`
    UPDATE pages
    SET locked_at = null, locked_by = null, locked_until = null
    WHERE locked_at IS NOT NULL
      AND locked_until IS NOT NULL
      AND locked_until < now()
  `);

  for (const row of expired) {
    await db.execute(sql`
      INSERT INTO audit_log (id, workspace_id, user_id, action, payload, created_at)
      VALUES (
        gen_random_uuid(),
        ${row.workspace_id},
        ${row.locked_by},
        'page.auto_unlocked',
        ${JSON.stringify({ pageId: row.page_id, originalLockerId: row.locked_by })}::jsonb,
        now()
      )
    `);
  }

  return { unlockedCount: expired.length };
}
```

```ts
const PAGE_AUTO_UNLOCK_CRON = '*/5 * * * *';

export async function registerPageAutoUnlockCron(
  db: PostgresJsDatabase<typeof schema>,
): Promise<void> {
  const nextRunAt = CronExpressionParser.parse(PAGE_AUTO_UNLOCK_CRON, { tz: 'UTC' })
    .next()
    .toDate();
  const command = 'pages:auto-unlock';
  const existing = await db
    .select({ id: schema.cronSchedules.id })
    .from(schema.cronSchedules)
    .where(eq(schema.cronSchedules.kind, 'page_auto_unlock'))
    .limit(1);
  if (existing[0]) {
    await db
      .update(schema.cronSchedules)
      .set({ command, cron: PAGE_AUTO_UNLOCK_CRON, enabled: true, nextRunAt })
      .where(eq(schema.cronSchedules.id, existing[0].id));
    return;
  }
  await db.insert(schema.cronSchedules).values({
    workspaceId: null,
    kind: 'page_auto_unlock',
    command,
    cron: PAGE_AUTO_UNLOCK_CRON,
    enabled: true,
    nextRunAt,
  });
}
```

```ts
import { NextResponse } from 'next/server';
import { z } from 'zod';
import { db } from '@/db';
import { requirePageAccess } from '@/lib/pages/access';
import { hasMinRole } from '@/lib/auth/require-role';
import { lockPage, PageLockedError, unlockPage } from '@/lib/pages/lock';

const lockBody = z.object({ lockedUntil: z.string().datetime().nullable().optional() });
const unlockBody = z.object({ adminOverride: z.boolean().default(false) });

export async function POST(req: Request, { params }: { params: Promise<{ id: string }> }) {
  const { id } = await params;
  const { ctx } = await requirePageAccess(id, 'editor');
  const { lockedUntil } = lockBody.parse(await req.json());
  await lockPage(db, {
    pageId: id,
    byUserId: ctx.userId,
    workspaceId: ctx.workspaceId,
    lockedUntil: lockedUntil ? new Date(lockedUntil) : null,
  });
  return NextResponse.json({ ok: true });
}

export async function DELETE(req: Request, { params }: { params: Promise<{ id: string }> }) {
  const { id } = await params;
  const { ctx } = await requirePageAccess(id, 'editor');
  const { adminOverride } = unlockBody.parse(await req.json());
  const effectiveOverride = adminOverride && hasMinRole('admin', ctx.role);
  try {
    await unlockPage(db, {
      pageId: id,
      byUserId: ctx.userId,
      workspaceId: ctx.workspaceId,
      adminOverride: effectiveOverride,
    });
    return NextResponse.json({ ok: true });
  } catch (err) {
    if (err instanceof PageLockedError) {
      return NextResponse.json({ error: 'PageLocked', state: err.state }, { status: 403 });
    }
    throw err;
  }
}
```

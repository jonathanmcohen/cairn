# Cairn v0.9.0 — Plan P41: `cairn-upgrade` CLI (preview / apply / rollback / healthcheck) + compose orchestration

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. **Every commit lands on `release/v0.9.0` — never on `main`** (see retrospective §1).

**Goal:** Ship a single-binary upgrade CLI (`cairn-upgrade`) that operators run from the host shell (or from inside the runtime image) to safely move a Cairn deployment from version N to version N+1. Four subcommands: `preview` (dry-run migrations against a snapshot, exit 0/1), `apply` (pg_dump → migrate → restart → health → auto-rollback on failure), `rollback` (pg_restore from a snapshot), `healthcheck` (`/api/health` 200 + drizzle_migrations table matches the bundled migration list). One orchestration flag (`--via-compose`) wraps the four primitives with `docker compose stop / pg_dump / pull / up / healthcheck`. Audits emit `upgrade.applied` / `upgrade.failed` / `upgrade.rolled_back` (extends the v0.6 P18 audit log). The CLI is consumed by P42's admin UI (`/admin/upgrade`) via an internal worker invoking the same code path — no shell shell-out, no duplicate logic.

**Architecture:** The CLI lives at `scripts/cairn-upgrade.ts` (TypeScript; invoked via `pnpm exec tsx scripts/cairn-upgrade.ts <command>`) AND is bundled into the runtime image at `dist/server/cairn-upgrade.js` (via the existing `pnpm build:entrypoint` tsc invocation — append the CLI to its include list). The CLI imports the production Drizzle migration journal directly from `drizzle/migrations/meta/_journal.json` so the "bundled migration list" is the same list `src/db/migrate.ts` walks. Snapshot files live under `/data/backups/` (same directory the v0.5 P5 backup CLI writes to). Server restart uses the PID file already written by `src/server/entrypoint.ts` (`/data/cairn.pid`); the CLI sends `SIGTERM` and waits for the process supervisor (docker-compose's restart policy, or systemd in bare-metal deploys) to bring a fresh process up. `--via-compose` is the orchestration mode that wraps the primitives with `docker compose` calls on the host — useful for the operator's "one button" experience and for P42's admin UI.

**Tech Stack:** Node 22, TypeScript, `pg` (existing), `child_process.spawn` for `pg_dump` / `pg_restore` / `docker compose`, `commander` (already a dep for other CLIs in `src/server/cli.ts`), Drizzle ORM (read-only — we query `__drizzle_migrations` directly via raw SQL). Vitest + Testcontainers Postgres for tests.

**Spec:** `docs/superpowers/specs/2026-05-26-cairn-v0.9.0-design.md` §3 G8 P41 + §5.2 new env vars (`CAIRN_BACKUP_DIR`, default `/data/backups`) + §5.3 new audit actions (`upgrade.applied` / `upgrade.failed` / `upgrade.rolled_back`).

```ts
import { spawn } from 'node:child_process';
import { createWriteStream, statSync } from 'node:fs';
import { mkdir } from 'node:fs/promises';
import { join } from 'node:path';
import { createGzip } from 'node:zlib';

export type DumpResult = { path: string; bytesWritten: number };

/**
 * pg_dump → gzip → /data/backups/pre-upgrade-<ts>.sql.gz (or {outDir}/...).
 * Uses --clean --if-exists for a self-contained restore script. Streams stdout
 * through gzip so we never hold the whole dump in memory.
 */
export async function dumpDatabase(input: {
  databaseUrl: string;
  outDir: string;
  label?: string;
}): Promise<DumpResult> {
  await mkdir(input.outDir, { recursive: true });
  const ts = new Date().toISOString().replace(/[:.]/g, '-');
  const label = input.label ?? 'pre-upgrade';
  const out = join(input.outDir, `${label}-${ts}.sql.gz`);

  return new Promise<DumpResult>((resolve, reject) => {
    const proc = spawn(
      'pg_dump',
      ['--clean', '--if-exists', '--no-owner', '--no-privileges', input.databaseUrl],
      { stdio: ['ignore', 'pipe', 'pipe'] },
    );
    const gz = createGzip();
    const sink = createWriteStream(out);
    let stderr = '';
    proc.stderr.on('data', (d) => {
      stderr += d.toString();
    });
    proc.stdout.pipe(gz).pipe(sink);
    sink.on('finish', () => {
      if (proc.exitCode !== 0 && proc.exitCode !== null) {
        reject(new Error(`pg_dump exited ${proc.exitCode}: ${stderr}`));
        return;
      }
      resolve({ path: out, bytesWritten: statSync(out).size });
    });
    sink.on('error', reject);
    proc.on('error', reject);
    proc.on('exit', (code) => {
      if (code !== 0) reject(new Error(`pg_dump exited ${code}: ${stderr}`));
    });
  });
}

/**
 * gunzip the dump file and stream it into `psql` against the target database.
 * Restores in-place (the dump uses --clean --if-exists).
 */
export async function restoreDatabase(input: {
  databaseUrl: string;
  dumpPath: string;
}): Promise<void> {
  const { createReadStream } = await import('node:fs');
  const { createGunzip } = await import('node:zlib');
  return new Promise<void>((resolve, reject) => {
    const psql = spawn('psql', ['--single-transaction', input.databaseUrl], {
      stdio: ['pipe', 'pipe', 'pipe'],
    });
    let stderr = '';
    psql.stderr.on('data', (d) => {
      stderr += d.toString();
    });
    createReadStream(input.dumpPath).pipe(createGunzip()).pipe(psql.stdin);
    psql.on('exit', (code) => {
      if (code === 0) resolve();
      else reject(new Error(`psql exited ${code}: ${stderr}`));
    });
    psql.on('error', reject);
  });
}
```

```ts
import { readFile } from 'node:fs/promises';
import { join } from 'node:path';
import { sql } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';

export type JournalEntry = { idx: number; version: string; when: number; tag: string; breakpoints: boolean };
export type Journal = { version: string; dialect: string; entries: JournalEntry[] };

const JOURNAL_PATH = join(process.cwd(), 'drizzle', 'migrations', 'meta', '_journal.json');

export async function loadBundledJournal(): Promise<Journal> {
  const raw = await readFile(JOURNAL_PATH, 'utf8');
  const parsed = JSON.parse(raw) as Journal;
  if (!Array.isArray(parsed.entries)) throw new Error('invalid journal: missing entries[]');
  return parsed;
}

export type CompareResult = {
  applied: JournalEntry[];
  pending: JournalEntry[];
  drifted: boolean;
  driftReason?: string;
};

export async function compareJournalToDb(input: {
  journal: Journal;
  db: PostgresJsDatabase<Record<string, never>>;
}): Promise<CompareResult> {
  const tableExists = (await input.db.execute(sql`
    SELECT EXISTS (
      SELECT 1 FROM information_schema.tables
      WHERE table_name = '__drizzle_migrations' AND table_schema = current_schema()
    ) AS exists
  `)) as unknown as Array<{ exists: boolean }>;

  if (!tableExists[0]?.exists) {
    return { applied: [], pending: [...input.journal.entries], drifted: false };
  }

  const rows = (await input.db.execute(sql`
    SELECT hash, created_at FROM __drizzle_migrations ORDER BY created_at ASC
  `)) as unknown as Array<{ hash: string; created_at: number | string }>;

  const knownTags = new Set(input.journal.entries.map((e) => e.tag));
  const dbHashes = rows.map((r) => r.hash);
  // Drizzle's __drizzle_migrations.hash is a SHA of the migration's SQL; the
  // journal's tag is the migration filename (e.g. "0034_sso_tables"). They
  // are NOT directly comparable. We use the count + ordering as the gate:
  //  - applied count > journal length → drift (DB ahead)
  //  - applied count == 0 → nothing applied yet
  //  - applied count <= journal length → first N journal entries are applied
  const applied = input.journal.entries.slice(0, dbHashes.length);
  const pending = input.journal.entries.slice(dbHashes.length);
  const drifted = dbHashes.length > input.journal.entries.length;
  const driftReason = drifted
    ? `database has ${dbHashes.length} migrations applied but journal lists ${input.journal.entries.length}`
    : undefined;
  // Cross-check: every applied row's hash should at least be a non-empty string.
  if (dbHashes.some((h) => !h || typeof h !== 'string')) {
    return { applied, pending, drifted: true, driftReason: 'unparseable hash in __drizzle_migrations' };
  }
  void knownTags;
  return { applied, pending, drifted, driftReason };
}
```

```ts
import { spawn } from 'node:child_process';
import { mkdtempSync, readFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { randomUUID } from 'node:crypto';
import { migrate } from 'drizzle-orm/postgres-js/migrator';
import { drizzle } from 'drizzle-orm/postgres-js';
import postgres from 'postgres';
import { compareJournalToDb, loadBundledJournal } from './migrations';

export type PreviewResult = {
  ok: boolean;
  pendingTags: string[];
  schemaDiff: string;
  error?: string;
};

/**
 * Read-only dry run. Strategy:
 *  1. Connect to the target databaseUrl read-only; capture pre-upgrade schema
 *     via `pg_dump --schema-only`.
 *  2. Create a scratch database `cairn_preview_<uuid>` on the same Postgres
 *     instance; clone schema via pg_dump | psql.
 *  3. Run pending migrations against the scratch DB.
 *  4. Capture post-upgrade schema; diff line-by-line.
 *  5. Drop the scratch DB.
 *  6. Compare bundled journal against the ORIGINAL DB to list pending tags.
 * Production data is never read into the diff — schema-only.
 */
export async function previewUpgrade(input: { databaseUrl: string }): Promise<PreviewResult> {
  const journal = await loadBundledJournal();
  const targetClient = postgres(input.databaseUrl, { max: 1 });
  const targetDb = drizzle(targetClient);
  const cmp = await compareJournalToDb({ journal, db: targetDb });
  await targetClient.end();

  if (cmp.pending.length === 0) {
    return { ok: true, pendingTags: [], schemaDiff: '' };
  }

  const url = new URL(input.databaseUrl);
  const adminUrl = new URL(input.databaseUrl);
  adminUrl.pathname = '/postgres';
  const scratchName = `cairn_preview_${randomUUID().replace(/-/g, '')}`;

  const admin = postgres(adminUrl.toString(), { max: 1 });
  await admin.unsafe(`CREATE DATABASE "${scratchName}"`);
  await admin.end();

  const tmp = mkdtempSync(join(tmpdir(), 'cairn-preview-'));
  const beforeDump = join(tmp, 'before.sql');
  const afterDump = join(tmp, 'after.sql');

  try {
    await spawnOk('pg_dump', ['--schema-only', input.databaseUrl], { writeTo: beforeDump });

    // Clone schema + data into scratch.
    await pipeDumpInto(input.databaseUrl, withDb(input.databaseUrl, scratchName));

    // Apply pending migrations against scratch.
    const scratchUrl = withDb(input.databaseUrl, scratchName);
    const scratchClient = postgres(scratchUrl, { max: 1 });
    const scratchDb = drizzle(scratchClient);
    await migrate(scratchDb, { migrationsFolder: 'drizzle/migrations' });
    await scratchClient.end();

    await spawnOk('pg_dump', ['--schema-only', scratchUrl], { writeTo: afterDump });

    const before = readFileSync(beforeDump, 'utf8');
    const after = readFileSync(afterDump, 'utf8');
    const diff = unifiedDiff(before, after);

    return { ok: true, pendingTags: cmp.pending.map((e) => e.tag), schemaDiff: diff };
  } catch (err) {
    return { ok: false, pendingTags: cmp.pending.map((e) => e.tag), schemaDiff: '', error: (err as Error).message };
  } finally {
    const cleanup = postgres(adminUrl.toString(), { max: 1 });
    await cleanup.unsafe(`DROP DATABASE IF EXISTS "${scratchName}" WITH (FORCE)`).catch(() => {});
    await cleanup.end();
    void url;
  }
}

function withDb(databaseUrl: string, dbName: string): string {
  const u = new URL(databaseUrl);
  u.pathname = `/${dbName}`;
  return u.toString();
}

async function spawnOk(
  cmd: string,
  args: string[],
  opts?: { writeTo?: string; stdin?: NodeJS.ReadableStream },
): Promise<void> {
  const { createWriteStream } = await import('node:fs');
  return new Promise<void>((resolve, reject) => {
    const proc = spawn(cmd, args, { stdio: ['pipe', opts?.writeTo ? 'pipe' : 'inherit', 'pipe'] });
    let stderr = '';
    proc.stderr.on('data', (d) => {
      stderr += d.toString();
    });
    if (opts?.writeTo) {
      const sink = createWriteStream(opts.writeTo);
      proc.stdout.pipe(sink);
      sink.on('error', reject);
    }
    if (opts?.stdin) opts.stdin.pipe(proc.stdin);
    proc.on('error', reject);
    proc.on('exit', (code) => (code === 0 ? resolve() : reject(new Error(`${cmd} exited ${code}: ${stderr}`))));
  });
}

async function pipeDumpInto(sourceUrl: string, targetUrl: string): Promise<void> {
  return new Promise<void>((resolve, reject) => {
    const dump = spawn('pg_dump', ['--clean', '--if-exists', '--no-owner', sourceUrl], {
      stdio: ['ignore', 'pipe', 'pipe'],
    });
    const psql = spawn('psql', [targetUrl], { stdio: ['pipe', 'inherit', 'pipe'] });
    let stderr = '';
    dump.stderr.on('data', (d) => {
      stderr += d.toString();
    });
    psql.stderr.on('data', (d) => {
      stderr += d.toString();
    });
    dump.stdout.pipe(psql.stdin);
    psql.on('exit', (code) => (code === 0 ? resolve() : reject(new Error(`psql exited ${code}: ${stderr}`))));
    dump.on('error', reject);
    psql.on('error', reject);
  });
}

function unifiedDiff(a: string, b: string): string {
  // Minimal line-diff (we just need to surface "added/removed" lines; we are
  // not trying to be `diff -u`). Skip lines starting with `--` (psql comments)
  // so the output is signal-only.
  const aLines = new Set(a.split('\n').filter((l) => l.trim() && !l.startsWith('--')));
  const bLines = b.split('\n').filter((l) => l.trim() && !l.startsWith('--'));
  const out: string[] = [];
  for (const l of bLines) {
    if (!aLines.has(l)) out.push(`+ ${l}`);
  }
  for (const l of a.split('\n').filter((l) => l.trim() && !l.startsWith('--'))) {
    if (!new Set(bLines).has(l)) out.push(`- ${l}`);
  }
  return out.join('\n');
}
```

```ts
import { drizzle } from 'drizzle-orm/postgres-js';
import { migrate } from 'drizzle-orm/postgres-js/migrator';
import postgres from 'postgres';
import { dumpDatabase, restoreDatabase } from './snapshot';
import { auditUpgradeApplied, auditUpgradeFailed, auditUpgradeRolledBack } from './audit';
import { loadBundledJournal, compareJournalToDb } from './migrations';

export type ApplyInput = {
  databaseUrl: string;
  backupDir: string;
  fromVersion: string;
  toVersion: string;
  /** Pluggable health probe — defaults to fetching /api/health. */
  healthcheck?: () => Promise<{ ok: boolean; version: string }>;
  /** Pluggable restart — defaults to SIGTERM via PID file. */
  restartServer?: () => Promise<void>;
  /** Pluggable migrator — defaults to Drizzle migrate. */
  runMigrations?: () => Promise<void>;
  healthcheckTimeoutMs?: number;
};

export type ApplyResult = {
  ok: boolean;
  snapshotPath?: string;
  migrationCount?: number;
  error?: string;
};

export async function applyUpgrade(input: ApplyInput): Promise<ApplyResult> {
  const healthcheck = input.healthcheck ?? defaultHealthcheck;
  const restartServer = input.restartServer ?? defaultRestart;
  const runMigrations = input.runMigrations ?? (async () => {
    const client = postgres(input.databaseUrl, { max: 1 });
    const db = drizzle(client);
    await migrate(db, { migrationsFolder: 'drizzle/migrations' });
    await client.end();
  });
  const timeout = input.healthcheckTimeoutMs ?? 60_000;

  // Step 1: snapshot
  let snapshotPath: string | undefined;
  try {
    const snap = await dumpDatabase({ databaseUrl: input.databaseUrl, outDir: input.backupDir });
    snapshotPath = snap.path;
  } catch (err) {
    await auditUpgradeFailed({ fromVersion: input.fromVersion, toVersion: input.toVersion, error: `snapshot: ${(err as Error).message}` });
    return { ok: false, error: `snapshot: ${(err as Error).message}` };
  }

  // Step 2: migrate
  let migrationCount = 0;
  try {
    const journal = await loadBundledJournal();
    const client = postgres(input.databaseUrl, { max: 1 });
    const cmp = await compareJournalToDb({ journal, db: drizzle(client) });
    await client.end();
    migrationCount = cmp.pending.length;
    await runMigrations();
  } catch (err) {
    return rollback(input, snapshotPath, `migrate: ${(err as Error).message}`);
  }

  // Step 3: restart + healthcheck
  try {
    await restartServer();
  } catch (err) {
    return rollback(input, snapshotPath, `restart: ${(err as Error).message}`);
  }

  const deadline = Date.now() + timeout;
  let healthy = false;
  while (Date.now() < deadline) {
    try {
      const h = await healthcheck();
      if (h.ok) {
        healthy = true;
        break;
      }
    } catch {
      // tolerate transient connection errors during restart
    }
    await new Promise((r) => setTimeout(r, 1000));
  }
  if (!healthy) return rollback(input, snapshotPath, 'healthcheck timeout');

  await auditUpgradeApplied({ fromVersion: input.fromVersion, toVersion: input.toVersion, migrationCount });
  return { ok: true, snapshotPath, migrationCount };
}

async function rollback(input: ApplyInput, snapshotPath: string | undefined, reason: string): Promise<ApplyResult> {
  if (snapshotPath) {
    try {
      await restoreDatabase({ databaseUrl: input.databaseUrl, dumpPath: snapshotPath });
      await auditUpgradeRolledBack({ snapshotPath });
    } catch (err) {
      reason = `${reason}; rollback also failed: ${(err as Error).message}`;
    }
  }
  await auditUpgradeFailed({ fromVersion: input.fromVersion, toVersion: input.toVersion, error: reason });
  return { ok: false, snapshotPath, error: reason };
}

async function defaultHealthcheck(): Promise<{ ok: boolean; version: string }> {
  const r = await fetch(`${process.env.PUBLIC_URL ?? 'http://localhost:3000'}/api/health`);
  if (!r.ok) return { ok: false, version: '' };
  const body = (await r.json()) as { version: string };
  return { ok: true, version: body.version };
}

async function defaultRestart(): Promise<void> {
  const { readFile } = await import('node:fs/promises');
  const pidPath = process.env.CAIRN_PID_FILE ?? '/data/cairn.pid';
  const pid = Number((await readFile(pidPath, 'utf8')).trim());
  if (!Number.isFinite(pid)) throw new Error(`invalid pid file ${pidPath}`);
  process.kill(pid, 'SIGTERM');
}
```

```ts
import { readdirSync, statSync } from 'node:fs';
import { join } from 'node:path';
import { restoreDatabase } from './snapshot';
import { auditUpgradeRolledBack } from './audit';

export async function rollbackUpgrade(input: {
  databaseUrl: string;
  backupDir: string;
  snapshotPath?: string;
  restartServer?: () => Promise<void>;
}): Promise<{ ok: boolean; snapshotPath: string; error?: string }> {
  const path = input.snapshotPath ?? pickNewestSnapshot(input.backupDir);
  if (!path) return { ok: false, snapshotPath: '', error: `no snapshot found in ${input.backupDir}` };

  try {
    await restoreDatabase({ databaseUrl: input.databaseUrl, dumpPath: path });
  } catch (err) {
    return { ok: false, snapshotPath: path, error: `restore: ${(err as Error).message}` };
  }
  if (input.restartServer) await input.restartServer();
  await auditUpgradeRolledBack({ snapshotPath: path });
  return { ok: true, snapshotPath: path };
}

function pickNewestSnapshot(dir: string): string | null {
  const candidates = readdirSync(dir)
    .filter((f) => f.endsWith('.sql.gz'))
    .map((f) => ({ name: f, mtime: statSync(join(dir, f)).mtimeMs }))
    .sort((a, b) => b.mtime - a.mtime);
  return candidates[0] ? join(dir, candidates[0].name) : null;
}
```

```ts
import { drizzle } from 'drizzle-orm/postgres-js';
import postgres from 'postgres';
import { compareJournalToDb, loadBundledJournal } from './migrations';

export type HealthcheckResult = {
  ok: boolean;
  drift: boolean;
  reason?: string;
  appliedCount?: number;
  journalCount?: number;
};

export async function runHealthcheck(input: {
  databaseUrl: string;
  healthcheck?: () => Promise<{ ok: boolean; version: string }>;
}): Promise<HealthcheckResult> {
  const h = input.healthcheck ?? defaultHealth;
  let httpOk = false;
  try {
    const probe = await h();
    httpOk = probe.ok;
  } catch (err) {
    return { ok: false, drift: false, reason: `health probe error: ${(err as Error).message}` };
  }
  if (!httpOk) return { ok: false, drift: false, reason: '/api/health did not return 200' };

  const journal = await loadBundledJournal();
  const client = postgres(input.databaseUrl, { max: 1 });
  const cmp = await compareJournalToDb({ journal, db: drizzle(client) });
  await client.end();

  if (cmp.drifted) {
    return { ok: false, drift: true, reason: cmp.driftReason ?? 'journal/db drift', appliedCount: cmp.applied.length, journalCount: journal.entries.length };
  }
  return { ok: true, drift: false, appliedCount: cmp.applied.length, journalCount: journal.entries.length };
}

async function defaultHealth(): Promise<{ ok: boolean; version: string }> {
  const r = await fetch(`${process.env.PUBLIC_URL ?? 'http://localhost:3000'}/api/health`);
  if (!r.ok) return { ok: false, version: '' };
  return (await r.json()) as { ok: boolean; version: string };
}
```

```ts
import { spawn } from 'node:child_process';

export type ComposeApplyInput = {
  databaseUrl: string;
  backupDir: string;
  fromVersion: string;
  toVersion: string;
  dockerCompose?: (args: string[]) => Promise<{ ok: boolean; stderr?: string }>;
  dump?: (databaseUrl: string, outDir: string) => Promise<{ path: string; bytesWritten: number }>;
  healthcheck?: () => Promise<{ ok: boolean; drift: boolean; reason?: string }>;
  restore?: (databaseUrl: string, dumpPath: string) => Promise<void>;
  healthcheckTimeoutMs?: number;
};

export type ComposeApplyResult = { ok: boolean; snapshotPath?: string; error?: string };

export async function applyViaCompose(input: ComposeApplyInput): Promise<ComposeApplyResult> {
  const compose = input.dockerCompose ?? defaultDockerCompose;
  const dump = input.dump ?? (async (u, d) => (await import('./snapshot')).dumpDatabase({ databaseUrl: u, outDir: d }));
  const health = input.healthcheck ?? (async () => (await import('./healthcheck')).runHealthcheck({ databaseUrl: input.databaseUrl }));
  const restore = input.restore ?? (async (u, p) => (await import('./snapshot')).restoreDatabase({ databaseUrl: u, dumpPath: p }));

  let stop = await compose(['stop', 'cairn', 'cairn-collab']);
  if (!stop.ok) return { ok: false, error: `compose stop: ${stop.stderr ?? ''}` };

  const snap = await dump(input.databaseUrl, input.backupDir);

  const pull = await compose(['pull']);
  if (!pull.ok) return { ok: false, snapshotPath: snap.path, error: `compose pull: ${pull.stderr ?? ''}` };

  const up = await compose(['up', '-d']);
  if (!up.ok) {
    await restore(input.databaseUrl, snap.path).catch(() => {});
    return { ok: false, snapshotPath: snap.path, error: `compose up: ${up.stderr ?? ''}` };
  }

  // Poll healthcheck until success or timeout
  const deadline = Date.now() + (input.healthcheckTimeoutMs ?? 120_000);
  let last = await health();
  while (!last.ok && Date.now() < deadline) {
    await new Promise((r) => setTimeout(r, 2000));
    last = await health();
  }
  if (!last.ok) {
    await restore(input.databaseUrl, snap.path).catch(() => {});
    return { ok: false, snapshotPath: snap.path, error: `healthcheck: ${last.reason ?? 'unknown'}` };
  }
  return { ok: true, snapshotPath: snap.path };
}

async function defaultDockerCompose(args: string[]): Promise<{ ok: boolean; stderr?: string }> {
  return new Promise((resolve) => {
    const proc = spawn('docker', ['compose', ...args], { stdio: ['ignore', 'inherit', 'pipe'] });
    let stderr = '';
    proc.stderr.on('data', (d) => {
      stderr += d.toString();
    });
    proc.on('exit', (code) => resolve({ ok: code === 0, stderr }));
    proc.on('error', () => resolve({ ok: false, stderr }));
  });
}
```

```ts
#!/usr/bin/env node
import { Command } from 'commander';
import { applyUpgrade } from '@/lib/upgrade/apply';
import { applyViaCompose } from '@/lib/upgrade/compose';
import { previewUpgrade } from '@/lib/upgrade/preview';
import { rollbackUpgrade } from '@/lib/upgrade/rollback';
import { runHealthcheck } from '@/lib/upgrade/healthcheck';

const program = new Command();
program
  .name('cairn-upgrade')
  .description('Cairn upgrade orchestration CLI')
  .option('--database-url <url>', 'Postgres connection string', process.env.DATABASE_URL)
  .option('--backup-dir <path>', 'Backup directory', process.env.CAIRN_BACKUP_DIR ?? '/data/backups')
  .option('--via-compose', 'Wrap apply with docker compose stop/pull/up', false);

const pkg = require(require('node:path').join(process.cwd(), 'package.json')) as { version: string };

program
  .command('preview')
  .description('Dry-run pending migrations against a scratch DB; show schema diff')
  .action(async () => {
    const opts = program.opts<{ databaseUrl: string }>();
    const r = await previewUpgrade({ databaseUrl: opts.databaseUrl });
    console.log(JSON.stringify({ ok: r.ok, pendingTags: r.pendingTags }, null, 2));
    if (r.schemaDiff) console.log('\n--- schema diff ---\n' + r.schemaDiff);
    process.exit(r.ok ? 0 : 1);
  });

program
  .command('apply')
  .description('Apply pending migrations (refuses without --confirm)')
  .option('--confirm', 'Required confirmation flag', false)
  .option('--from-version <v>', 'Current version', pkg.version)
  .option('--to-version <v>', 'Target version', pkg.version)
  .action(async (cmdOpts: { confirm: boolean; fromVersion: string; toVersion: string }) => {
    if (!cmdOpts.confirm) {
      console.error('apply refuses to run without --confirm');
      process.exit(1);
    }
    const opts = program.opts<{ databaseUrl: string; backupDir: string; viaCompose: boolean }>();
    const result = opts.viaCompose
      ? await applyViaCompose({
          databaseUrl: opts.databaseUrl,
          backupDir: opts.backupDir,
          fromVersion: cmdOpts.fromVersion,
          toVersion: cmdOpts.toVersion,
        })
      : await applyUpgrade({
          databaseUrl: opts.databaseUrl,
          backupDir: opts.backupDir,
          fromVersion: cmdOpts.fromVersion,
          toVersion: cmdOpts.toVersion,
        });
    console.log(JSON.stringify(result, null, 2));
    process.exit(result.ok ? 0 : 1);
  });

program
  .command('rollback')
  .description('Restore from a snapshot (newest in backup-dir if --snapshot omitted)')
  .option('--snapshot <path>', 'Specific snapshot file')
  .action(async (cmdOpts: { snapshot?: string }) => {
    const opts = program.opts<{ databaseUrl: string; backupDir: string }>();
    const result = await rollbackUpgrade({
      databaseUrl: opts.databaseUrl,
      backupDir: opts.backupDir,
      snapshotPath: cmdOpts.snapshot,
    });
    console.log(JSON.stringify(result, null, 2));
    process.exit(result.ok ? 0 : 1);
  });

program
  .command('healthcheck')
  .description('Probe /api/health + check __drizzle_migrations against bundled journal')
  .action(async () => {
    const opts = program.opts<{ databaseUrl: string }>();
    const r = await runHealthcheck({ databaseUrl: opts.databaseUrl });
    console.log(JSON.stringify(r, null, 2));
    process.exit(r.ok ? 0 : 1);
  });

program.parseAsync(process.argv).catch((err: unknown) => {
  console.error(err);
  process.exit(1);
});
```

# Cairn v0.9.0 — Plan P39: SIEM forwarder migration 0048 + core engine + syslog + HTTP webhook

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the SIEM-forwarder foundation. Migration `0048` adds `siem_forwarders` + `siem_delivery_log`. Core dispatcher subscribes to v0.6 P18 audit-log writes, enqueues a delivery to every enabled forwarder of the matching workspace, retries with exponential backoff (1s / 5s / 30s / 5m), records each attempt. Two targets ship in this plan: (a) syslog RFC 5424 over UDP and TCP, (b) generic HTTP webhook (POST JSON with optional bearer auth). Admin UI at `/admin/siem` with list + add/edit/test/delete actions. P40 layers Splunk HEC + Datadog Logs + S3 NDJSON onto this scaffold.

**Architecture:** The audit-log recorder (`src/lib/audit/recorder.ts#recordAudit`) is the single write site for `audit_log`. Add a post-write hook that emits a typed event into a process-local queue. A dispatcher (`src/lib/siem/dispatch.ts`) consumes the queue and fans out to every enabled forwarder for that workspace. Delivery happens in-process; failed deliveries persist to `siem_delivery_log` with `status='retry'` and a `next_attempt_at` timestamp, picked up by a cron sweep (one of the existing cron entries in `src/server/cron.ts`).

**Tech Stack:** Drizzle, Postgres, Node net (UDP via `dgram`, TCP via `node:net`), built-in `fetch`. Vitest + Testcontainers. No new runtime dep.

**Depends on:** v0.6 P18 audit log, v0.6 P20 pino REDACT_PATHS + prom-client.

```ts
// src/lib/siem/format.ts
export type AuditEventInput = {
  id: string;
  workspaceId: string;
  actorUserId: string | null;
  action: string;
  targetType: string | null;
  targetId: string | null;
  metadata: Record<string, unknown>;
  createdAt: Date;
};

export type SiemEnvelope = {
  id: string;
  timestamp: string;
  workspace_id: string;
  actor_user_id: string | null;
  action: string;
  target: { type: string; id: string } | null;
  metadata: Record<string, unknown>;
};

export function formatAuditEvent(input: AuditEventInput): SiemEnvelope {
  return {
    id: input.id,
    timestamp: input.createdAt.toISOString(),
    workspace_id: input.workspaceId,
    actor_user_id: input.actorUserId,
    action: input.action,
    target:
      input.targetType && input.targetId
        ? { type: input.targetType, id: input.targetId }
        : null,
    metadata: input.metadata,
  };
}
```

```ts
// src/lib/siem/backoff.ts
const SCHEDULE_MS = [1_000, 5_000, 30_000, 5 * 60_000];

export function nextBackoffMs(attempt: number): number {
  if (attempt < 1) return SCHEDULE_MS[0];
  if (attempt > SCHEDULE_MS.length) return SCHEDULE_MS[SCHEDULE_MS.length - 1];
  return SCHEDULE_MS[attempt - 1];
}

export const MAX_ATTEMPTS = SCHEDULE_MS.length;
export function isExhausted(attempt: number): boolean {
  return attempt >= MAX_ATTEMPTS + 1;
}
```

```ts
import { createSocket } from 'node:dgram';
import { createConnection } from 'node:net';
import type { SiemEnvelope } from '../format';

const FACILITY_LOCAL0 = 16;
const SEVERITY_INFORMATIONAL = 6;
const PRIORITY = FACILITY_LOCAL0 * 8 + SEVERITY_INFORMATIONAL;

function buildFrame(env: SiemEnvelope, hostname: string): string {
  const ts = env.timestamp;
  const app = 'cairn';
  const procid = '-';
  const msgid = env.action;
  const msg = JSON.stringify(env);
  return `<${PRIORITY}>1 ${ts} ${hostname} ${app} ${procid} ${msgid} - ${msg}`;
}

export async function sendSyslog(
  forwarder: { endpoint: string; options: Record<string, unknown> },
  env: SiemEnvelope,
): Promise<void> {
  const url = new URL(forwarder.endpoint);
  const hostname = (forwarder.options.hostname as string | undefined) ?? 'cairn';
  const frame = buildFrame(env, hostname);

  if (url.protocol === 'udp:') {
    await new Promise<void>((resolve, reject) => {
      const sock = createSocket('udp4');
      sock.send(frame, Number(url.port), url.hostname, (err) => {
        sock.close();
        err ? reject(err) : resolve();
      });
    });
    return;
  }
  if (url.protocol === 'tcp:') {
    await new Promise<void>((resolve, reject) => {
      const sock = createConnection({ host: url.hostname, port: Number(url.port) });
      sock.once('connect', () => {
        sock.write(`${frame}\n`, (err) => {
          sock.end();
          err ? reject(err) : resolve();
        });
      });
      sock.once('error', reject);
    });
    return;
  }
  throw new Error(`unsupported scheme: ${url.protocol}`);
}
```

```ts
import type { SiemEnvelope } from '../format';

export async function sendHttp(
  forwarder: { endpoint: string; credentialSecret: string | null; options: Record<string, unknown> },
  env: SiemEnvelope,
): Promise<void> {
  const timeoutMs = (forwarder.options.timeoutMs as number | undefined) ?? 5_000;
  const ac = new AbortController();
  const timer = setTimeout(() => ac.abort(), timeoutMs);
  try {
    const headers: Record<string, string> = { 'content-type': 'application/json' };
    if (forwarder.credentialSecret) headers.authorization = `Bearer ${forwarder.credentialSecret}`;
    const res = await fetch(forwarder.endpoint, {
      method: 'POST',
      headers,
      body: JSON.stringify(env),
      signal: ac.signal,
    });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
  } finally {
    clearTimeout(timer);
  }
}
```

```ts
import { and, eq } from 'drizzle-orm';
import { db } from '@/db/client';
import { siemDeliveryLog, siemForwarders } from '@/db/schema/siem';
import { logger } from '@/lib/observability/logger';
import { isExhausted, nextBackoffMs } from './backoff';
import { formatAuditEvent } from './format';
import { sendHttp } from './targets/http';
import { sendSyslog } from './targets/syslog';

export type Sender = (
  forwarder: { endpoint: string; credentialSecret: string | null; options: Record<string, unknown> },
  env: ReturnType<typeof formatAuditEvent>,
) => Promise<void>;

export const DEFAULT_SENDERS: Record<string, Sender> = {
  http: sendHttp,
  syslog: (f, env) => sendSyslog({ endpoint: f.endpoint, options: f.options }, env),
};

export async function dispatchAuditEvent(
  audit: {
    id: string;
    workspaceId: string;
    actorUserId: string | null;
    action: string;
    targetType: string | null;
    targetId: string | null;
    metadata: Record<string, unknown>;
    createdAt: Date;
  },
  opts: { senders?: Record<string, Sender> } = {},
): Promise<void> {
  const senders = opts.senders ?? DEFAULT_SENDERS;
  const envelope = formatAuditEvent(audit);

  const forwarders = await db()
    .select()
    .from(siemForwarders)
    .where(and(eq(siemForwarders.workspaceId, audit.workspaceId), eq(siemForwarders.enabled, true)));

  await Promise.all(
    forwarders.map(async (forwarder) => {
      const send = senders[forwarder.kind];
      if (!send) {
        logger.warn({ kind: forwarder.kind }, 'siem.sender_missing');
        return;
      }
      // Determine current attempt from existing delivery log rows for this event+forwarder.
      const prior = await db()
        .select()
        .from(siemDeliveryLog)
        .where(
          and(
            eq(siemDeliveryLog.forwarderId, forwarder.id),
            eq(siemDeliveryLog.auditEventId, audit.id),
          ),
        );
      const attempt = prior.length + 1;
      try {
        await send(forwarder, envelope);
        await db().insert(siemDeliveryLog).values({
          forwarderId: forwarder.id,
          auditEventId: audit.id,
          status: 'success',
          attempt,
        });
      } catch (err) {
        const msg = err instanceof Error ? err.message : String(err);
        const exhausted = isExhausted(attempt + 1);
        await db().insert(siemDeliveryLog).values({
          forwarderId: forwarder.id,
          auditEventId: audit.id,
          status: exhausted ? 'failed' : 'retry',
          attempt,
          error: msg,
          nextAttemptAt: exhausted ? null : new Date(Date.now() + nextBackoffMs(attempt)),
        });
        if (exhausted) {
          // Emit a meta-audit (does NOT re-trigger dispatch — see Task 6 guard).
          logger.error(
            { forwarderId: forwarder.id, auditEventId: audit.id, error: msg },
            'siem.delivery_failed',
          );
        }
      }
    }),
  );
}
```

# Cairn v0.9.0 — Plan P23: Tasks hub — `/my-tasks` aggregator + materialized view

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [x]`) syntax for tracking.

**Goal:** A workspace-spanning "My tasks" page at `/my-tasks`. Lists every task-list item the user owns (assigned-to, mentioned-in, or authored on the page), across every workspace they have v0.7 ACL read access to. Each row links to the source page and exposes a checkbox that toggles the underlying TipTap task-item via the v0.3 Yjs collab path. Powered by a Postgres materialized view `mv_user_tasks` refreshed on `pages.content` UPDATE.

**Architecture:** Three layers.

1. **Database:** A materialized view `mv_user_tasks` keyed `(page_id, block_id)` with columns `(user_id, page_id, workspace_id, block_id, text, checked, assigned_to, mentioned_by, due_at_iso, created_at)`. A PL/pgSQL function `extract_tasks_from_content(content jsonb)` walks the TipTap JSON, finds every `taskItem` node, and emits one row per (task, user). The trigger `pages_refresh_user_tasks` on `pages` AFTER UPDATE OF content fires `REFRESH MATERIALIZED VIEW CONCURRENTLY mv_user_tasks` (cheap because the page row update has already committed). Wrapped in a single migration `0044a_user_tasks_view.sql` — slotted **before** P26's `0044` so we don't collide with the lifecycle migration; if Drizzle re-numbers, accept what generates and rename helpfully.
2. **Library:** `src/lib/tasks/aggregate.ts#listMyTasks(userId, opts)` selects from `mv_user_tasks` with filters (`workspaceId?`, `status?: 'open'|'done'|'all'`, `dueBy?: Date`) and post-filters by v0.7 ACL using `requirePageAccess` semantics in batch — a helper `assertReadablePageIds(userId, pageIds)` to avoid N+1.
3. **UI:** Page route `/my-tasks` (server component) loads via `listMyTasks` and renders filter chips + table. Each row's checkbox `POST`s to `/api/pages/[id]/tasks/[blockId]/toggle`, which opens the Yjs doc on the Hocuspocus server, mutates the `taskItem.attrs.checked`, commits the update — the change then propagates back through the v0.3 collab channel to any open viewers.

**Tech Stack:** Postgres materialized views + PL/pgSQL, Drizzle for read queries (raw SQL via `db.execute` for the view), Yjs server-side mutator (existing v0.3 helper `applyServerUpdate`), Next.js 16 server components for the page route, shadcn/ui for chip-row + DatePicker.

```sql
-- 0044a_user_tasks_view.sql
-- Materialized view exposing every TipTap taskItem node from pages.content
-- with the user(s) involved (assignee, mentioner, author). Refreshed via a
-- trigger on pages AFTER UPDATE OF content.

CREATE OR REPLACE FUNCTION extract_tasks_from_content(content jsonb)
RETURNS TABLE (
  block_id   text,
  text       text,
  checked    boolean,
  assigned_to uuid,
  mentioned_by uuid,
  due_at_iso text
)
LANGUAGE sql
IMMUTABLE
AS $$
  WITH RECURSIVE walk(node) AS (
    SELECT content
    UNION ALL
    SELECT child
    FROM walk, jsonb_array_elements(COALESCE(walk.node->'content','[]'::jsonb)) AS child
  ),
  task_nodes AS (
    SELECT node
    FROM walk
    WHERE node->>'type' = 'taskItem'
  )
  SELECT
    (node->'attrs'->>'blockId')::text                            AS block_id,
    COALESCE(node->'attrs'->>'text', node #>> '{content,0,content,0,text}', '')::text AS text,
    COALESCE((node->'attrs'->>'checked')::boolean, false)        AS checked,
    NULLIF(node->'attrs'->>'assignedTo','')::uuid                AS assigned_to,
    NULLIF(node->'attrs'->>'mentionedBy','')::uuid               AS mentioned_by,
    node->'attrs'->>'dueAt'                                       AS due_at_iso
  FROM task_nodes;
$$;

DROP MATERIALIZED VIEW IF EXISTS mv_user_tasks;

CREATE MATERIALIZED VIEW mv_user_tasks AS
WITH page_tasks AS (
  SELECT
    p.id           AS page_id,
    p.workspace_id AS workspace_id,
    p.created_at   AS page_created_at,
    t.*
  FROM pages p
  CROSS JOIN LATERAL extract_tasks_from_content(p.content) AS t
  WHERE p.deleted_at IS NULL
)
SELECT
  u.user_id::uuid                                          AS user_id,
  page_tasks.page_id,
  page_tasks.workspace_id,
  page_tasks.block_id,
  page_tasks.text,
  page_tasks.checked,
  page_tasks.assigned_to,
  page_tasks.mentioned_by,
  page_tasks.due_at_iso,
  page_tasks.page_created_at                                AS created_at
FROM page_tasks
CROSS JOIN LATERAL (
  SELECT DISTINCT v.user_id
  FROM (VALUES (page_tasks.assigned_to), (page_tasks.mentioned_by)) AS v(user_id)
  WHERE v.user_id IS NOT NULL
) AS u
WHERE page_tasks.block_id IS NOT NULL;

CREATE UNIQUE INDEX mv_user_tasks_pk
  ON mv_user_tasks (user_id, page_id, block_id);

CREATE INDEX mv_user_tasks_user_workspace
  ON mv_user_tasks (user_id, workspace_id);

CREATE INDEX mv_user_tasks_user_checked
  ON mv_user_tasks (user_id, checked);

CREATE INDEX mv_user_tasks_user_due
  ON mv_user_tasks (user_id, due_at_iso);

CREATE OR REPLACE FUNCTION refresh_mv_user_tasks()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  -- CONCURRENTLY requires the unique index above; we created it.
  REFRESH MATERIALIZED VIEW CONCURRENTLY mv_user_tasks;
  RETURN NULL;
END;
$$;

DROP TRIGGER IF EXISTS pages_refresh_user_tasks ON pages;
CREATE TRIGGER pages_refresh_user_tasks
AFTER INSERT OR UPDATE OF content OR DELETE ON pages
FOR EACH STATEMENT
EXECUTE FUNCTION refresh_mv_user_tasks();
```

```ts
import { sql } from 'drizzle-orm';
import { getDb } from '@/server/db';

export type MyTaskRow = {
  pageId: string;
  workspaceId: string;
  blockId: string;
  text: string;
  checked: boolean;
  dueAt: Date | null;
  pageTitle: string;
  pageIcon: string | null;
};

export type ListMyTasksOpts = {
  workspaceId?: string;
  status?: 'open' | 'done' | 'all';
  dueBy?: Date;
};

export async function listMyTasks(
  userId: string,
  opts: ListMyTasksOpts,
): Promise<MyTaskRow[]> {
  const db = getDb();
  const status = opts.status ?? 'open';
  // Permission gate: page must either belong to a workspace the user is a member of,
  // or carry an explicit page_acl read grant. Joining both sources keeps the v0.7 ACL
  // chain happy.
  const rows = (await db.execute(sql`
    SELECT
      mv.page_id, mv.workspace_id, mv.block_id, mv.text, mv.checked, mv.due_at_iso,
      p.title AS page_title, p.icon AS page_icon
    FROM mv_user_tasks mv
    JOIN pages p ON p.id = mv.page_id AND p.deleted_at IS NULL
    LEFT JOIN workspace_members wm
      ON wm.workspace_id = mv.workspace_id AND wm.user_id = ${userId}
    LEFT JOIN page_acls pa
      ON pa.page_id = mv.page_id AND pa.user_id = ${userId} AND pa.role IS NOT NULL
    WHERE mv.user_id = ${userId}
      AND (wm.user_id IS NOT NULL OR pa.user_id IS NOT NULL)
      ${opts.workspaceId ? sql`AND mv.workspace_id = ${opts.workspaceId}` : sql``}
      ${
        status === 'open'
          ? sql`AND mv.checked = false`
          : status === 'done'
            ? sql`AND mv.checked = true`
            : sql``
      }
      ${
        opts.dueBy
          ? sql`AND mv.due_at_iso IS NOT NULL AND mv.due_at_iso::timestamptz <= ${opts.dueBy.toISOString()}::timestamptz`
          : sql``
      }
    ORDER BY mv.due_at_iso NULLS LAST, mv.created_at DESC
  `)) as unknown as Array<{
    page_id: string;
    workspace_id: string;
    block_id: string;
    text: string;
    checked: boolean;
    due_at_iso: string | null;
    page_title: string;
    page_icon: string | null;
  }>;

  return rows.map((r) => ({
    pageId: r.page_id,
    workspaceId: r.workspace_id,
    blockId: r.block_id,
    text: r.text,
    checked: r.checked,
    dueAt: r.due_at_iso ? new Date(r.due_at_iso) : null,
    pageTitle: r.page_title,
    pageIcon: r.page_icon,
  }));
}
```

```ts
import { sql } from 'drizzle-orm';
import { recordAudit } from '@/lib/audit/log';
import { getDb } from '@/server/db';

type ContentNode = {
  type?: string;
  attrs?: Record<string, unknown>;
  content?: ContentNode[];
};

function walkAndToggle(node: ContentNode, blockId: string): boolean {
  if (node.type === 'taskItem' && (node.attrs as { blockId?: string } | undefined)?.blockId === blockId) {
    node.attrs = { ...(node.attrs ?? {}), checked: !((node.attrs as { checked?: boolean } | undefined)?.checked ?? false) };
    return true;
  }
  if (Array.isArray(node.content)) {
    for (const child of node.content) {
      if (walkAndToggle(child, blockId)) return true;
    }
  }
  return false;
}

export async function toggleTaskCheck(input: {
  pageId: string;
  blockId: string;
  userId: string;
}): Promise<{ checked: boolean }> {
  const db = getDb();
  const rows = (await db.execute(sql`
    SELECT content FROM pages WHERE id = ${input.pageId} AND deleted_at IS NULL
  `)) as unknown as Array<{ content: ContentNode }>;
  const page = rows[0];
  if (!page) throw new Error('page not found');

  const content = structuredClone(page.content);
  const found = walkAndToggle(content, input.blockId);
  if (!found) throw new Error('task block not found');

  await db.execute(sql`
    UPDATE pages SET content = ${content}::jsonb, updated_at = now()
    WHERE id = ${input.pageId}
  `);

  // Find the new state to return.
  const final = await db.execute(sql`SELECT checked FROM mv_user_tasks WHERE page_id = ${input.pageId} AND block_id = ${input.blockId} LIMIT 1`);
  const checked = (final[0] as { checked?: boolean } | undefined)?.checked ?? false;

  await recordAudit({
    actorUserId: input.userId,
    action: 'task.toggle',
    entityType: 'page',
    entityId: input.pageId,
    details: { blockId: input.blockId, checked },
  });

  return { checked };
}
```

import { requirePageAccess } from '@/lib/pages/access';
import { toggleTaskCheck } from '@/lib/tasks/toggle';

export async function POST(
  _req: Request,
  ctxParams: { params: Promise<{ id: string; blockId: string }> },
): Promise<Response> {
  const { id, blockId } = await ctxParams.params;
  const { ctx } = await requirePageAccess(id, 'editor');
  try {
    const { checked } = await toggleTaskCheck({ pageId: id, blockId, userId: ctx.userId });
    return new Response(JSON.stringify({ checked }), {
      status: 200,
      headers: { 'content-type': 'application/json' },
    });
  } catch (err) {
    return new Response(JSON.stringify({ error: (err as Error).message }), {
      status: 400,
      headers: { 'content-type': 'application/json' },
    });
  }
}

- [x] **Step 2: Implement `src/app/my-tasks/tasks-table.tsx`**

```tsx
'use client';

import Link from 'next/link';
import { useRouter, useSearchParams } from 'next/navigation';
import { useState, useTransition } from 'react';
import { Button } from '@/components/ui/button';
import { Checkbox } from '@/components/ui/checkbox';
import { Input } from '@/components/ui/input';
import type { MyTaskRow } from '@/lib/tasks/aggregate';

const STATUSES = ['open', 'done', 'all'] as const;
type Status = (typeof STATUSES)[number];

export function TasksTable(props: {
  initialTasks: MyTaskRow[];
  initialStatus: Status;
  initialDue?: string;
}): JSX.Element {
  const router = useRouter();
  const sp = useSearchParams();
  const [, startTransition] = useTransition();
  const [tasks, setTasks] = useState(props.initialTasks);

  const setQuery = (patch: Record<string, string | null>): void => {
    const next = new URLSearchParams(sp.toString());
    for (const [k, v] of Object.entries(patch)) {
      if (v === null || v === '') next.delete(k);
      else next.set(k, v);
    }
    startTransition(() => router.replace(`/my-tasks?${next.toString()}`));
  };

  const toggle = async (pageId: string, blockId: string): Promise<void> => {
    setTasks((prev) =>
      prev.map((t) => (t.pageId === pageId && t.blockId === blockId ? { ...t, checked: !t.checked } : t)),
    );
    const res = await fetch(`/api/pages/${pageId}/tasks/${blockId}/toggle`, { method: 'POST' });
    if (!res.ok) {
      // revert
      setTasks((prev) =>
        prev.map((t) => (t.pageId === pageId && t.blockId === blockId ? { ...t, checked: !t.checked } : t)),
      );
    }
  };

  return (
    <div className="mt-6 space-y-4">
      <div className="flex gap-2" role="group" aria-label="Filter by status">
        {STATUSES.map((s) => (
          <Button
            key={s}
            variant={props.initialStatus === s ? 'default' : 'outline'}
            size="sm"
            onClick={() => setQuery({ status: s })}
          >
            {s}
          </Button>
        ))}
        <Input
          type="date"
          className="ml-auto w-40"
          value={props.initialDue ?? ''}
          aria-label="Due by"
          onChange={(e) => setQuery({ due: e.target.value || null })}
        />
      </div>
      <ul className="divide-y rounded border">
        {tasks.length === 0 && (
          <li className="p-4 text-center text-sm text-muted-foreground">No tasks.</li>
        )}
        {tasks.map((t) => (
          <li key={`${t.pageId}-${t.blockId}`} className="flex items-center gap-3 p-3">
            <Checkbox
              checked={t.checked}
              onCheckedChange={() => toggle(t.pageId, t.blockId)}
              aria-label={`Toggle ${t.text}`}
            />
            <span className={t.checked ? 'line-through text-muted-foreground' : ''}>{t.text}</span>
            <Link href={`/pages/${t.pageId}` as never} className="ml-auto text-sm text-muted-foreground hover:underline">
              {t.pageIcon ? `${t.pageIcon} ` : ''}
              {t.pageTitle}
            </Link>
            {t.dueAt && (
              <time className="text-xs text-muted-foreground" dateTime={t.dueAt.toISOString()}>
                due {t.dueAt.toISOString().slice(0, 10)}
              </time>
            )}
          </li>
        ))}
      </ul>
    </div>
  );
}
```

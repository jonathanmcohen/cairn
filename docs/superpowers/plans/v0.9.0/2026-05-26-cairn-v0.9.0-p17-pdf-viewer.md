# Cairn v0.9.0 — Plan P17: PDF viewer + annotation (migration 0042)

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. Mark `- [ ]` → `- [x]` as work lands.

**Goal:** Drop PDFs into a Cairn page, render every page client-side via `pdfjs-dist`, and let the current user annotate them (highlight, comment, shape). Annotations are persisted per-user (multi-user Yjs sync is deferred to v1.0). Re-uses the v0.5 `/api/files` upload + signed-URL pipeline.

**Architecture:** New TipTap node `PdfNode` carries a `fileId` attribute. Renderer fetches the signed URL via the existing `/api/files/[id]/signed-url` route, then hands it to `pdfjs-dist` which renders each page to a `<canvas>` with a transparent SVG overlay. The overlay reads/writes annotations to `pdf_annotations` (migration 0042) scoped by `(file_id, page_number, created_by)`. Slash command `/pdf` opens a file picker; drag-drop of `.pdf` files on the editor surface auto-creates the node + uploads the file.

**Tech Stack:** `pdfjs-dist` (new dep, worker bundled), TipTap 3, Next.js 16 App Router, Drizzle ORM, Postgres 16, Zod v4, Vitest 4 + Testcontainers Postgres, jsdom for client-side renderer unit tests.

```ts
import { and, eq } from 'drizzle-orm';
import type { Database } from '@/db/client';
import { pdfAnnotations } from '@/db/schema';
import type { AnnotationKind, Rect } from './schema';

export type AnnotationRow = {
  id: string;
  pageId: string;
  fileId: string;
  pageNumber: number;
  rect: Rect;
  kind: AnnotationKind;
  content: string | null;
  createdBy: string;
  createdAt: Date;
  updatedAt: Date;
};

export async function listAnnotations(
  db: Database,
  input: { fileId: string; userId: string },
): Promise<AnnotationRow[]> {
  const rows = await db
    .select()
    .from(pdfAnnotations)
    .where(and(eq(pdfAnnotations.fileId, input.fileId), eq(pdfAnnotations.createdBy, input.userId)));
  return rows as AnnotationRow[];
}

export async function createAnnotation(
  db: Database,
  input: {
    pageId: string;
    fileId: string;
    pageNumber: number;
    rect: Rect;
    kind: AnnotationKind;
    content: string | null;
    createdBy: string;
  },
): Promise<AnnotationRow> {
  const [row] = await db.insert(pdfAnnotations).values(input).returning();
  return row as AnnotationRow;
}

export async function updateAnnotation(
  db: Database,
  input: { id: string; userId: string; rect?: Rect; content?: string | null },
): Promise<AnnotationRow> {
  const [row] = await db
    .update(pdfAnnotations)
    .set({
      ...(input.rect !== undefined ? { rect: input.rect } : {}),
      ...(input.content !== undefined ? { content: input.content } : {}),
      updatedAt: new Date(),
    })
    .where(and(eq(pdfAnnotations.id, input.id), eq(pdfAnnotations.createdBy, input.userId)))
    .returning();
  if (!row) throw new Error('annotation not found');
  return row as AnnotationRow;
}

export async function deleteAnnotation(
  db: Database,
  input: { id: string; userId: string },
): Promise<void> {
  const result = await db
    .delete(pdfAnnotations)
    .where(and(eq(pdfAnnotations.id, input.id), eq(pdfAnnotations.createdBy, input.userId)))
    .returning({ id: pdfAnnotations.id });
  if (result.length === 0) throw new Error('annotation not found');
}
```

```ts
import { NextResponse } from 'next/server';
import { auth } from '@/lib/auth/config';
import { requirePageAccess } from '@/lib/pages/access';
import { getDb } from '@/db/client';
import { createAnnotation, listAnnotations } from '@/lib/pdf/annotations';
import { createAnnotationInput } from '@/lib/pdf/schema';
import { files } from '@/db/schema';
import { eq } from 'drizzle-orm';

export const runtime = 'nodejs';

export async function GET(
  _req: Request,
  ctx: { params: Promise<{ fileId: string }> },
): Promise<Response> {
  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
  const { fileId } = await ctx.params;
  const db = getDb();
  const [file] = await db.select().from(files).where(eq(files.id, fileId)).limit(1);
  if (!file?.pageId) return NextResponse.json({ error: 'not found' }, { status: 404 });
  await requirePageAccess(file.pageId, 'viewer');
  const annotations = await listAnnotations(db, { fileId, userId: session.user.id });
  return NextResponse.json({ annotations });
}

export async function POST(
  req: Request,
  ctx: { params: Promise<{ fileId: string }> },
): Promise<Response> {
  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
  const { fileId } = await ctx.params;
  const json = await req.json();
  const parsed = createAnnotationInput.safeParse(json);
  if (!parsed.success) return NextResponse.json({ error: parsed.error.format() }, { status: 400 });
  if (parsed.data.fileId !== fileId) return NextResponse.json({ error: 'file id mismatch' }, { status: 400 });
  await requirePageAccess(parsed.data.pageId, 'editor');
  const db = getDb();
  const created = await createAnnotation(db, { ...parsed.data, createdBy: session.user.id });
  return NextResponse.json({ annotation: created }, { status: 201 });
}
```

```ts
import { NextResponse } from 'next/server';
import { auth } from '@/lib/auth/config';
import { getDb } from '@/db/client';
import { deleteAnnotation, updateAnnotation } from '@/lib/pdf/annotations';
import { updateAnnotationInput } from '@/lib/pdf/schema';

export const runtime = 'nodejs';

export async function PATCH(
  req: Request,
  ctx: { params: Promise<{ fileId: string; annotationId: string }> },
): Promise<Response> {
  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
  const { annotationId } = await ctx.params;
  const json = await req.json();
  const parsed = updateAnnotationInput.safeParse(json);
  if (!parsed.success) return NextResponse.json({ error: parsed.error.format() }, { status: 400 });
  try {
    const updated = await updateAnnotation(getDb(), {
      id: annotationId,
      userId: session.user.id,
      ...parsed.data,
    });
    return NextResponse.json({ annotation: updated });
  } catch {
    return NextResponse.json({ error: 'not found' }, { status: 404 });
  }
}

export async function DELETE(
  _req: Request,
  ctx: { params: Promise<{ fileId: string; annotationId: string }> },
): Promise<Response> {
  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
  const { annotationId } = await ctx.params;
  try {
    await deleteAnnotation(getDb(), { id: annotationId, userId: session.user.id });
    return new Response(null, { status: 204 });
  } catch {
    return NextResponse.json({ error: 'not found' }, { status: 404 });
  }
}
```

```tsx
'use client';

import { useEffect, useRef, useState } from 'react';
import type { AnnotationKind, Rect } from '@/lib/pdf/schema';

type LocalAnnotation = {
  id: string;
  pageNumber: number;
  rect: Rect;
  kind: AnnotationKind;
  content: string | null;
};

type Props = { fileId: string; defaultPage: number };

export default function PdfRenderer({ fileId, defaultPage }: Props) {
  const [signedUrl, setSignedUrl] = useState<string | null>(null);
  const [pages, setPages] = useState<number[]>([]);
  const [tool, setTool] = useState<AnnotationKind>('highlight');
  const [annotations, setAnnotations] = useState<LocalAnnotation[]>([]);
  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());

  useEffect(() => {
    let cancelled = false;
    void (async () => {
      const res = await fetch(`/api/files/${fileId}/signed-url`);
      if (!res.ok) return;
      const { url } = (await res.json()) as { url: string };
      if (!cancelled) setSignedUrl(url);
      const aRes = await fetch(`/api/pdf/${fileId}/annotations`);
      if (aRes.ok) {
        const body = (await aRes.json()) as { annotations: LocalAnnotation[] };
        if (!cancelled) setAnnotations(body.annotations);
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [fileId]);

  useEffect(() => {
    if (!signedUrl) return;
    let cancelled = false;
    void (async () => {
      const pdfjs = await import('pdfjs-dist');
      const workerSrc = (await import('pdfjs-dist/build/pdf.worker.min.mjs?url')).default;
      pdfjs.GlobalWorkerOptions.workerSrc = workerSrc;
      const doc = await pdfjs.getDocument(signedUrl).promise;
      if (cancelled) return;
      setPages(Array.from({ length: doc.numPages }, (_, i) => i + 1));
      for (let n = 1; n <= doc.numPages; n++) {
        const page = await doc.getPage(n);
        const canvas = canvasRefs.current.get(n);
        if (!canvas) continue;
        const viewport = page.getViewport({ scale: 1.5 });
        canvas.width = viewport.width;
        canvas.height = viewport.height;
        const ctx = canvas.getContext('2d');
        if (!ctx) continue;
        await page.render({ canvasContext: ctx, viewport }).promise;
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [signedUrl]);

  async function createOnDrag(pageNumber: number, rect: Rect) {
    const res = await fetch(`/api/pdf/${fileId}/annotations`, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({
        pageId: (window as unknown as { __cairnPageId: string }).__cairnPageId,
        fileId,
        pageNumber,
        rect,
        kind: tool,
        content: null,
      }),
    });
    if (res.ok) {
      const body = (await res.json()) as { annotation: LocalAnnotation };
      setAnnotations((prev) => [...prev, body.annotation]);
    }
  }

  return (
    <div className="cairn-pdf-viewer">
      <div role="toolbar" aria-label="PDF annotation tools" className="flex gap-2 p-2 border-b">
        {(['highlight', 'comment', 'shape'] as const).map((k) => (
          <button
            key={k}
            type="button"
            aria-pressed={tool === k}
            onClick={() => setTool(k)}
            className={tool === k ? 'bg-primary text-primary-foreground px-2 py-1 rounded' : 'px-2 py-1 rounded border'}
          >
            {k}
          </button>
        ))}
      </div>
      {pages.map((n) => (
        <div key={n} className="relative my-2" data-page={n} data-default-page={defaultPage}>
          <canvas
            ref={(el) => {
              if (el) canvasRefs.current.set(n, el);
              else canvasRefs.current.delete(n);
            }}
          />
          <PageOverlay
            pageNumber={n}
            annotations={annotations.filter((a) => a.pageNumber === n)}
            onCreate={(rect) => createOnDrag(n, rect)}
          />
        </div>
      ))}
    </div>
  );
}

function PageOverlay({
  pageNumber,
  annotations,
  onCreate,
}: {
  pageNumber: number;
  annotations: LocalAnnotation[];
  onCreate: (rect: Rect) => void;
}) {
  const svgRef = useRef<SVGSVGElement | null>(null);
  const dragRef = useRef<{ x: number; y: number } | null>(null);
  return (
    <svg
      ref={svgRef}
      className="absolute inset-0 w-full h-full"
      aria-label={`Annotations on page ${pageNumber}`}
      onPointerDown={(e) => {
        const rect = e.currentTarget.getBoundingClientRect();
        dragRef.current = { x: (e.clientX - rect.left) / rect.width, y: (e.clientY - rect.top) / rect.height };
      }}
      onPointerUp={(e) => {
        if (!dragRef.current) return;
        const rect = e.currentTarget.getBoundingClientRect();
        const x2 = (e.clientX - rect.left) / rect.width;
        const y2 = (e.clientY - rect.top) / rect.height;
        const x = Math.min(dragRef.current.x, x2);
        const y = Math.min(dragRef.current.y, y2);
        const w = Math.abs(x2 - dragRef.current.x);
        const h = Math.abs(y2 - dragRef.current.y);
        dragRef.current = null;
        if (w > 0.005 && h > 0.005) onCreate({ x, y, w, h });
      }}
    >
      {annotations.map((a) => (
        <rect
          key={a.id}
          x={`${a.rect.x * 100}%`}
          y={`${a.rect.y * 100}%`}
          width={`${a.rect.w * 100}%`}
          height={`${a.rect.h * 100}%`}
          fill={a.kind === 'highlight' ? 'rgba(250,204,21,0.4)' : 'none'}
          stroke={a.kind === 'shape' ? 'currentColor' : 'none'}
          strokeWidth={2}
        />
      ))}
    </svg>
  );
}
```

```ts
export const pdfMenuItem = {
  command: '/pdf',
  title: 'PDF',
  description: 'Embed a PDF with annotations',
  run: async (editor: import('@tiptap/core').Editor): Promise<void> => {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = 'application/pdf';
    input.onchange = async () => {
      const file = input.files?.[0];
      if (!file) return;
      const fd = new FormData();
      fd.append('file', file);
      const res = await fetch('/api/files', { method: 'POST', body: fd });
      if (!res.ok) return;
      const { id } = (await res.json()) as { id: string };
      editor.chain().focus().insertContent({ type: 'pdf', attrs: { fileId: id, defaultPage: 1 } }).run();
    };
    input.click();
  },
};
```

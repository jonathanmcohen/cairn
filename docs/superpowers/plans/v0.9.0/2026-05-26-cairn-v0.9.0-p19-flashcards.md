**Goal:** Spaced-repetition flashcards as a first-class block. Users embed `Flashcard` nodes (front/back/deck_tag) anywhere; the system schedules per-user reviews via the SM-2 algorithm; a sidebar "Review due" counter and a study session route `/flashcards/study?deck=<tag>` drive the review loop; the first card-due-of-day fires a v0.8 notification bell + lands in the v0.6 email digest.

**Architecture:** Migration `0043_flashcards.sql` creates `flashcard_cards` (one row per editor block) + `flashcard_reviews` (per-user scheduling state). The SM-2 algorithm at `src/lib/flashcards/sm2.ts` is pure: `scheduleNext(review, grade) → { ease, interval, due_at }`. A cron job (registered with the v0.7 P14 scheduler) fires daily at 09:00 workspace-local: any user with `count(due_reviews) > 0` and no notification already fired today gets a notification + a row in the v0.6 P11 email digest. TipTap node `Flashcard` upserts its row into `flashcard_cards` on save via a server-side block listener.

**Tech Stack:** Drizzle ORM, Postgres 16, TipTap 3, Next.js 16 App Router (server actions for grade submission), Vitest 4 + Testcontainers, existing notification helper (`src/lib/notifications/...`), existing digest mailer (`src/lib/notifications/digest-mailer.ts`).

- [ ] **Step 3: Implement `src/lib/flashcards/sm2.ts`**

```ts
/**
 * SM-2 spaced-repetition algorithm.
 *
 *   grade 0 = Again (forgot)        → reset interval, lower ease
 *   grade 1 = Hard                   → keep ease, small interval bump
 *   grade 2 = Good                   → standard interval bump
 *   grade 3 = Easy                   → raise ease, larger interval bump
 *
 * Interval recurrence (grade ≥ 1):
 *   - first review (interval=0):    1 day
 *   - second review (interval=1):   6 days
 *   - otherwise:                    round(interval * ease)
 *
 * Ease delta per grade:
 *   - 0: -0.20 (clamped to ≥ 1.3)
 *   - 1: -0.15
 *   - 2:  0
 *   - 3: +0.15
 */
export type ReviewState = { ease: number; interval: number };
export type ReviewNext = {
  ease: number;
  interval: number;
  dueAt: Date;
  lastReviewedAt: Date;
  lastGrade: number;
};

const DAY_MS = 86_400_000;
const MIN_EASE = 1.3;
const EASE_DELTA: Record<number, number> = { 0: -0.2, 1: -0.15, 2: 0, 3: 0.15 };

export function scheduleNext(
  prev: ReviewState,
  grade: 0 | 1 | 2 | 3,
  now: Date = new Date(),
): ReviewNext {
  const newEase = Math.max(MIN_EASE, prev.ease + (EASE_DELTA[grade] ?? 0));
  if (grade === 0) {
    return {
      ease: newEase,
      interval: 0,
      dueAt: now,
      lastReviewedAt: now,
      lastGrade: grade,
    };
  }
  let interval: number;
  if (prev.interval === 0) interval = 1;
  else if (prev.interval === 1) interval = 6;
  else interval = Math.round(prev.interval * newEase);
  return {
    ease: newEase,
    interval,
    dueAt: new Date(now.getTime() + interval * DAY_MS),
    lastReviewedAt: now,
    lastGrade: grade,
  };
}
```

import { and, eq } from 'drizzle-orm';
import type { Database } from '@/db/client';
import { flashcardCards } from '@/db/schema';

export type UpsertCardInput = {
  pageId: string;
  blockId: string;
  front: string;
  back: string;
  deckTag: string | null;
  createdBy: string;
};

export async function upsertCard(db: Database, input: UpsertCardInput) {
  const [existing] = await db
    .select()
    .from(flashcardCards)
    .where(and(eq(flashcardCards.pageId, input.pageId), eq(flashcardCards.blockId, input.blockId)))
    .limit(1);
  if (existing) {
    const [row] = await db
      .update(flashcardCards)
      .set({
        front: input.front,
        back: input.back,
        deckTag: input.deckTag,
        updatedAt: new Date(),
      })
      .where(eq(flashcardCards.id, existing.id))
      .returning();
    return row!;
  }
  const [row] = await db.insert(flashcardCards).values(input).returning();
  return row!;
}

import { and, eq, isNull, lte, or, sql } from 'drizzle-orm';
import type { Database } from '@/db/client';
import { flashcardCards, flashcardReviews } from '@/db/schema';

export type DueCard = {
  id: string;
  pageId: string;
  blockId: string;
  front: string;
  back: string;
  deckTag: string | null;
  ease: number;
  interval: number;
};

export async function listDueForUser(
  db: Database,
  userId: string,
  opts: { deckTag?: string } = {},
): Promise<DueCard[]> {
  const now = new Date();
  const rows = await db
    .select({
      id: flashcardCards.id,
      pageId: flashcardCards.pageId,
      blockId: flashcardCards.blockId,
      front: flashcardCards.front,
      back: flashcardCards.back,
      deckTag: flashcardCards.deckTag,
      ease: flashcardReviews.ease,
      interval: flashcardReviews.interval,
      dueAt: flashcardReviews.dueAt,
    })
    .from(flashcardCards)
    .leftJoin(
      flashcardReviews,
      and(eq(flashcardReviews.cardId, flashcardCards.id), eq(flashcardReviews.userId, userId)),
    )
    .where(
      and(
        opts.deckTag ? eq(flashcardCards.deckTag, opts.deckTag) : sql`true`,
        or(isNull(flashcardReviews.dueAt), lte(flashcardReviews.dueAt, now)),
      ),
    );
  return rows.map((r) => ({
    id: r.id,
    pageId: r.pageId,
    blockId: r.blockId,
    front: r.front,
    back: r.back,
    deckTag: r.deckTag,
    ease: r.ease ?? 2.5,
    interval: r.interval ?? 0,
  }));
}

export async function countDueForUser(db: Database, userId: string): Promise<number> {
  const rows = await listDueForUser(db, userId);
  return rows.length;
}

import { NextResponse } from 'next/server';
import { and, eq } from 'drizzle-orm';
import { z } from 'zod';
import { auth } from '@/lib/auth/config';
import { getDb } from '@/db/client';
import { flashcardReviews } from '@/db/schema';
import { scheduleNext } from '@/lib/flashcards/sm2';

export const runtime = 'nodejs';

const body = z.object({
  cardId: z.uuid(),
  grade: z.union([z.literal(0), z.literal(1), z.literal(2), z.literal(3)]),
});

export async function POST(req: Request): Promise<Response> {
  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
  const parsed = body.safeParse(await req.json());
  if (!parsed.success) return NextResponse.json({ error: parsed.error.format() }, { status: 400 });
  const db = getDb();
  const [existing] = await db
    .select()
    .from(flashcardReviews)
    .where(and(eq(flashcardReviews.cardId, parsed.data.cardId), eq(flashcardReviews.userId, session.user.id)))
    .limit(1);
  const prev = existing ? { ease: existing.ease, interval: existing.interval } : { ease: 2.5, interval: 0 };
  const next = scheduleNext(prev, parsed.data.grade);
  if (existing) {
    await db
      .update(flashcardReviews)
      .set({
        ease: next.ease,

async function reconcileFlashcards(db: Database, pageId: string, userId: string, doc: unknown): Promise<void> {
  const found: Array<{ blockId: string; front: string; back: string; deck_tag: string | null }> = [];
  function walk(n: { type?: string; attrs?: Record<string, unknown>; content?: unknown[] }): void {
    if (n.type === 'flashcard' && n.attrs?.blockId) {
      found.push({
        blockId: String(n.attrs.blockId),
        front: String(n.attrs.front ?? ''),
        back: String(n.attrs.back ?? ''),
        deck_tag: (n.attrs.deck_tag as string | null) ?? null,
      });
    }
    for (const c of (n.content as Array<{ type?: string }> | undefined) ?? []) walk(c);
  }
  walk(doc as { type?: string });
  for (const f of found) {
    await upsertCard(db, { pageId, blockId: f.blockId, front: f.front, back: f.back, deckTag: f.deck_tag, createdBy: userId });
  }
  // Delete cards whose blockId no longer appears in the doc.
  const blockIds = found.map((f) => f.blockId);
  await db
    .delete(flashcardCards)
    .where(
      and(
        eq(flashcardCards.pageId, pageId),
        blockIds.length > 0 ? notInArray(flashcardCards.blockId, blockIds) : undefined as never,
      ),
    );
}

- [ ] **Step 3: Implement `src/lib/flashcards/notify-due.ts`**

```ts
import { and, eq, gte, isNull, lte, or, sql } from 'drizzle-orm';
import type { Database } from '@/db/client';
import { flashcardCards, flashcardReviews, notifications, users } from '@/db/schema';

export type NotifyResult = { notified: number };

export async function notifyDueFlashcards(db: Database): Promise<NotifyResult> {
  const todayStart = new Date();
  todayStart.setUTCHours(0, 0, 0, 0);

  // Users with at least one due card.
  const dueUsers = await db
    .selectDistinct({ userId: users.id })
    .from(users)
    .innerJoin(flashcardCards, eq(flashcardCards.createdBy, users.id))
    .leftJoin(
      flashcardReviews,
      and(eq(flashcardReviews.cardId, flashcardCards.id), eq(flashcardReviews.userId, users.id)),
    )
    .where(or(isNull(flashcardReviews.dueAt), lte(flashcardReviews.dueAt, new Date())));

  let count = 0;
  for (const { userId } of dueUsers) {
    const [already] = await db
      .select({ id: notifications.id })
      .from(notifications)
      .where(
        and(
          eq(notifications.userId, userId),
          eq(notifications.kind, 'flashcards_due'),
          gte(notifications.createdAt, todayStart),
        ),
      )
      .limit(1);
    if (already) continue;
    await db.insert(notifications).values({
      userId,
      kind: 'flashcards_due',
      body: sql`'{"message":"You have flashcards due for review."}'::jsonb`,
    });
    count++;
  }
  return { notified: count };
}
```

# Cairn v0.9.0 — Plan P34: Static export pipeline + CLI + admin UI + MkDocs target

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [x]`) syntax for tracking.

**Goal:** Ship a generic workspace-to-static-site export pipeline + a CLI entrypoint + an admin UI button + the **MkDocs** output target. The Docusaurus target lands in the sibling P35; the pipeline is engineered today so P35 only adds a new `target` strategy.

The pipeline walks a workspace's page tree, serializes each page to markdown via the existing `proseToMarkdown` (`src/lib/markdown/from-prose.ts`), downloads every referenced asset (images / PDFs / audio / video) via signed URLs into `docs/assets/`, rewrites in-doc asset links to relative paths, emits a target-specific frontmatter header on each page, builds a buildable project tree (mkdocs.yml + docs/ + assets/ for MkDocs), and zips the result with `archiver` (already a runtime dep).

**Architecture:**

- `src/lib/export/static-site.ts` — `exportWorkspace(workspaceId, target, opts) → Readable` orchestrator. Streams a zip via `archiver`'s `ZipArchive` (already used by `src/lib/markdown/export-subtree.ts`).
- `src/lib/export/page-walk.ts` — recursive-CTE walk over `pages` filtered by `workspace_id`, ordered by parent → child → position, returning `{id, parentId, title, content, slug, depth}`. Pulls cover URLs separately.
- `src/lib/export/assets.ts` — scans a TipTap doc for `image` / `file` / `videoBlock` / `audioBlock` nodes (the v0.8 + v0.9 set), reads `attrs.src` (HMAC-signed `/api/files/[id]?sig=&exp=`), resolves the underlying file via `files.id`, and emits `{ srcPath, destPath, mimeType, bytes }` records. The pipeline rewrites the in-doc `src` to `./assets/<filename>` before serializing to markdown.
- `src/lib/export/frontmatter.ts` — `applyFrontmatter(target, page, opts) → string` returns markdown with a YAML frontmatter block prepended (via `gray-matter`'s `stringify`). MkDocs profile uses `title` + `nav_order`. The Docusaurus profile (P35) reuses this with a different shape.
- `src/lib/export/targets/mkdocs.ts` — `buildMkDocsTree({pages, assets, workspaceName}) → { 'mkdocs.yml': string, 'docs/<path>.md': string, 'docs/assets/<file>': Buffer }` map. The orchestrator writes this map into the zip archive.
- `scripts/cairn-export.ts` — node CLI (`pnpm exec tsx scripts/cairn-export.ts --workspace <id> --target mkdocs --out path/to.zip`). Pure orchestration: parse args, call `exportWorkspace`, stream to fs.
- `src/app/(app)/settings/workspace/export-static-site/page.tsx` — admin UI with a target selector (MkDocs only in P34; P35 unlocks Docusaurus) + a "Generate" button that fetches `POST /api/exports/static-site` → server streams the zip back → browser downloads.
- `src/app/api/exports/static-site/route.ts` — POST handler. Auth via `requireRole('admin')`. Body: `{ workspaceId, target }`. Streams the zip with `Content-Type: application/zip` + `Content-Disposition: attachment; filename=cairn-<workspace>-<target>.zip`.

```ts
import { sql as rawSql } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';
import type * as schema from '@/db/schema';

export type WalkedPage = {
  id: string;
  parentId: string | null;
  title: string;
  content: unknown;
  depth: number;
  position: number;
};

/**
 * Recursive-CTE walk over `pages` filtered to `workspaceId`. Depth-first
 * pre-order: a parent appears before any of its descendants, and siblings
 * are sorted by `position`. Soft-deleted pages are excluded.
 *
 * Note (per CLAUDE.md): recursive CTEs go through `db.execute(rawSql)`;
 * Drizzle's builder can't express them. The result is cast to the row shape.
 */
export async function walkWorkspacePages(
  db: PostgresJsDatabase<typeof schema>,
  workspaceId: string,
): Promise<WalkedPage[]> {
  const rows = (await db.execute(rawSql`
    WITH RECURSIVE tree AS (
      SELECT id, parent_id, title, content, position, 0 AS depth,
             ARRAY[position]::int[] AS path
      FROM pages
      WHERE workspace_id = ${workspaceId}::uuid
        AND parent_id IS NULL
        AND deleted_at IS NULL
      UNION ALL
      SELECT p.id, p.parent_id, p.title, p.content, p.position, t.depth + 1,
             t.path || p.position
      FROM pages p
      INNER JOIN tree t ON p.parent_id = t.id
      WHERE p.workspace_id = ${workspaceId}::uuid
        AND p.deleted_at IS NULL
    )
    SELECT id, parent_id, title, content, depth, position
    FROM tree
    ORDER BY path;
  `)) as unknown as Array<{
    id: string;
    parent_id: string | null;
    title: string;
    content: unknown;
    depth: number;
    position: number;
  }>;
  return rows.map((r) => ({
    id: r.id,
    parentId: r.parent_id,
    title: r.title,
    content: r.content,
    depth: r.depth,
    position: r.position,
  }));
}
```

```ts
import { inArray, sql } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';
import * as schema from '@/db/schema';

export type AssetRef = {
  fileId: string;
  destFilename: string;
  mimeType: string | null;
  storageKey: string;
};

type Node = {
  type?: string;
  attrs?: Record<string, unknown>;
  content?: Node[];
};

const MEDIA_NODE_TYPES = new Set(['image', 'file', 'videoBlock', 'audioBlock', 'pdfBlock']);
const FILE_URL_RE = /^\/api\/files\/([0-9a-f-]{36})(?:\?|$)/;

function collectMediaSrcs(node: Node, acc: { node: Node; fileId: string }[]): void {
  if (!node || typeof node !== 'object') return;
  if (node.type && MEDIA_NODE_TYPES.has(node.type)) {
    const src = node.attrs?.src;
    if (typeof src === 'string') {
      const m = src.match(FILE_URL_RE);
      if (m) acc.push({ node, fileId: m[1]! });
    }
  }
  if (Array.isArray(node.content)) {
    for (const child of node.content) collectMediaSrcs(child, acc);
  }
}

function sanitize(filename: string): string {
  return filename.replace(/[^A-Za-z0-9._-]+/g, '_').slice(0, 80) || 'file';
}

/**
 * Mutates `doc` in place: every media node's `attrs.src` becomes
 * `./assets/<unique-filename>`. Returns the rewritten doc + the list of asset
 * refs the orchestrator must download + write into `docs/assets/`.
 *
 * Filename collision avoidance: prefix every dest filename with the file id so
 * two files with the same `original_name` never collide.
 */
export async function extractAndRewriteAssets(
  db: PostgresJsDatabase<typeof schema>,
  workspaceId: string,
  doc: unknown,
): Promise<{ rewritten: unknown; assets: AssetRef[] }> {
  const hits: { node: Node; fileId: string }[] = [];
  collectMediaSrcs(doc as Node, hits);
  if (hits.length === 0) return { rewritten: doc, assets: [] };

  const fileIds = [...new Set(hits.map((h) => h.fileId))];
  const rows = await db
    .select({
      id: schema.files.id,
      workspaceId: schema.files.workspaceId,
      storageKey: schema.files.storageKey,
      originalName: schema.files.originalName,
      mimeType: schema.files.mimeType,
    })
    .from(schema.files)
    .where(inArray(schema.files.id, fileIds));

  const byId = new Map(rows.map((r) => [r.id, r]));
  const assets: AssetRef[] = [];
  for (const { node, fileId } of hits) {
    const f = byId.get(fileId);
    if (!f || f.workspaceId !== workspaceId) continue; // cross-workspace ref is dropped silently
    const dest = `${fileId}-${sanitize(f.originalName ?? 'file')}`;
    node.attrs = { ...(node.attrs ?? {}), src: `./assets/${dest}` };
    assets.push({
      fileId,
      destFilename: dest,
      mimeType: f.mimeType,
      storageKey: f.storageKey,
    });
  }
  // De-duplicate by destFilename.
  const seen = new Set<string>();
  const uniq = assets.filter((a) => (seen.has(a.destFilename) ? false : (seen.add(a.destFilename), true)));
  return { rewritten: doc, assets: uniq };
}
```

```ts
import matter from 'gray-matter';

export type ExportTarget = 'mkdocs' | 'docusaurus';

export type FrontmatterInput = {
  id: string;
  title: string;
  slug: string;
  position: number;
  depth: number;
};

function buildMkdocs(input: FrontmatterInput): Record<string, unknown> {
  return {
    title: input.title,
    nav_order: input.position,
  };
}

function buildDocusaurus(input: FrontmatterInput): Record<string, unknown> {
  // Placeholder for P35 — exports the field shape, but P34 should never call
  // through this branch. Implemented here so P35's diff is purely additive.
  return {
    id: input.slug,
    slug: `/${input.slug}`,
    title: input.title,
    sidebar_position: input.position,
  };
}

export function applyFrontmatter(target: ExportTarget, input: FrontmatterInput, body: string): string {
  let data: Record<string, unknown>;
  switch (target) {
    case 'mkdocs':
      data = buildMkdocs(input);
      break;
    case 'docusaurus':
      data = buildDocusaurus(input);
      break;
    default: {
      const _exhaustive: never = target;
      throw new Error(`unknown export target: ${String(_exhaustive)}`);
    }
  }
  return matter.stringify(body, data);
}
```

```ts
import { dump as yamlDump } from 'js-yaml';
import type { AssetRef } from '../assets';
import { applyFrontmatter } from '../frontmatter';
import type { WalkedPage } from '../page-walk';

export type RenderedPage = WalkedPage & { slug: string; markdown: string };

export type MkDocsTree = {
  files: Record<string, string | Buffer>;
};

function slugify(title: string, id: string): string {
  const base = title
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '')
    .slice(0, 50);
  return base ? `${base}-${id.slice(0, 8)}` : id.slice(0, 8);
}

function buildNav(pages: RenderedPage[]): unknown[] {
  // MkDocs nav supports nested arrays of {title: filename} entries. We build a
  // map by id → children and emit one entry per page in pre-order. Children of
  // a parent are emitted as a nested array under the parent's title.
  type NavEntry = Record<string, string | NavEntry[]>;
  const byId = new Map(pages.map((p) => [p.id, p]));
  const childrenByParent = new Map<string | null, RenderedPage[]>();
  for (const p of pages) {
    const arr = childrenByParent.get(p.parentId) ?? [];
    arr.push(p);
    childrenByParent.set(p.parentId, arr);
  }
  function emit(parentId: string | null): NavEntry[] {
    const kids = childrenByParent.get(parentId) ?? [];
    return kids.map<NavEntry>((k) => {
      const own: NavEntry = { [k.title]: `${k.slug}.md` };
      const grandkids = emit(k.id);
      if (grandkids.length === 0) return own;
      // Parent + children: wrap as a nested entry with the parent as the first item.
      return { [k.title]: [{ Index: `${k.slug}.md` }, ...grandkids] };
    });
  }
  // Suppress unused; keeps the local binding for symmetry.
  void byId;
  return emit(null);
}

export function buildMkDocsTree(args: {
  workspaceName: string;
  pages: WalkedPage[];
  assets: Array<AssetRef & { contents: Buffer }>;
}): MkDocsTree {
  const rendered: RenderedPage[] = args.pages.map((p) => {
    const slug = slugify(p.title, p.id);
    // proseToMarkdown is called by the orchestrator BEFORE buildMkDocsTree —
    // here we expect each page to already carry a `markdown` string. The
    // public type therefore extends WalkedPage with `markdown` — but to keep
    // this builder pure, we accept the markdown via the orchestrator pre-pass.
    return { ...p, slug, markdown: '' };
  });

  const files: Record<string, string | Buffer> = {};
  for (const p of rendered) {
    files[`docs/${p.slug}.md`] = applyFrontmatter('mkdocs', { id: p.id, title: p.title, slug: p.slug, position: p.position, depth: p.depth }, p.markdown);
  }
  for (const a of args.assets) {
    files[`docs/assets/${a.destFilename}`] = a.contents;
  }
  files['mkdocs.yml'] = yamlDump({
    site_name: args.workspaceName,
    docs_dir: 'docs',
    nav: buildNav(rendered),
    theme: { name: 'material' },
  });
  return { files };
}
```

```ts
import { PassThrough, type Readable } from 'node:stream';
import { ZipArchive } from 'archiver';
import { eq } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';
import * as schema from '@/db/schema';
import { getStorage } from '@/lib/files/storage';
import { proseToMarkdown } from '@/lib/markdown/from-prose';
import { extractAndRewriteAssets } from './assets';
import { applyFrontmatter, type ExportTarget } from './frontmatter';
import { walkWorkspacePages } from './page-walk';
import { buildMkDocsTree } from './targets/mkdocs';

export type ExportArgs = {
  workspaceId: string;
  target: ExportTarget;
};

function slugify(title: string, id: string): string {
  const base = title
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '')
    .slice(0, 50);
  return base ? `${base}-${id.slice(0, 8)}` : id.slice(0, 8);
}

export async function exportWorkspace(
  db: PostgresJsDatabase<typeof schema>,
  args: ExportArgs,
): Promise<Readable> {
  // 1. Workspace metadata (site_name etc.).
  const [ws] = await db
    .select({ name: schema.workspaces.name })
    .from(schema.workspaces)
    .where(eq(schema.workspaces.id, args.workspaceId))
    .limit(1);
  if (!ws) throw new Error(`workspace not found: ${args.workspaceId}`);

  // 2. Walk + render every page.
  const pages = await walkWorkspacePages(db, args.workspaceId);
  const allAssets: Array<{ destFilename: string; storageKey: string; contents: Buffer; fileId: string; mimeType: string | null }> = [];
  const rendered: Array<{
    id: string;
    parentId: string | null;
    title: string;
    slug: string;
    position: number;
    depth: number;
    markdown: string;
  }> = [];

  const storage = getStorage();
  for (const p of pages) {
    const { rewritten, assets } = await extractAndRewriteAssets(db, args.workspaceId, p.content);
    for (const a of assets) {
      if (allAssets.some((x) => x.destFilename === a.destFilename)) continue;
      const buf = await storage.get(a.storageKey);
      allAssets.push({ ...a, contents: buf });
    }
    const slug = slugify(p.title, p.id);
    const md = proseToMarkdown(rewritten);
    rendered.push({
      id: p.id,
      parentId: p.parentId,
      title: p.title,
      slug,
      position: p.position,
      depth: p.depth,
      markdown: md,
    });
  }

  // 3. Build target tree.
  let files: Record<string, string | Buffer>;
  switch (args.target) {
    case 'mkdocs': {
      const tree = buildMkDocsTree({
        workspaceName: ws.name,
        pages: rendered.map((r) => ({ id: r.id, parentId: r.parentId, title: r.title, content: null, position: r.position, depth: r.depth })),
        assets: allAssets,
      });
      // Replace empty markdown bodies with the real rendered text.
      files = { ...tree.files };
      for (const r of rendered) {
        files[`docs/${r.slug}.md`] = applyFrontmatter(
          'mkdocs',
          { id: r.id, title: r.title, slug: r.slug, position: r.position, depth: r.depth },
          r.markdown,
        );
      }
      break;
    }
    case 'docusaurus':
      // P35 wires this branch.
      throw new Error('docusaurus target not implemented in P34');
    default: {
      const _exh: never = args.target;
      throw new Error(`unknown target: ${String(_exh)}`);
    }
  }

  // 4. Stream as a zip.
  const pass = new PassThrough();
  const archive = new ZipArchive({ zlib: { level: 9 } });
  archive.pipe(pass);
  for (const [path, body] of Object.entries(files)) {
    archive.append(body, { name: path });
  }
  void archive.finalize();
  return pass;
}
```

```ts
/**
 * Cairn static-site export CLI. Pure orchestration: parse args, call
 * exportWorkspace, stream to fs.
 *
 *   pnpm exec tsx scripts/cairn-export.ts --workspace <uuid> --target mkdocs --out out.zip
 */
import { createWriteStream } from 'node:fs';
import { pipeline } from 'node:stream/promises';
import { getDb } from '@/db/client';
import { exportWorkspace } from '@/lib/export/static-site';

type Args = { workspace?: string; target?: 'mkdocs' | 'docusaurus'; out?: string };

function parseArgs(argv: string[]): Args {
  const out: Args = {};
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i]!;
    const v = argv[i + 1];
    if (a === '--workspace' && v) {
      out.workspace = v;
      i++;
    } else if (a === '--target' && (v === 'mkdocs' || v === 'docusaurus')) {
      out.target = v;
      i++;
    } else if (a === '--out' && v) {
      out.out = v;
      i++;
    }
  }
  return out;
}

async function main(): Promise<void> {
  const args = parseArgs(process.argv.slice(2));
  if (!args.workspace || !args.target || !args.out) {
    process.stderr.write('usage: cairn-export --workspace <uuid> --target mkdocs|docusaurus --out path.zip\n');
    process.exit(2);
  }
  const db = getDb();
  const stream = await exportWorkspace(db, { workspaceId: args.workspace, target: args.target });
  await pipeline(stream, createWriteStream(args.out));
  process.stderr.write(`wrote ${args.out}\n`);
}

void main();
```

```ts
import type { Readable } from 'node:stream';
import { NextResponse } from 'next/server';
import { z } from 'zod';
import { getDb } from '@/db/client';
import { HttpError, requireRole } from '@/lib/auth/require-role';
import { exportWorkspace } from '@/lib/export/static-site';

const Body = z.object({
  workspaceId: z.uuid(),
  target: z.enum(['mkdocs', 'docusaurus']),
});

export async function POST(req: Request): Promise<Response> {
  try {
    const body = Body.parse(await req.json());
    const ctx = await requireRole(body.workspaceId, 'admin');
    void ctx;
    const stream = await exportWorkspace(getDb(), {
      workspaceId: body.workspaceId,
      target: body.target,
    });
    // Convert Node Readable → web Response (per CLAUDE.md gotcha).
    // @ts-expect-error — Next 16 accepts a Node Readable as the body init.
    const res = new Response(stream as Readable, {
      status: 200,
      headers: {
        'content-type': 'application/zip',
        'content-disposition': `attachment; filename="cairn-${body.workspaceId.slice(0, 8)}-${body.target}.zip"`,
      },
    });
    return res;
  } catch (err) {
    if (err instanceof HttpError) {
      return NextResponse.json({ error: err.message }, { status: err.status });
    }
    if (err instanceof z.ZodError) {
      return NextResponse.json({ error: 'invalid body' }, { status: 400 });
    }
    return NextResponse.json(
      { error: err instanceof Error ? err.message : 'unknown' },
      { status: 500 },
    );
  }
}
```

# Cairn v0.9.0 — Plan P22: Bulk file drag-drop + audio block + extended MIME allowlist

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Drag-drop ≥2 files into the editor at once → progress modal shows per-file status (pending → uploading → done | failed) → on completion each file lands as the correct block type (image → `cairnImage`, pdf → `pdf` (v0.9 P17), audio → `audio` (NEW), video → `video` (v0.8 P24), anything else → `file`). Parallel upload concurrency capped at 4. Audio block is a new TipTap node that renders `<audio controls src={signedUrl}>` using the same HMAC-signed URL pipeline as the v0.8 video node. MIME allowlist extended to cover the five common audio formats.

**Architecture:** A new `BulkUploader` client component owns the modal + state machine. The existing editor drop handler (v0.5 P2 image-paste flow + v0.6 P10 file-drop) gets an early branch: if the dragged file count is ≥2, prevent the default insertion and open the bulk modal instead, passing the `FileList` and the editor instance. The modal walks the file list, classifies each by MIME, posts to `/api/upload` (existing endpoint, MIME-allowlist extended in `src/lib/files/upload.ts`), and once each finishes calls `editor.chain().insertContent(...)` with the appropriate node spec at the end of the document (or at the original drop position, captured in `localStorage` before the modal opens). Audio node mirrors v0.8 video node's signed-URL render path.

**Tech Stack:** TipTap 3 (existing), `@radix-ui/react-dialog` (existing via shadcn), Vitest + jsdom for component tests, fake `<input type="file">` via `DataTransfer` in jsdom-extended.

import { mergeAttributes, Node } from '@tiptap/core';
import { ReactNodeViewRenderer } from '@tiptap/react';
import { AudioView } from './audio-view';

export type AudioAttrs = {
  fileId: string;
  mime: string;
  name?: string;
};

export const AudioNode = Node.create({
  name: 'cairnAudio',
  group: 'block',
  atom: true,
  draggable: true,
  selectable: true,

  addAttributes() {
    return {
      fileId: {
        default: '',
        parseHTML: (el) => (el as HTMLElement).getAttribute('data-file-id') ?? '',
        renderHTML: (attrs) => ({ 'data-file-id': attrs.fileId }),
      },
      mime: {
        default: 'audio/mpeg',
        parseHTML: (el) => (el as HTMLElement).getAttribute('data-mime') ?? 'audio/mpeg',
        renderHTML: (attrs) => ({ 'data-mime': attrs.mime }),
      },
      name: {
        default: null,
        parseHTML: (el) => (el as HTMLElement).getAttribute('data-name'),
        renderHTML: (attrs) =>
          attrs.name ? { 'data-name': attrs.name } : {},
      },
    };
  },

  parseHTML() {
    return [{ tag: 'div[data-cairn-audio]' }];
  },

  renderHTML({ HTMLAttributes }) {
    return [
      'div',
      mergeAttributes(HTMLAttributes, { 'data-cairn-audio': '' }),
    ];
  },

  addNodeView() {
    return ReactNodeViewRenderer(AudioView);
  },
});

```tsx
'use client';

import { NodeViewWrapper, type NodeViewProps } from '@tiptap/react';
import { useEffect, useState } from 'react';

export function AudioView(props: NodeViewProps): JSX.Element {
  const attrs = props.node.attrs as { fileId: string; mime: string; name?: string };
  const [signedUrl, setSignedUrl] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (!attrs.fileId) return;
    const ac = new AbortController();
    fetch(`/api/files/${attrs.fileId}/sign`, { signal: ac.signal })
      .then(async (r) => {
        if (!r.ok) throw new Error(`sign failed (${r.status})`);
        return (await r.json()) as { url: string };
      })
      .then((data) => setSignedUrl(data.url))
      .catch((err: unknown) => {
        if ((err as Error).name !== 'AbortError') setError((err as Error).message);
      });
    return () => ac.abort();
  }, [attrs.fileId]);

  return (
    <NodeViewWrapper className="cairn-audio my-2" data-drag-handle>
      {error && <p className="text-sm text-destructive">Audio unavailable: {error}</p>}
      {signedUrl ? (
        <audio controls preload="metadata" src={signedUrl} className="w-full">
          <track kind="captions" />
        </audio>
      ) : (
        !error && <div className="h-12 animate-pulse rounded bg-muted" />
      )}
      {attrs.name && <p className="text-xs text-muted-foreground mt-1">{attrs.name}</p>}
    </NodeViewWrapper>
  );
}
```

```tsx
'use client';

import { useEffect, useId, useRef, useState } from 'react';
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog';
import { Progress } from '@/components/ui/progress';

const MAX_CONCURRENT = 4;

export type BulkKind = 'image' | 'audio' | 'video' | 'pdf' | 'file';

export type BulkResult = {
  name: string;
  kind: BulkKind;
  status: 'done' | 'failed';
  fileId?: string;
  error?: string;
  mime: string;
};

function classifyKind(mime: string): BulkKind {
  if (mime.startsWith('image/')) return 'image';
  if (mime.startsWith('audio/')) return 'audio';
  if (mime.startsWith('video/')) return 'video';
  if (mime === 'application/pdf') return 'pdf';
  return 'file';
}

type Row = {
  file: File;
  kind: BulkKind;
  status: 'pending' | 'uploading' | 'done' | 'failed';
  fileId?: string;
  error?: string;
};

async function uploadOne(file: File): Promise<{ id: string }> {
  const form = new FormData();
  form.append('file', file);
  const res = await fetch('/api/upload', { method: 'POST', body: form });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return (await res.json()) as { id: string };
}

export function BulkUploader(props: {
  open: boolean;
  files: File[];
  onOpenChange: (open: boolean) => void;
  onComplete: (results: BulkResult[]) => void;
}): JSX.Element {
  const [rows, setRows] = useState<Row[]>(() =>
    props.files.map((file) => ({ file, kind: classifyKind(file.type), status: 'pending' })),
  );
  const calledOnComplete = useRef(false);
  const titleId = useId();

  useEffect(() => {
    if (!props.open) return;
    if (calledOnComplete.current) return;

    let cancelled = false;
    const queue = [...props.files.keys()];
    let running = 0;

    const update = (i: number, patch: Partial<Row>): void => {
      setRows((prev) => {
        const next = [...prev];
        next[i] = { ...next[i]!, ...patch };
        return next;
      });
    };

    const finishIfDone = (): void => {
      setRows((prev) => {
        const allTerminal = prev.every((r) => r.status === 'done' || r.status === 'failed');
        if (allTerminal && !calledOnComplete.current) {
          calledOnComplete.current = true;
          const results: BulkResult[] = prev.map((r) => ({
            name: r.file.name,
            kind: r.kind,
            status: r.status === 'done' ? 'done' : 'failed',
            fileId: r.fileId,
            error: r.error,
            mime: r.file.type,
          }));
          props.onComplete(results);
        }
        return prev;
      });
    };

    const pump = (): void => {
      while (running < MAX_CONCURRENT && queue.length > 0) {
        const i = queue.shift()!;
        running += 1;
        update(i, { status: 'uploading' });
        uploadOne(props.files[i]!)
          .then(({ id }) => {
            if (cancelled) return;
            update(i, { status: 'done', fileId: id });
          })
          .catch((err: unknown) => {
            if (cancelled) return;
            update(i, { status: 'failed', error: (err as Error).message });
          })
          .finally(() => {
            if (cancelled) return;
            running -= 1;
            pump();
            // schedule a microtask check so the row update has applied first
            queueMicrotask(finishIfDone);
          });
      }
    };

    pump();
    return () => {
      cancelled = true;
    };
  }, [props.open, props.files, props.onComplete]);

  const doneCount = rows.filter((r) => r.status === 'done' || r.status === 'failed').length;
  const pct = rows.length === 0 ? 100 : Math.round((doneCount / rows.length) * 100);

  return (
    <Dialog open={props.open} onOpenChange={props.onOpenChange}>
      <DialogContent aria-labelledby={titleId}>
        <DialogHeader>
          <DialogTitle id={titleId}>
            Uploading {rows.length} file{rows.length === 1 ? '' : 's'}
          </DialogTitle>
        </DialogHeader>
        <Progress value={pct} />
        <ul className="mt-3 space-y-1">
          {rows.map((row) => (
            <li key={row.file.name} className="flex items-center justify-between text-sm">
              <span className="truncate">{row.file.name}</span>
              <span
                className={
                  row.status === 'failed'
                    ? 'text-destructive'
                    : row.status === 'done'
                      ? 'text-muted-foreground'
                      : ''
                }
              >
                {row.status}
                {row.error ? ` — ${row.error}` : ''}
              </span>
            </li>
          ))}
        </ul>
      </DialogContent>
    </Dialog>
  );
}
```

- [x] **Step 1: Implement `src/components/editor/use-bulk-drop-handler.ts`**

```ts
'use client';

import type { Editor } from '@tiptap/core';
import { useEffect, useState } from 'react';
import type { BulkResult } from './bulk-uploader';

type Args = {
  editor: Editor | null;
};

type State = {
  open: boolean;
  files: File[];
  dropPos: number | null;
};

export function useBulkDropHandler({ editor }: Args): {
  open: boolean;
  files: File[];
  onOpenChange: (open: boolean) => void;
  onComplete: (results: BulkResult[]) => void;
} {
  const [state, setState] = useState<State>({ open: false, files: [], dropPos: null });

  useEffect(() => {
    if (!editor) return;
    const view = editor.view;
    const handler = (e: DragEvent): boolean => {
      const dt = e.dataTransfer;
      if (!dt) return false;
      const files = Array.from(dt.files);
      if (files.length < 2) return false;
      e.preventDefault();
      const coords = { left: e.clientX, top: e.clientY };
      const dropPos = view.posAtCoords(coords)?.pos ?? null;
      setState({ open: true, files, dropPos });
      return true;
    };
    view.dom.addEventListener('drop', handler);
    return () => view.dom.removeEventListener('drop', handler);
  }, [editor]);

  return {
    open: state.open,
    files: state.files,
    onOpenChange: (open) => setState((s) => ({ ...s, open })),
    onComplete: (results) => {
      if (!editor) return;
      const pos = state.dropPos ?? editor.state.doc.content.size;
      const nodes = results
        .filter((r) => r.status === 'done' && r.fileId)
        .map((r) => buildNode(r));
      editor.chain().insertContentAt(pos, nodes).run();
      setState({ open: false, files: [], dropPos: null });
    },
  };
}

function buildNode(r: BulkResult): { type: string; attrs: Record<string, unknown> } {
  const base = { fileId: r.fileId, mime: r.mime, name: r.name };
  switch (r.kind) {
    case 'image':
      return { type: 'cairnImage', attrs: { ...base } };
    case 'audio':
      return { type: 'cairnAudio', attrs: { ...base } };
    case 'video':
      return { type: 'cairnVideo', attrs: { ...base } };
    case 'pdf':
      return { type: 'pdf', attrs: { ...base } };
    default:
      return { type: 'file', attrs: { ...base } };
  }
}
```

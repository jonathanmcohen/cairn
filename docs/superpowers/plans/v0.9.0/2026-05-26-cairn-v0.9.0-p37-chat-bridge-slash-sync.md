# Cairn v0.9.0 — Plan P37: Chat bridge — slash commands + channel↔page sync (migration 0047)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extend v0.9 P36 chat-bridge (Slack + Discord install + outbound/inbound bot) with: (a) migration `0047_chat_bridge.sql` adding `chat_bridge_installs` + `chat_channel_links`, (b) slash commands `/cairn search <q>` (top-5 search results) and `/cairn create page <title>` (creates a page + returns link), (c) channel↔page sync mode where channel messages are appended as comments on a linked page and page comments are posted back to the channel. Dedupe so post-back comments don't echo. Per-workspace rate-limit. Admin UI at `/admin/chat-bridge/channels`.

**Architecture:** P36 introduced the `@slack/bolt` + `discord.js` install + bot-token surface. P37 adds two tables: `chat_bridge_installs` stores the install secrets (per-workspace, per-platform) including a Cairn-scoped PAT minted at install time (hashed). `chat_channel_links` ties a channel to a `pages.id` with a `link_mode` ∈ `('notify' | 'sync')`. Slash handlers live in `src/lib/chat/slash.ts` (platform-agnostic core + per-platform adapters). Sync engine in `src/lib/chat/sync.ts`: bidirectional, with a `chat_message_id` dedupe column on comments to avoid echo loops + per-workspace 30-msg/min rate-limit (Redis-style in-memory bucket, persisted in `chat_bridge_installs.options jsonb`).

**Tech Stack:** Drizzle + Postgres for storage, `@slack/bolt` + `discord.js` (deps from P36), `sanitize-html` (existing — v0.5.1 T3) for inbound message sanitization, Vitest + Testcontainers for tests.

**Depends on:** v0.5 P2 webhook infrastructure (delivery audit), v0.5 P1 + v0.7 PATs (PAT mint helper), v0.6 P20 pino REDACT_PATHS, v0.9 P36 chat-bridge install scaffold.

```ts
export type SlashCommand =
  | { kind: 'search'; query: string }
  | { kind: 'create_page'; title: string }
  | { kind: 'error'; message: string };

const MAX_TITLE = 255;
const MAX_QUERY = 200;

export function parseSlashCommand(raw: string): SlashCommand {
  const text = raw.trim().replace(/\s+/g, ' ');
  if (!text) return { kind: 'error', message: 'empty command' };

  if (text.startsWith('search ')) {
    const query = text.slice('search '.length).trim();
    if (!query) return { kind: 'error', message: 'search query required' };
    return { kind: 'search', query: query.slice(0, MAX_QUERY) };
  }
  if (text === 'search') return { kind: 'error', message: 'search query required' };

  if (text.startsWith('create page ')) {
    const title = text.slice('create page '.length).trim();
    if (!title) return { kind: 'error', message: 'page title required' };
    return { kind: 'create_page', title: title.slice(0, MAX_TITLE) };
  }
  if (text === 'create page') return { kind: 'error', message: 'page title required' };

  return { kind: 'error', message: `unknown command: ${text.split(' ')[0]}` };
}
```

```ts
import { createHmac } from 'node:crypto';

export function signSlack(body: string, signingSecret: string): { ts: string; v0: string } {
  const ts = Math.floor(Date.now() / 1000).toString();
  const base = `v0:${ts}:${body}`;
  const v0 = `v0=${createHmac('sha256', signingSecret).update(base).digest('hex')}`;
  return { ts, v0 };
}
```

```ts
import { and, eq } from 'drizzle-orm';
import { db } from '@/db/client';
import { chatBridgeInstalls } from '@/db/schema/chat-bridge';
import { pages } from '@/db/schema/pages';
import { searchPages } from '@/lib/pages/search';

export type SlashContext = {
  workspaceId: string;
  installId: string;
  invokingChatUserId: string;
  channelId: string;
};

export type SlashResponse =
  | { kind: 'message'; text: string; ephemeral: boolean }
  | { kind: 'message_with_link'; text: string; pageId: string; pageUrl: string };

export async function executeSlashCommand(
  cmd: ReturnType<typeof parseSlashCommand>,
  ctx: SlashContext,
  publicUrl: string,
): Promise<SlashResponse> {
  if (cmd.kind === 'error') {
    return { kind: 'message', text: `error: ${cmd.message}`, ephemeral: true };
  }
  if (cmd.kind === 'search') {
    const results = await searchPages({ workspaceId: ctx.workspaceId, query: cmd.query, limit: 5 });
    if (results.length === 0) {
      return { kind: 'message', text: `no results for "${cmd.query}"`, ephemeral: true };
    }
    const lines = results.map(
      (r, i) => `${i + 1}. <${publicUrl}/pages/${r.id}|${r.title}>`,
    );
    return {
      kind: 'message',
      text: [`Top ${results.length} results for "${cmd.query}":`, ...lines].join('\n'),
      ephemeral: true,
    };
  }
  // create_page
  const [install] = await db()
    .select()
    .from(chatBridgeInstalls)
    .where(eq(chatBridgeInstalls.id, ctx.installId))
    .limit(1);
  if (!install) {
    return { kind: 'message', text: 'install not found', ephemeral: true };
  }
  const [page] = await db()
    .insert(pages)
    .values({
      workspaceId: ctx.workspaceId,
      title: cmd.title,
      content: { type: 'doc', content: [] },
      createdBy: install.installedBy,
      updatedBy: install.installedBy,
    })
    .returning();
  return {
    kind: 'message_with_link',
    text: `Created "${cmd.title}": ${publicUrl}/pages/${page.id}`,
    pageId: page.id,
    pageUrl: `${publicUrl}/pages/${page.id}`,
  };
}
```

```ts
  // Reject if older than 5 minutes
  if (Math.abs(Date.now() / 1000 - Number(ts)) > 300) return false;
  const base = `v0:${ts}:${rawBody}`;
  const expected = `v0=${createHmac('sha256', secret).update(base).digest('hex')}`;
  if (expected.length !== sig.length) return false;
  return timingSafeEqual(Buffer.from(expected), Buffer.from(sig));
}

export async function POST(req: NextRequest): Promise<NextResponse> {
  const rawBody = await req.text();
  const params = new URLSearchParams(rawBody);
  const teamId = params.get('team_id') ?? '';
  const text = params.get('text') ?? '';
  const userId = params.get('user_id') ?? '';
  const channelId = params.get('channel_id') ?? '';

  const installs = await db()
    .select()
    .from(chatBridgeInstalls)
    .where(and(eq(chatBridgeInstalls.platform, 'slack'), eq(chatBridgeInstalls.teamId, teamId)))
    .limit(1);
  const install = installs[0];
  if (!install) {
    return NextResponse.json({ error: 'unknown team' }, { status: 401 });
  }
  if (!verifySlackSig(rawBody, req.headers, install.signingSecret)) {
    return NextResponse.json({ error: 'bad signature' }, { status: 401 });
  }
  const rl = await checkRateLimit({ workspaceId: install.workspaceId });
  if (!rl.allowed) {
    return NextResponse.json({ response_type: 'ephemeral', text: 'rate limit exceeded' });
  }
  const cmd = parseSlashCommand(text);
  const response = await executeSlashCommand(
    cmd,
    {
      workspaceId: install.workspaceId,
      installId: install.id,
      invokingChatUserId: userId,
      channelId,
    },
    env().PUBLIC_URL,
  );
  await recordAudit({
    workspaceId: install.workspaceId,
    actorUserId: install.installedBy,
    action: 'chat.slash_invoked',
    targetType: 'chat_install',
    targetId: install.id,
    metadata: { command: cmd.kind, channelId },
  });
  logger.info({ workspaceId: install.workspaceId, command: cmd.kind }, 'chat.slash_invoked');
  return NextResponse.json({
    response_type: 'ephemeral',
    text: response.text,
  });
}
```

```ts
const WINDOW_MS = 60_000;
const DEFAULT_LIMIT = 30;

type Bucket = { count: number; resetAt: number };
const buckets = new Map<string, Bucket>();

export async function checkRateLimit(input: {
  workspaceId: string;
  limit?: number;
}): Promise<{ allowed: boolean; remaining: number; resetAt: number }> {
  const limit = input.limit ?? DEFAULT_LIMIT;
  const now = Date.now();
  const key = input.workspaceId;
  const b = buckets.get(key);
  if (!b || b.resetAt <= now) {
    const fresh = { count: 1, resetAt: now + WINDOW_MS };
    buckets.set(key, fresh);
    return { allowed: true, remaining: limit - 1, resetAt: fresh.resetAt };
  }
  if (b.count >= limit) {
    return { allowed: false, remaining: 0, resetAt: b.resetAt };
  }
  b.count += 1;
  return { allowed: true, remaining: limit - b.count, resetAt: b.resetAt };
}

export function resetRateLimitForTests(): void {
  buckets.clear();
}
```

```ts
import { and, eq } from 'drizzle-orm';
import sanitizeHtml from 'sanitize-html';
import { db } from '@/db/client';
import { chatBridgeInstalls, chatChannelLinks } from '@/db/schema/chat-bridge';
import { isMessageProcessed, markMessageProcessed } from './dedupe';
import { checkRateLimit } from './ratelimit';

export type IngestResult =
  | { kind: 'inserted'; commentId: string }
  | { kind: 'duplicate' }
  | { kind: 'skipped_mode' }
  | { kind: 'skipped_bot' }
  | { kind: 'skipped_rate_limit' }
  | { kind: 'link_not_found' };

export async function ingestChannelMessage(input: {
  installId: string;
  channelId: string;
  messageId: string;
  authorChatUserId: string;
  body: string;
}): Promise<IngestResult> {
  const [install] = await db()
    .select()
    .from(chatBridgeInstalls)
    .where(eq(chatBridgeInstalls.id, input.installId))
    .limit(1);
  if (!install) return { kind: 'link_not_found' };

  const botUserId =
    (install.options as { botUserId?: string } | null)?.botUserId ?? null;
  if (botUserId && botUserId === input.authorChatUserId) {
    return { kind: 'skipped_bot' };
  }

  const [link] = await db()
    .select()
    .from(chatChannelLinks)
    .where(
      and(
        eq(chatChannelLinks.installId, input.installId),
        eq(chatChannelLinks.channelId, input.channelId),
      ),
    )
    .limit(1);
  if (!link) return { kind: 'link_not_found' };
  if (link.linkMode !== 'sync') return { kind: 'skipped_mode' };

  const rl = await checkRateLimit({ workspaceId: install.workspaceId });
  if (!rl.allowed) return { kind: 'skipped_rate_limit' };

  if (await isMessageProcessed({ pageId: link.pageId, chatMessageId: input.messageId })) {
    return { kind: 'duplicate' };
  }

  const sanitized = sanitizeHtml(input.body, {
    allowedTags: ['b', 'i', 'em', 'strong', 'a', 'code', 'pre'],
    allowedAttributes: { a: ['href'] },
  });

  const { commentId } = await markMessageProcessed({
    workspaceId: install.workspaceId,
    pageId: link.pageId,
    authorUserId: install.installedBy,
    chatMessageId: input.messageId,
    body: sanitized,
  });
  return { kind: 'inserted', commentId };
}

/**
 * Post a freshly-created page comment back to the linked channel(s).
 * Skips comments that already have a chat_message_id (i.e. comments that
 * originated FROM a channel — those would create an echo loop).
 */
export async function postCommentToChannels(input: {
  workspaceId: string;
  pageId: string;
  commentId: string;
  body: string;
  postFn: (args: { platform: 'slack' | 'discord'; channelId: string; body: string }) => Promise<string>;
}): Promise<{ posted: number }> {
  const links = await db()
    .select({ link: chatChannelLinks, install: chatBridgeInstalls })
    .from(chatChannelLinks)
    .innerJoin(chatBridgeInstalls, eq(chatChannelLinks.installId, chatBridgeInstalls.id))
    .where(
      and(
        eq(chatChannelLinks.workspaceId, input.workspaceId),
        eq(chatChannelLinks.pageId, input.pageId),
        eq(chatChannelLinks.linkMode, 'sync'),
      ),
    );

  let posted = 0;
  for (const row of links) {
    const rl = await checkRateLimit({ workspaceId: input.workspaceId });
    if (!rl.allowed) continue;
    await input.postFn({
      platform: row.install.platform as 'slack' | 'discord',
      channelId: row.link.channelId,
      body: input.body,
    });
    posted += 1;
  }
  return { posted };
}
```

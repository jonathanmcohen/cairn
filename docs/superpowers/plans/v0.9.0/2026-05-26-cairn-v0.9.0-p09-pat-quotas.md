# Cairn v0.9.0 — Plan P9: PAT quotas migration 0037 + scope rate-limits engine

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extend Personal Access Tokens (shipped in v0.7) with per-token daily / monthly request caps AND per-scope per-minute rate limits. A PAT request that exceeds either limit is rejected with HTTP `429 Too Many Requests` + a `Retry-After` header. Usage is rolled up in a tiny per-window table that resets at window boundaries; the existing PAT dispatcher checks the quota before invoking the route handler.

**Architecture:** Migration `0037_pat_quotas.sql` adds three nullable columns to `personal_access_tokens` (`daily_request_limit int`, `monthly_request_limit int`, `scope_rate_limits jsonb`) and a new `pat_quota_usage` rollup table keyed by `(token_id, window_start, window_kind)`. `src/lib/auth/pat-quota.ts` exposes `checkQuota(db, tokenId, scopeId)` returning `{allowed, retryAfterSec?}` — it computes the active day/month windows, reads usage, and (atomically) increments. The PAT dispatcher (`src/lib/auth/pat.ts`) calls `checkQuota` after `verifyPat` and before the handler; on `!allowed` it short-circuits with a `429` Response carrying `Retry-After`. Scope rate-limits live in jsonb of shape `{ "pages.write": {perMinute: 60}, "pages.read": {perMinute: 600} }` — per-minute buckets are tracked as in-process `Map<string, {count, resetAt}>` (one process = one bucket, acceptable for single-node homelab deployments matching `CAIRN_RATE_LIMIT_BUCKET_STRATEGY=memory`).

**Tech Stack:** Drizzle ORM migration, Postgres 16 jsonb, Vitest 4 + Testcontainers Postgres for the quota tests, no new runtime deps. Reuses `src/lib/auth/rate-limit.ts` patterns (v0.5.1 T5).

```ts
/**
 * Pure date helpers for PAT quota windows. UTC-pinned so the window boundary
 * is deterministic regardless of process TZ (single-node homelab, but tests
 * run with whatever TZ the container inherits).
 */

export function dayWindowStart(ts: Date): Date {
  return new Date(Date.UTC(ts.getUTCFullYear(), ts.getUTCMonth(), ts.getUTCDate(), 0, 0, 0, 0));
}

export function monthWindowStart(ts: Date): Date {
  return new Date(Date.UTC(ts.getUTCFullYear(), ts.getUTCMonth(), 1, 0, 0, 0, 0));
}

export function nextDayBoundarySec(ts: Date): number {
  const next = new Date(dayWindowStart(ts).getTime() + 24 * 60 * 60 * 1000);
  return Math.max(1, Math.ceil((next.getTime() - ts.getTime()) / 1000));
}

export function nextMonthBoundarySec(ts: Date): number {
  const cur = monthWindowStart(ts);
  const next = new Date(Date.UTC(cur.getUTCFullYear(), cur.getUTCMonth() + 1, 1, 0, 0, 0, 0));
  return Math.max(1, Math.ceil((next.getTime() - ts.getTime()) / 1000));
}
```

```ts
/**
 * In-process per-(tokenId, scopeId) per-minute bucket. Single-node homelab
 * acceptable: bucket lives in module-level Map; restarts reset all buckets,
 * which is OK because the DB rollup covers the per-day / per-month caps.
 *
 * Window is a rolling 60-second window pinned to UTC minute starts so the
 * boundary is deterministic in tests.
 */

export type ScopeLimit = { perMinute: number } | undefined;

type Bucket = { count: number; resetAt: number };

const buckets = new Map<string, Bucket>();

function keyFor(tokenId: string, scopeId: string): string {
  return `${tokenId}::${scopeId}`;
}

function minuteStartMs(now: number): number {
  return now - (now % 60_000);
}

export type ConsumeResult = { allowed: true } | { allowed: false; retryAfterSec: number };

export function consumeScopeBucket(
  tokenId: string,
  scopeId: string,
  limit: ScopeLimit,
): ConsumeResult {
  if (!limit || limit.perMinute <= 0) return { allowed: true };
  const now = Date.now();
  const windowStart = minuteStartMs(now);
  const key = keyFor(tokenId, scopeId);
  let b = buckets.get(key);
  if (!b || b.resetAt <= now) {
    b = { count: 0, resetAt: windowStart + 60_000 };
    buckets.set(key, b);
  }
  if (b.count >= limit.perMinute) {
    return { allowed: false, retryAfterSec: Math.max(1, Math.ceil((b.resetAt - now) / 1000)) };
  }
  b.count += 1;
  return { allowed: true };
}

/** Test-only: clear all buckets. */
export function resetScopeBucketsForTests(): void {
  buckets.clear();
}
```

```ts
  nextDayBoundarySec,
  nextMonthBoundarySec,
} from './pat-quota-windows';
import { consumeScopeBucket } from './pat-scope-bucket';

export type QuotaResult = { allowed: true } | { allowed: false; retryAfterSec: number };

type TokenLimits = {
  dailyRequestLimit: number | null;
  monthlyRequestLimit: number | null;
  scopeRateLimits: Record<string, { perMinute: number }> | null;
};

async function loadLimits(
  db: PostgresJsDatabase<typeof schema>,
  tokenId: string,
): Promise<TokenLimits | null> {
  const [row] = await db
    .select({
      dailyRequestLimit: schema.personalAccessTokens.dailyRequestLimit,
      monthlyRequestLimit: schema.personalAccessTokens.monthlyRequestLimit,
      scopeRateLimits: schema.personalAccessTokens.scopeRateLimits,
    })
    .from(schema.personalAccessTokens)
    .where(eq(schema.personalAccessTokens.id, tokenId));
  return row ?? null;
}

async function readUsage(
  db: PostgresJsDatabase<typeof schema>,
  tokenId: string,
  windowKind: 'day' | 'month',
  windowStart: Date,
): Promise<number> {
  const [row] = await db
    .select({ requests: schema.patQuotaUsage.requests })
    .from(schema.patQuotaUsage)
    .where(
      and(
        eq(schema.patQuotaUsage.tokenId, tokenId),
        eq(schema.patQuotaUsage.windowKind, windowKind),
        eq(schema.patQuotaUsage.windowStart, windowStart),
      ),
    );
  return row?.requests ?? 0;
}

async function bumpUsage(
  db: PostgresJsDatabase<typeof schema>,
  tokenId: string,
  windowKind: 'day' | 'month',
  windowStart: Date,
): Promise<void> {
  // Atomic upsert: insert with requests=1 or increment if the (token, window) row exists.
  await db
    .insert(schema.patQuotaUsage)
    .values({ tokenId, windowKind, windowStart, requests: 1, bytes: 0 })
    .onConflictDoUpdate({
      target: [
        schema.patQuotaUsage.tokenId,
        schema.patQuotaUsage.windowStart,
        schema.patQuotaUsage.windowKind,
      ],
      set: { requests: sql`${schema.patQuotaUsage.requests} + 1` },
    });
}

/**
 * Check + atomically bump the quota for `tokenId` on `scopeId`. Returns
 * `{allowed: true}` and increments the rollup on success; returns
 * `{allowed: false, retryAfterSec}` and does NOT increment on cap hit.
 *
 * Order of checks (cheapest first):
 *   1) in-process scope bucket (perMinute)
 *   2) DB-backed daily cap
 *   3) DB-backed monthly cap
 */
export async function checkQuota(
  db: PostgresJsDatabase<typeof schema>,
  tokenId: string,
  scopeId: string,
): Promise<QuotaResult> {
  const limits = await loadLimits(db, tokenId);
  if (!limits) return { allowed: true }; // token gone — caller will 401 separately

  const scopeLimit = limits.scopeRateLimits?.[scopeId];
  const bucket = consumeScopeBucket(tokenId, scopeId, scopeLimit);
  if (!bucket.allowed) {
    return { allowed: false, retryAfterSec: bucket.retryAfterSec };
  }

  const now = new Date();
  const dayStart = dayWindowStart(now);
  const monthStart = monthWindowStart(now);

  if (limits.dailyRequestLimit !== null) {
    const used = await readUsage(db, tokenId, 'day', dayStart);
    if (used >= limits.dailyRequestLimit) {
      return { allowed: false, retryAfterSec: nextDayBoundarySec(now) };
    }
  }
  if (limits.monthlyRequestLimit !== null) {
    const used = await readUsage(db, tokenId, 'month', monthStart);
    if (used >= limits.monthlyRequestLimit) {
      return { allowed: false, retryAfterSec: nextMonthBoundarySec(now) };
    }
  }

  // All caps cleared — bump both window rollups so future calls see the latest counts.
  await bumpUsage(db, tokenId, 'day', dayStart);
  await bumpUsage(db, tokenId, 'month', monthStart);
  return { allowed: true };
}
```

```ts
import { getDb as _getDbForDispatch } from '@/db/client';
import { checkQuota } from './pat-quota';

export type DispatchPatInput = {
  token: string;
  scope: string;
};

export type DispatchPatResult =
  | { kind: 'ok'; tokenId: string; userId: string; workspaceId: string; scopes: string[] }
  | { kind: 'invalid' }
  | { kind: 'rate-limited'; response: Response };

/**
 * Validate a `cairn_pat_*` bearer token, enforce its scope, then run the PAT
 * quota check. Returns:
 *   - `{kind:'ok', ...}` when verified + under quota.
 *   - `{kind:'invalid'}` when the token is unknown / revoked / lacks the scope.
 *   - `{kind:'rate-limited', response}` with a 429 + Retry-After Response
 *     when the daily / monthly / scope cap is hit.
 *
 * Route handlers should return `result.response` directly on `rate-limited`
 * so the client gets the standard 429.
 */
export async function dispatchPat(input: DispatchPatInput): Promise<DispatchPatResult> {
  const db = _getDbForDispatch();
  // Reuse the existing verify path — function name in this file pre-P9 was
  // `consumePatRequest` (returns null or {tokenId,userId,workspaceId,scopes}).
  // If the export name differs in the file, adapt this call site accordingly.
  const verified = await consumePatRequest(db, input.token, input.scope);
  if (!verified) return { kind: 'invalid' };

  const quota = await checkQuota(db, verified.tokenId, input.scope);
  if (!quota.allowed) {
    const response = new Response(
      JSON.stringify({ error: 'rate_limited', retryAfterSec: quota.retryAfterSec }),
      {
        status: 429,
        headers: {
          'Content-Type': 'application/json',
          'Retry-After': String(quota.retryAfterSec),
        },
      },
    );
    return { kind: 'rate-limited', response };
  }
  return {
    kind: 'ok',
    tokenId: verified.tokenId,
    userId: verified.userId,
    workspaceId: verified.workspaceId,
    scopes: verified.scopes,
  };
}
```

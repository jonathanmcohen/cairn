# Cairn v0.9.0 — Plan P16: Image gallery + lightbox block

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Single TipTap block that holds 1..n images in a responsive grid, with a click-to-open portal-mounted lightbox modal supporting prev/next/zoom + keyboard navigation. Dropping a batch of N image files anywhere over the editor creates ONE gallery node containing all of them as child `cairnImage` nodes (the existing image block from v0.2).

**Architecture:**
- New TipTap node `gallery` at `src/components/editor/blocks/gallery-node.ts`. Group `block`, content `cairnImage+` — i.e. a structural container that only accepts the existing image block as children.
- Renderer `gallery.tsx` lays children out in a Tailwind grid (`grid grid-cols-3 gap-2 sm:grid-cols-4`). Each child renders normally (the existing `cairnImage` node-view), wrapped in a click handler that opens the lightbox.
- New lightbox at `src/components/editor/lightbox.tsx` — portal-mounted via `createPortal`, focus-trapped, ESC restores focus to the trigger image. Arrow keys cycle prev/next. `+` / `-` zoom; clicking outside closes.
- The existing image-extension's drop handler (in `src/components/editor/image-extension.ts`) is modified: when multiple image files are dropped at once, it composes a single `gallery` node containing one `cairnImage` per file instead of inserting N sibling image nodes.
- Slash-command `/gallery` inserts an empty `gallery` node — the placeholder UI prompts the user to drop or pick files.

**Existing surface (read before editing):**
- `src/components/editor/image-extension.ts` — current image drop handler (single-file insert).
- `src/components/editor/blocks/` — where new schema-only nodes live (e.g. `mermaid-node.ts`).
- `src/components/editor/extensions.ts` + `extensions-lazy.ts` — extension registration.
- `src/components/editor/editor-node-names.ts` — `EDITOR_NODE_NAMES` array for static walkers.
- `src/components/editor/slash-extension.ts` — slash-command registry.
- The `cairnImage` node name (verify via `grep -n "name: 'cairnImage'" src/components/editor/image-extension.ts`; if it's actually `image` adapt this plan accordingly).

```tsx
'use client';

import type { NodeViewProps } from '@tiptap/react';
import { NodeViewContent, NodeViewWrapper } from '@tiptap/react';
import { useState } from 'react';
import { Lightbox } from '@/components/editor/lightbox';

type ChildImage = { src: string; alt: string };

function extractChildren(node: NodeViewProps['node']): ChildImage[] {
  const out: ChildImage[] = [];
  node.content.forEach((child) => {
    if (child.type.name !== 'cairnImage' && child.type.name !== 'image') return;
    const src = (child.attrs.src ?? child.attrs.fileId ?? '') as string;
    const alt = (child.attrs.alt ?? '') as string;
    if (src) out.push({ src, alt });
  });
  return out;
}

export function GalleryView({ node, editor }: NodeViewProps) {
  const images = extractChildren(node);
  const [openIndex, setOpenIndex] = useState<number | null>(null);

  if (images.length === 0) {
    return (
      <NodeViewWrapper
        className="my-3 rounded-md border-2 border-dashed p-6 text-center text-sm text-muted-foreground"
        data-empty
      >
        Drop images here or use the image button.
        {/* NodeViewContent is required so TipTap renders the (empty) child slot. */}
        <NodeViewContent className="sr-only" />
      </NodeViewWrapper>
    );
  }

  return (
    <NodeViewWrapper className="my-3">
      <div className="grid grid-cols-3 gap-2 sm:grid-cols-4" role="list">
        {images.map((img, i) => (
          <button
            key={`${img.src}-${i}`}
            type="button"
            role="listitem"
            className="block aspect-square overflow-hidden rounded-md focus:outline-none focus:ring-2 focus:ring-ring"
            onClick={() => setOpenIndex(i)}
            aria-label={`Open image ${i + 1} of ${images.length}${img.alt ? ': ' + img.alt : ''}`}
          >
            <img
              src={img.src}
              alt={img.alt}
              className="h-full w-full object-cover"
              loading="lazy"
            />
          </button>
        ))}
      </div>
      {/* TipTap still needs a hidden content slot to keep child nodes editable
          via keyboard nav even when the grid is the visible surface. */}
      <NodeViewContent className="sr-only" />
      {openIndex !== null && (
        <Lightbox
          images={images}
          startIndex={openIndex}
          onClose={() => setOpenIndex(null)}
        />
      )}
      {editor.isEditable && (
        <p className="mt-1 text-xs text-muted-foreground">
          {images.length} {images.length === 1 ? 'image' : 'images'}
        </p>
      )}
    </NodeViewWrapper>
  );
}
```

(The `sr-only NodeViewContent` is a workaround for TipTap requiring a content slot when `content` is non-empty. The grid is the visible surface; the underlying ProseMirror children remain selectable via keyboard if the user tabs into the screen-reader-only slot.)

```tsx
'use client';

import { useEffect, useRef, useState } from 'react';
import { createPortal } from 'react-dom';

type Img = { src: string; alt: string };

export function Lightbox({
  images,
  startIndex,
  onClose,
}: {
  images: Img[];
  startIndex: number;
  onClose: () => void;
}) {
  const [index, setIndex] = useState(startIndex);
  const [scale, setScale] = useState(1);
  const dialogRef = useRef<HTMLDivElement | null>(null);
  const previousFocus = useRef<Element | null>(null);

  // Capture previously focused element and focus the dialog on mount; restore
  // on unmount.
  useEffect(() => {
    previousFocus.current = document.activeElement;
    dialogRef.current?.focus();
    return () => {
      if (previousFocus.current instanceof HTMLElement) {
        previousFocus.current.focus();
      }
    };
  }, []);

  // Keyboard handlers.
  useEffect(() => {
    function onKey(e: KeyboardEvent) {
      switch (e.key) {
        case 'Escape':
          e.preventDefault();
          onClose();
          break;
        case 'ArrowRight':
          e.preventDefault();
          setIndex((i) => (i + 1) % images.length);
          setScale(1);
          break;
        case 'ArrowLeft':
          e.preventDefault();
          setIndex((i) => (i - 1 + images.length) % images.length);
          setScale(1);
          break;
        case '+':
        case '=':
          e.preventDefault();
          setScale((s) => Math.min(s + 0.25, 4));
          break;
        case '-':
          e.preventDefault();
          setScale((s) => Math.max(s - 0.25, 0.25));
          break;
      }
    }
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [images.length, onClose]);

  if (typeof document === 'undefined') return null;
  const current = images[index]!;

  return createPortal(
    <div
      ref={dialogRef}
      role="dialog"
      aria-modal="true"
      aria-label={`Image ${index + 1} of ${images.length}`}
      tabIndex={-1}
      className="fixed inset-0 z-50 flex flex-col items-center justify-center bg-black/90 outline-none"
    >
      <div
        data-testid="lightbox-backdrop"
        className="absolute inset-0"
        onClick={onClose}
        aria-hidden
      />
      <div className="relative z-10 flex max-h-full max-w-full items-center justify-center p-4">
        <img
          src={current.src}
          alt={current.alt}
          style={{ transform: `scale(${scale})`, transition: 'transform 120ms ease' }}
          className="max-h-[85vh] max-w-[90vw] origin-center"
        />
      </div>
      <div className="relative z-10 mt-2 flex items-center gap-4 text-sm text-white">
        <button
          type="button"
          className="rounded p-2 hover:bg-white/10"
          onClick={() => {
            setIndex((i) => (i - 1 + images.length) % images.length);
            setScale(1);
          }}
          aria-label="Previous image"
        >
          ‹
        </button>
        <span>{index + 1} / {images.length}</span>
        <button
          type="button"
          className="rounded p-2 hover:bg-white/10"
          onClick={() => {
            setIndex((i) => (i + 1) % images.length);
            setScale(1);
          }}
          aria-label="Next image"
        >
          ›
        </button>
        <span className="ml-4 text-xs text-white/60">+/- to zoom · Esc to close</span>
      </div>
    </div>,
    document.body,
  );
}
```

```ts
export type GalleryInsertResult =
  | { type: 'cairnImage'; attrs: { fileId: string } }
  | { type: 'gallery'; content: Array<{ type: 'cairnImage'; attrs: { fileId: string } }> };

export async function composeGalleryInsert(input: {
  files: File[];
  uploadFn: (file: File) => Promise<string>;
}): Promise<GalleryInsertResult> {
  const imageFiles = input.files.filter((f) => f.type.startsWith('image/'));
  if (imageFiles.length === 0) {
    return { type: 'gallery', content: [] };
  }
  const ids = await Promise.all(imageFiles.map(input.uploadFn));
  if (ids.length === 1) {
    return { type: 'cairnImage', attrs: { fileId: ids[0]! } };
  }
  return {
    type: 'gallery',
    content: ids.map((fileId) => ({ type: 'cairnImage' as const, attrs: { fileId } })),
  };
}
```

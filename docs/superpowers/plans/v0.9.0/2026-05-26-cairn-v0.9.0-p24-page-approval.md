# Cairn v0.9.0 — Plan P24: Page approval + signed audit (migration 0045 — shared with P25)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A reviewable, tamper-evident page approval workflow. Page owner moves a page to `status='review'` → the page header surfaces an approval panel for admins → admin chooses `approve` / `reject` / `requested_changes` → the decision is recorded with a HMAC-SHA256 signature over `(page_id|version_snapshot_id|approver_id|decision|approved_at)` and a v0.6 audit entry. The migration in this plan ALSO adds `page_templates.visibility` (P25 consumes that column) so the two land atomically in `0045_approvals_template_visibility.sql`.

**Architecture:** New table `page_approvals` keyed `(id uuid pk)` with FKs to `pages`, `page_versions` (v0.5 P4 snapshot table), and `users`. Each decision captures the version snapshot id at the time of approval — the signature includes it, so post-decision edits to the page don't invalidate the signature but the UI can clearly say "approved at version X" when the page has since advanced. The library function `requestApproval(pageId)` transitions `pages.status='review'` (this column lands in P26's 0044 migration; this plan declares a soft-dep on P26 and our migration runs after if Drizzle's sequence puts 0044 first). The approval panel mounts in the page header when `status='review'` and the viewer has admin role per v0.7 ACL.

**Tech Stack:** Drizzle migrations + table, Node `crypto.createHmac` for signing, existing `recordAudit` from v0.6 P18, Next.js 16 server actions for the admin decision form, Vitest + Testcontainers for full DB roundtrip.

```ts
import { createHmac, timingSafeEqual } from 'node:crypto';

export type ApprovalDecision = 'approved' | 'rejected' | 'requested_changes';

export type ApprovalSignatureParts = {
  pageId: string;
  versionSnapshotId: string;
  approverUserId: string;
  decision: ApprovalDecision;
  approvedAtISO: string;
};

function canonical(parts: ApprovalSignatureParts): string {
  return [
    parts.pageId,
    parts.versionSnapshotId,
    parts.approverUserId,
    parts.decision,
    parts.approvedAtISO,
  ].join('|');
}

export function signApproval(parts: ApprovalSignatureParts, secret: string): string {
  return createHmac('sha256', secret).update(canonical(parts)).digest('hex');
}

export function verifyApprovalSignature(
  parts: ApprovalSignatureParts,
  signature: string,
  secret: string,
): boolean {
  const expected = signApproval(parts, secret);
  if (expected.length !== signature.length) return false;
  return timingSafeEqual(Buffer.from(expected, 'hex'), Buffer.from(signature, 'hex'));
}
```

```ts
import { and, desc, eq } from 'drizzle-orm';
import { sql } from 'drizzle-orm';
import { recordAudit } from '@/lib/audit/log';
import { env } from '@/lib/env';
import { getDb } from '@/server/db';
import { pageApprovals } from '@/db/schema/page-approvals';
import { pageVersions } from '@/db/schema/page-versions';
import {
  type ApprovalDecision,
  signApproval,
} from './approval-signature';

async function pagesHasStatusColumn(): Promise<boolean> {
  const db = getDb();
  const rows = (await db.execute(sql`
    SELECT column_name FROM information_schema.columns
    WHERE table_name = 'pages' AND column_name = 'status'
  `)) as unknown as Array<{ column_name: string }>;
  return rows.length > 0;
}

export async function requestApproval(pageId: string, userId: string): Promise<void> {
  if (await pagesHasStatusColumn()) {
    const db = getDb();
    await db.execute(sql`UPDATE pages SET status = 'review' WHERE id = ${pageId}`);
  }
  await recordAudit({
    actorUserId: userId,
    action: 'page.approval_requested',
    entityType: 'page',
    entityId: pageId,
  });
}

export async function decide(input: {
  pageId: string;
  approverUserId: string;
  decision: ApprovalDecision;
  comment?: string;
}): Promise<{ id: string; signatureHmac: string }> {
  const db = getDb();
  const versions = await db
    .select({ id: pageVersions.id })
    .from(pageVersions)
    .where(eq(pageVersions.pageId, input.pageId))
    .orderBy(desc(pageVersions.createdAt))
    .limit(1);
  const versionSnapshotId = versions[0]?.id;
  if (!versionSnapshotId) throw new Error('no version snapshot available');

  const approvedAt = new Date();
  const approvedAtISO = approvedAt.toISOString();
  const sig = signApproval(
    {
      pageId: input.pageId,
      versionSnapshotId,
      approverUserId: input.approverUserId,
      decision: input.decision,
      approvedAtISO,
    },
    env().AUTH_SECRET,
  );

  const inserted = await db
    .insert(pageApprovals)
    .values({
      pageId: input.pageId,
      versionSnapshotId,
      approverUserId: input.approverUserId,
      decision: input.decision,
      comment: input.comment ?? null,
      signatureHmac: sig,
      approvedAt,
    })
    .returning({ id: pageApprovals.id });

  // Lifecycle transition on the page row (only when P26 has landed).
  if (await pagesHasStatusColumn()) {
    const nextStatus =
      input.decision === 'approved'
        ? 'published'
        : input.decision === 'rejected'
          ? 'draft'
          : 'review';
    await db.execute(sql`UPDATE pages SET status = ${nextStatus} WHERE id = ${input.pageId}`);
  }

  await recordAudit({
    actorUserId: input.approverUserId,
    action:
      input.decision === 'approved'
        ? 'page.approved'
        : input.decision === 'rejected'
          ? 'page.rejected'
          : 'page.changes_requested',
    entityType: 'page',
    entityId: input.pageId,
    details: { versionSnapshotId, signatureHmac: sig },
  });

  return { id: inserted[0]!.id, signatureHmac: sig };
}

export async function listApprovals(pageId: string): Promise<
  Array<{
    id: string;
    decision: ApprovalDecision;
    approverUserId: string;
    approvedAt: Date;
    comment: string | null;
    versionSnapshotId: string;
  }>
> {
  const db = getDb();
  const rows = await db
    .select()
    .from(pageApprovals)
    .where(eq(pageApprovals.pageId, pageId))
    .orderBy(desc(pageApprovals.approvedAt));
  return rows.map((r) => ({
    id: r.id,
    decision: r.decision,
    approverUserId: r.approverUserId,
    approvedAt: r.approvedAt,
    comment: r.comment,
    versionSnapshotId: r.versionSnapshotId,
  }));
}
```

```ts
import { requirePageAccess } from '@/lib/pages/access';
import { listApprovals, requestApproval } from '@/lib/pages/approval';

const PostSchema = z.object({ action: z.literal('request') });

export async function POST(
  req: Request,
  ctxParams: { params: Promise<{ id: string }> },
): Promise<Response> {
  const { id } = await ctxParams.params;
  const { ctx } = await requirePageAccess(id, 'editor');
  const parsed = PostSchema.safeParse(await req.json());
  if (!parsed.success) return new Response('bad request', { status: 400 });
  await requestApproval(id, ctx.userId);
  return new Response(JSON.stringify({ ok: true }), {
    status: 200,
    headers: { 'content-type': 'application/json' },
  });
}

export async function GET(
  _req: Request,
  ctxParams: { params: Promise<{ id: string }> },
): Promise<Response> {
  const { id } = await ctxParams.params;
  await requirePageAccess(id, 'viewer');
  const history = await listApprovals(id);
  return new Response(JSON.stringify({ history }), {
    status: 200,
    headers: { 'content-type': 'application/json' },
  });
}
```

```tsx
'use client';

import { useEffect, useState } from 'react';
import { Button } from '@/components/ui/button';
import { Textarea } from '@/components/ui/textarea';

type Decision = 'approved' | 'rejected' | 'requested_changes';

type HistoryItem = {
  id: string;
  decision: Decision;
  approverUserId: string;
  approvedAt: string;
  comment: string | null;
  versionSnapshotId: string;
};

export function ApprovalPanel(props: {
  pageId: string;
  canDecide: boolean;
  inReview: boolean;
}): JSX.Element | null {
  const [history, setHistory] = useState<HistoryItem[]>([]);
  const [comment, setComment] = useState('');
  const [submitting, setSubmitting] = useState(false);

  useEffect(() => {
    if (!props.inReview && history.length === 0) return;
    let cancelled = false;
    fetch(`/api/pages/${props.pageId}/approval`)
      .then((r) => r.json())
      .then((data: { history: HistoryItem[] }) => {
        if (!cancelled) setHistory(data.history);
      })
      .catch(() => {});
    return () => {
      cancelled = true;
    };
  }, [props.pageId, props.inReview]);

  if (!props.inReview && history.length === 0) return null;

  const submit = async (decision: Decision): Promise<void> => {
    setSubmitting(true);
    try {
      const res = await fetch(`/api/pages/${props.pageId}/approval/decide`, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ decision, comment: comment || undefined }),
      });
      if (!res.ok) throw new Error(`failed (${res.status})`);
      const fresh = await fetch(`/api/pages/${props.pageId}/approval`).then((r) => r.json());
      setHistory(fresh.history);
      setComment('');
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <aside aria-label="Page approval" className="rounded border bg-amber-50/30 p-4">
      <h2 className="text-sm font-semibold uppercase tracking-wide">Approval</h2>
      {props.canDecide && props.inReview && (
        <div className="mt-3 space-y-2">
          <Textarea
            placeholder="Comment (optional)"
            value={comment}
            onChange={(e) => setComment(e.target.value)}
            aria-label="Approval comment"
          />
          <div className="flex gap-2">
            <Button onClick={() => submit('approved')} disabled={submitting}>
              Approve
            </Button>
            <Button
              variant="outline"
              onClick={() => submit('requested_changes')}
              disabled={submitting}
            >
              Request changes
            </Button>
            <Button
              variant="destructive"
              onClick={() => submit('rejected')}
              disabled={submitting}
            >
              Reject
            </Button>
          </div>
        </div>
      )}
      {history.length > 0 && (
        <ol className="mt-4 space-y-2 text-sm">
          {history.map((h) => (
            <li key={h.id} className="rounded border bg-background p-2">
              <span className="font-mono">{h.decision}</span> by {h.approverUserId}{' '}
              <time dateTime={h.approvedAt}>{new Date(h.approvedAt).toLocaleString()}</time>
              {h.comment && <p className="mt-1 text-muted-foreground">{h.comment}</p>}
            </li>
          ))}
        </ol>
      )}
    </aside>
  );
}
```

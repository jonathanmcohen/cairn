```ts
/**
 * Typed keyboard-shortcut registry — the single source of truth for the
 * dispatcher, the ⌘/ sheet, and the ⌘K palette's action entries.
 *
 * Pure data + a small API. No React, no DOM beyond reading a plain key-event
 * shape. `Mod` is the platform-agnostic command/control modifier.
 */

/** Scopes a shortcut is active in. `global` is always live; `editor` is reserved. */
export type ShortcutScope = 'global' | 'editor';

/**
 * `action` entries also surface in the ⌘K palette; `command` entries are
 * keyboard-only (kept off the palette to avoid clutter).
 */
export type ShortcutKind = 'action' | 'command';

export type ShortcutEntry = {
  /** Stable unique id (also the de-dupe key for idempotent re-registration). */
  id: string;
  /** Human-authored binding, e.g. `Mod+Shift+L`. Normalized on registration. */
  keys: string;
  scope: ShortcutScope;
  kind: ShortcutKind;
  /** i18n key resolved via `t()` at render time (never a literal string). */
  labelKey: string;
  /** Side effect to run when matched/selected. */
  run: () => void;
};

/** Minimal key-event shape the matcher needs (a real KeyboardEvent satisfies it). */
export type KeyEventLike = {
  key: string;
  metaKey: boolean;
  ctrlKey: boolean;
  shiftKey: boolean;
  altKey: boolean;
};

const registry = new Map<string, ShortcutEntry>();

/** Lowercase the key, canonically order modifiers: mod, alt, shift, then key. */
export function normalizeKeys(keys: string): string {
  const parts = keys.split('+').map((p) => p.trim().toLowerCase());
  const order = ['mod', 'alt', 'shift'];
  const mods = parts.filter((p) => order.includes(p)).sort((a, b) => order.indexOf(a) - order.indexOf(b));
  const key = parts.find((p) => !order.includes(p)) ?? '';
  return [...mods, key].filter(Boolean).join('+');
}

/** Register (or replace by id). Throws on a same-scope, same-normalized-keys collision. */
export function registerShortcut(entry: ShortcutEntry): void {
  const norm = normalizeKeys(entry.keys);
  for (const existing of registry.values()) {
    if (existing.id === entry.id) continue; // re-register replaces, never conflicts
    if (existing.scope === entry.scope && normalizeKeys(existing.keys) === norm) {
      throw new Error(
        `shortcut conflict: "${entry.keys}" in scope "${entry.scope}" already bound by "${existing.id}"`,
      );
    }
  }
  registry.set(entry.id, entry);
}

/** All entries for a scope, registration order preserved. */
export function getShortcuts(scope?: ShortcutScope): ShortcutEntry[] {
  const all = [...registry.values()];
  return scope ? all.filter((e) => e.scope === scope) : all;
}

/** The normalized binding of a key event (or null when no Mod modifier is held). */
function eventKeys(e: KeyEventLike): string {
  const mods: string[] = [];
  if (e.metaKey || e.ctrlKey) mods.push('mod');
  if (e.altKey) mods.push('alt');
  if (e.shiftKey) mods.push('shift');
  return normalizeKeys([...mods, e.key.toLowerCase()].join('+'));
}

/** First entry in `scope` whose binding matches the event, else null. */
export function matchShortcut(e: KeyEventLike, scope: ShortcutScope): ShortcutEntry | null {
  const target = eventKeys(e);
  for (const entry of getShortcuts(scope)) {
    if (normalizeKeys(entry.keys) === target) return entry;
  }
  return null;
}

/** Test-only: clear the registry between cases. */
export function resetRegistry(): void {
  registry.clear();
}
```

```ts
'use client';

import { registerShortcut } from '@/lib/shortcuts/registry';

/**
 * Register the app's global shortcuts exactly once. `run()` callbacks are wired
 * by the dispatcher (which owns the router/theme/workspace handles) via the
 * `handlers` it passes — here we register the metadata + a thunk that reads the
 * currently-installed handler so the registry stays pure data.
 */
export type ShortcutHandlers = {
  newPage: () => void;
  toggleTheme: () => void;
  openSheet: () => void;
  switchWorkspace: () => void;
  openFavorites: () => void;
};

let handlers: ShortcutHandlers | null = null;

/** Install the live handlers (called by the dispatcher on mount). */
export function setShortcutHandlers(h: ShortcutHandlers): void {
  handlers = h;
}

let registered = false;

/** Idempotently register the global shortcut set. Safe to call on every mount. */
export function ensureAppShortcuts(): void {
  if (registered) return;
  registered = true;
  registerShortcut({
    id: 'page.new',
    keys: 'Mod+N',
    scope: 'global',
    kind: 'action',
    labelKey: 'shortcut.newPage',
    run: () => handlers?.newPage(),
  });
  registerShortcut({
    id: 'theme.toggle',
    keys: 'Mod+Shift+L',
    scope: 'global',
    kind: 'action',
    labelKey: 'shortcut.toggleTheme',
    run: () => handlers?.toggleTheme(),
  });
  registerShortcut({
    id: 'workspace.switch',
    keys: 'Mod+Shift+O',
    scope: 'global',
    kind: 'action',
    labelKey: 'shortcut.switchWorkspace',
    run: () => handlers?.switchWorkspace(),
  });
  registerShortcut({
    id: 'nav.favorites',
    keys: 'Mod+Shift+F',
    scope: 'global',
    kind: 'action',
    labelKey: 'shortcut.openFavorites',
    run: () => handlers?.openFavorites(),
```

```tsx
'use client';

import type { Route } from 'next';
import { useRouter } from 'next/navigation';
import { useTheme } from 'next-themes';
import { createContext, type ReactNode, useCallback, useContext, useEffect, useMemo, useState } from 'react';
import { matchShortcut } from '@/lib/shortcuts/registry';
import { ensureAppShortcuts, setShortcutHandlers } from './app-shortcuts';

type SheetState = { open: boolean; setOpen: (v: boolean) => void };
const SheetCtx = createContext<SheetState | null>(null);

/** Read the ⌘/ sheet open-state (used by ShortcutSheet). */
export function useShortcutSheet(): SheetState {
  const ctx = useContext(SheetCtx);
  if (!ctx) throw new Error('useShortcutSheet must be used within ShortcutDispatcher');
  return ctx;
}

export function ShortcutDispatcher({ children }: { children?: ReactNode }) {
  const router = useRouter();
  const { theme, setTheme } = useTheme();
  const [sheetOpen, setSheetOpen] = useState(false);

  const newPage = useCallback(() => {
    void (async () => {
      const res = await fetch('/api/pages', {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: '{}',
      });
      if (res.ok) {
        const created = (await res.json()) as { id: string };
        router.push(`/pages/${created.id}` as Route);
        router.refresh();
      }
    })();
  }, [router]);

  const toggleTheme = useCallback(() => {
    setTheme(theme === 'dark' ? 'light' : 'dark');
  }, [theme, setTheme]);

  const openFavorites = useCallback(() => {
    router.push('/favorites' as Route);
  }, [router]);

  const switchWorkspace = useCallback(() => {
    router.push('/workspaces' as Route);
  }, [router]);

  const openSheet = useCallback(() => setSheetOpen(true), []);

  useEffect(() => {
    ensureAppShortcuts();
    setShortcutHandlers({ newPage, toggleTheme, openFavorites, switchWorkspace, openSheet });
  }, [newPage, toggleTheme, openFavorites, switchWorkspace, openSheet]);

  useEffect(() => {
    function onKey(e: KeyboardEvent) {
      // Only act on Mod-modified chords so we never swallow plain typing.
      if (!e.metaKey && !e.ctrlKey) return;
      const hit = matchShortcut(e, 'global');
      if (hit) {
        e.preventDefault();
        hit.run();
      }
    }
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, []);

  const sheet = useMemo<SheetState>(() => ({ open: sheetOpen, setOpen: setSheetOpen }), [sheetOpen]);

  return <SheetCtx.Provider value={sheet}>{children}</SheetCtx.Provider>;
}
```

```tsx
'use client';

import { useEffect } from 'react';
import { getShortcuts, type ShortcutScope } from '@/lib/shortcuts/registry';
import { useShortcutSheet } from './dispatcher';

const SCOPES: { scope: ShortcutScope; titleKey: string }[] = [
  { scope: 'global', titleKey: 'shortcuts.group.global' },
  { scope: 'editor', titleKey: 'shortcuts.group.editor' },
];

/** Pretty-print a `Mod+Shift+L` binding for the platform (⌘ on mac, Ctrl elsewhere). */
function prettyKeys(keys: string): string {
  const isMac = typeof navigator !== 'undefined' && /Mac|iPhone|iPad/.test(navigator.platform);
  return keys
    .split('+')
    .map((p) => {
      const k = p.trim();
      if (k === 'Mod') return isMac ? '⌘' : 'Ctrl';
      if (k === 'Shift') return isMac ? '⇧' : 'Shift';
      if (k === 'Alt') return isMac ? '⌥' : 'Alt';
      return k.toUpperCase();
    })
    .join(isMac ? '' : '+');
}

export function ShortcutSheet({ label = (k: string) => k }: { label?: (key: string) => string }) {
  const { open, setOpen } = useShortcutSheet();

  useEffect(() => {
    function onKey(e: KeyboardEvent) {
      if (e.key === 'Escape') setOpen(false);
    }
    if (open) window.addEventListener('keydown', onKey);
```

```ts
import type { Locale } from './config';

export type Messages = Record<string, string>;
export type TVars = Record<string, string | number>;
export type TFunction = (key: string, vars?: TVars) => string;

/** Replace `{name}` placeholders; leave unmatched placeholders intact. */
function interpolate(template: string, vars: TVars): string {
  return template.replace(/\{(\w+)\}/g, (match, name: string) =>
    name in vars ? String(vars[name]) : match,
  );
}

/**
 * Build a `t()` over a flat-key catalog.
 *
 * - `t('a.b')` → catalog['a.b'], else the key itself (visible-in-dev fallback).
 * - `vars.count` triggers plural selection: tries `key.<category>` (e.g.
 *   `items.one`/`items.other` via `Intl.PluralRules`), then the bare `key`.
 * - `{name}` placeholders interpolate from `vars`.
 */
export function createT(locale: Locale, messages: Messages): TFunction {
  const plural = new Intl.PluralRules(locale);
  return (key, vars = {}) => {
    let template: string | undefined;
    if (typeof vars.count === 'number') {
      const category = plural.select(vars.count);
      template = messages[`${key}.${category}`] ?? messages[`${key}.other`] ?? messages[key];
    } else {
      template = messages[key];
    }
    if (template === undefined) return key;
    return interpolate(template, vars);
  };
}
```

- [x] **Step 6: Write `src/lib/i18n/resolve.ts`**

```ts
import { DEFAULT_LOCALE, isLocale, type Locale } from './config';

/** Parse an `Accept-Language` header into base languages, highest-q first. */
function parseAcceptLanguage(header: string): string[] {
  return header
    .split(',')
    .map((part) => {
      const [tag, ...params] = part.trim().split(';');
      const q = params.find((p) => p.trim().startsWith('q='));
      const weight = q ? Number.parseFloat(q.split('=')[1] ?? '1') : 1;
      return { lang: (tag ?? '').trim().toLowerCase(), q: Number.isNaN(weight) ? 1 : weight };
    })
    .filter((e) => e.lang.length > 0)
    .sort((a, b) => b.q - a.q)
    .map((e) => e.lang);
}

/**
 * Resolve the active locale: a valid `cairn_locale` cookie wins; else the
 * highest-q supported `Accept-Language` (region subtag ignored); else `en`.
 */
export function resolveLocale(
  cookieValue: string | null | undefined,
  acceptLanguage: string | null | undefined,
): Locale {
  if (cookieValue && isLocale(cookieValue)) return cookieValue;
  if (acceptLanguage) {
    for (const lang of parseAcceptLanguage(acceptLanguage)) {
      const base = lang.split('-')[0] ?? lang;
      if (isLocale(base)) return base;
    }
  }
  return DEFAULT_LOCALE;
}
```

```tsx
'use client';

import { createContext, type ReactNode, useContext, useMemo } from 'react';
import type { Locale } from './config';
import { createT, type Messages, type TFunction } from './t';

type I18nValue = { locale: Locale; t: TFunction };
const I18nCtx = createContext<I18nValue | null>(null);

export function I18nProvider({
  locale,
  messages,
  children,
}: {
  locale: Locale;
  messages: Messages;
  children: ReactNode;
}) {
  const value = useMemo<I18nValue>(() => ({ locale, t: createT(locale, messages) }), [locale, messages]);
  return <I18nCtx.Provider value={value}>{children}</I18nCtx.Provider>;
}

/** The translate function for the active locale. */
export function useT(): TFunction {
  const ctx = useContext(I18nCtx);
  if (!ctx) throw new Error('useT must be used within I18nProvider');
  return ctx.t;
}

/** The active locale (for `dir`-aware UI bits). */
export function useLocale(): Locale {
  const ctx = useContext(I18nCtx);
  if (!ctx) throw new Error('useLocale must be used within I18nProvider');
  return ctx.locale;
}
```

```tsx
import { useLocale, useT } from '@/lib/i18n/provider';

/** Settings control: writes the cairn_locale cookie and reloads to re-resolve `dir`. */
export function LocaleSwitcher() {
  const t = useT();
  const current = useLocale();

  function setLocale(next: Locale) {
    if (next === current) return;
    // 1-year cookie, root path, SameSite=Lax so it rides the next navigation.
    document.cookie = `${LOCALE_COOKIE}=${next}; path=/; max-age=31536000; samesite=lax`;
    // Full reload so the server re-resolves <html lang dir> + reloads the catalog.
    window.location.reload();
  }

  return (
    <label className="flex items-center gap-2 text-sm">
      <span className="text-muted-foreground">{t('locale.label')}</span>
      <select
        value={current}
        onChange={(e) => setLocale(e.target.value as Locale)}
        className="rounded border bg-background px-2 py-1"
      >
        {LOCALES.map((loc) => (
          <option key={loc} value={loc}>
            {t(`locale.${loc}`)}
          </option>
        ))}
      </select>
    </label>
  );
}
```

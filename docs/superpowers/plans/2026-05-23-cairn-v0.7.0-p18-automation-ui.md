> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the user-facing surface for the automation rules engine: a `/settings/automation` page listing rules, a New/Edit rule form (trigger + conditions + action), a rules list with enable/disable toggle + delete, and the CRUD REST API behind it. CRUD round-trip + cross-workspace 403/404 covered by tests.

**Architecture:** Server-component page lists rules + renders an `<AddRuleButton>` that opens the client `<RuleForm>` modal. The list itself (`<RuleList>`) is a client component so toggle/delete are interactive. Two API route files — `route.ts` for `GET`/`POST` and `[ruleId]/route.ts` for `PATCH`/`DELETE`. All routes are admin-gated via the existing `requireRole('admin')` helper. Last-run-status pulls the newest `automation_runs` row per rule via a single LATERAL join.

**Tech Stack:** Next.js 16 (App Router, React 19), Tailwind v4 + shadcn/ui, existing `requireRole`/`getAuthContext` from v0.2. No new deps. v0.6 filter-UI primitives (whichever component houses the Property/Operator/Value rows in the search/db-filter UI) are imported as-is.

```ts
import { eq } from 'drizzle-orm';
import { type NextRequest, NextResponse } from 'next/server';
import { z } from 'zod';
import { getDb } from '@/db/client';
import * as schema from '@/db/schema';
import { requireRole } from '@/lib/auth/require-role';

const triggerEvents = [
  'page.created',
  'page.updated',
  'page.deleted',
  'row.created',
  'row.updated',
  'row.deleted',
  'comment.created',
] as const;

const actionTypes = ['notify', 'send_webhook', 'set_property', 'create_page'] as const;

const RuleInput = z.object({
  name: z.string().min(1).max(200),
  triggerEvent: z.enum(triggerEvents),
  condition: z.record(z.unknown()).default({}),
  actionType: z.enum(actionTypes),
  actionConfig: z.record(z.unknown()),
  enabled: z.boolean().optional().default(true),
});

export async function GET(_req: NextRequest) {
  try {
    const ctx = await requireRole('viewer'); // any member can read; matches webhook list in v0.5
    const db = getDb();
    const rules = await db
      .select()
      .from(schema.automationRules)
      .where(eq(schema.automationRules.workspaceId, ctx.workspaceId));
    return NextResponse.json({ rules });
  } catch (err) {
    return errorResponse(err);
  }
}

export async function POST(req: NextRequest) {
  try {
    const ctx = await requireRole('admin');
    const body = RuleInput.parse(await req.json());
    const db = getDb();
    const [row] = await db
      .insert(schema.automationRules)
      .values({
        workspaceId: ctx.workspaceId,
        name: body.name,
        triggerEvent: body.triggerEvent,
        condition: body.condition,
        actionType: body.actionType,
        actionConfig: body.actionConfig,
        enabled: body.enabled,
        createdBy: ctx.userId,
      })
      .returning();
    return NextResponse.json(row, { status: 201 });
  } catch (err) {
    return errorResponse(err);
  }
}

function errorResponse(err: unknown): NextResponse {
  if (err instanceof z.ZodError) return NextResponse.json({ error: err.flatten() }, { status: 400 });
  const status =
    err && typeof err === 'object' && 'status' in err && typeof (err as { status: number }).status === 'number'
      ? (err as { status: number }).status
      : 500;
  const message = err instanceof Error ? err.message : 'internal error';
  return NextResponse.json({ error: message }, { status });
}
```

```ts
import { and, eq } from 'drizzle-orm';
import { type NextRequest, NextResponse } from 'next/server';
import { z } from 'zod';
import { getDb } from '@/db/client';
import * as schema from '@/db/schema';
import { requireRole } from '@/lib/auth/require-role';

const RuleUpdate = z
  .object({
    name: z.string().min(1).max(200).optional(),
    triggerEvent: z
      .enum(['page.created', 'page.updated', 'page.deleted', 'row.created', 'row.updated', 'row.deleted', 'comment.created'])
      .optional(),
    condition: z.record(z.unknown()).optional(),
    actionType: z.enum(['notify', 'send_webhook', 'set_property', 'create_page']).optional(),
    actionConfig: z.record(z.unknown()).optional(),
    enabled: z.boolean().optional(),
  })
  .refine((o) => Object.keys(o).length > 0, { message: 'no fields to update' });

type Ctx = { params: Promise<{ ruleId: string }> };

export async function PATCH(req: NextRequest, { params }: Ctx) {
  try {
    const ctx = await requireRole('admin');
    const { ruleId } = await params;
    const patch = RuleUpdate.parse(await req.json());
    const db = getDb();
    // Cross-workspace = 404 (per cairn convention — never 403, never leak existence).
    const [updated] = await db
      .update(schema.automationRules)
      .set(patch)
      .where(and(eq(schema.automationRules.id, ruleId), eq(schema.automationRules.workspaceId, ctx.workspaceId)))
      .returning();
    if (!updated) return NextResponse.json({ error: 'not found' }, { status: 404 });
    return NextResponse.json(updated);
  } catch (err) {
    return errorResponse(err);
  }
}

export async function DELETE(_req: NextRequest, { params }: Ctx) {
  try {
    const ctx = await requireRole('admin');
    const { ruleId } = await params;
    const db = getDb();
    const [deleted] = await db
      .delete(schema.automationRules)
      .where(and(eq(schema.automationRules.id, ruleId), eq(schema.automationRules.workspaceId, ctx.workspaceId)))
      .returning({ id: schema.automationRules.id });
    if (!deleted) return NextResponse.json({ error: 'not found' }, { status: 404 });
    return new NextResponse(null, { status: 204 });
  } catch (err) {
    return errorResponse(err);
  }
}

function errorResponse(err: unknown): NextResponse {
  if (err instanceof z.ZodError) return NextResponse.json({ error: err.flatten() }, { status: 400 });
  const status =
    err && typeof err === 'object' && 'status' in err && typeof (err as { status: number }).status === 'number'
      ? (err as { status: number }).status
      : 500;
  const message = err instanceof Error ? err.message : 'internal error';
  return NextResponse.json({ error: message }, { status });
}
```

```tsx
import { desc, eq, sql as rawSql } from 'drizzle-orm';
import { getDb } from '@/db/client';
import * as schema from '@/db/schema';
import { requireRole } from '@/lib/auth/require-role';
import { RuleList } from '@/components/automation/rule-list';

export const dynamic = 'force-dynamic';

type RuleRow = schema.AutomationRule & {
  lastStatus: schema.AutomationRunStatus | null;
  lastRunAt: Date | null;
};

async function loadRules(workspaceId: string): Promise<RuleRow[]> {
  const db = getDb();
  // One LATERAL join to pull the newest run per rule. Drizzle 0.45 lacks a
  // first-class LATERAL builder, so this is a single raw SQL query.
  const rows = (await db.execute(rawSql`
    SELECT r.*, lr.status AS last_status, lr.created_at AS last_run_at
    FROM automation_rules r
    LEFT JOIN LATERAL (
      SELECT status, created_at
      FROM automation_runs
      WHERE rule_id = r.id
      ORDER BY created_at DESC
      LIMIT 1
    ) lr ON TRUE
    WHERE r.workspace_id = ${workspaceId}::uuid
    ORDER BY r.created_at DESC
  `)) as unknown as Array<{
    id: string;
    workspace_id: string;
    name: string;
    trigger_event: string;
    condition: schema.AutomationCondition;
    action_type: string;
    action_config: Record<string, unknown>;
    enabled: boolean;
    created_by: string | null;
    created_at: Date;
    last_status: schema.AutomationRunStatus | null;
    last_run_at: Date | null;
  }>;
  return rows.map((r) => ({
    id: r.id,
    workspaceId: r.workspace_id,
    name: r.name,
    triggerEvent: r.trigger_event,
    condition: r.condition,
    actionType: r.action_type as schema.AutomationActionType,
    actionConfig: r.action_config,
    enabled: r.enabled,
    createdBy: r.created_by,
    createdAt: r.created_at,
    lastStatus: r.last_status,
    lastRunAt: r.last_run_at,
  }));
}

export default async function AutomationSettingsPage() {
  const ctx = await requireRole('viewer'); // gate at workspace-member level; mutations require admin
  const canMutate = ctx.role === 'admin' || ctx.role === 'owner';
  const rules = await loadRules(ctx.workspaceId);

  return (
    <div className="mx-auto max-w-4xl space-y-6 p-6">
      <header className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-semibold">Automation</h1>
          <p className="text-muted-foreground text-sm">
            Run actions when events fire in this workspace.
          </p>
        </div>
      </header>
      <RuleList initialRules={rules} canMutate={canMutate} />
    </div>
  );
}
```

```tsx
'use client';

import { useState } from 'react';
import { Badge } from '@/components/ui/badge';
import { Button } from '@/components/ui/button';
import { Switch } from '@/components/ui/switch';
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
import type * as schema from '@/db/schema';
import { RuleForm } from './rule-form';

type RuleRow = schema.AutomationRule & {
  lastStatus: schema.AutomationRunStatus | null;
  lastRunAt: Date | null;
};

export function RuleList({
  initialRules,
  canMutate,
}: {
  initialRules: RuleRow[];
  canMutate: boolean;
}) {
  const [rules, setRules] = useState(initialRules);
  const [editing, setEditing] = useState<RuleRow | null>(null);
  const [creating, setCreating] = useState(false);

  async function setEnabled(rule: RuleRow, enabled: boolean) {
    const res = await fetch(`/api/automation/rules/${rule.id}`, {
      method: 'PATCH',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ enabled }),
    });
    if (res.ok) {
      const updated = await res.json();
      setRules((rs) => rs.map((r) => (r.id === rule.id ? { ...r, ...updated } : r)));
    }
  }

  async function remove(rule: RuleRow) {
    if (!confirm(`Delete rule "${rule.name}"?`)) return;
    const res = await fetch(`/api/automation/rules/${rule.id}`, { method: 'DELETE' });
    if (res.ok) setRules((rs) => rs.filter((r) => r.id !== rule.id));
  }

  return (
    <div className="space-y-4">
      {canMutate ? (
        <div className="flex justify-end">
          <Button onClick={() => setCreating(true)}>New rule</Button>
        </div>
      ) : null}
      {rules.length === 0 ? (
        <p className="text-muted-foreground text-sm">No automation rules yet.</p>
      ) : (
        <Table>
          <TableHeader>
            <TableRow>
              <TableHead>Name</TableHead>
              <TableHead>Trigger</TableHead>
              <TableHead>Action</TableHead>
              <TableHead>Last run</TableHead>
              <TableHead>Enabled</TableHead>
              <TableHead className="w-24" />
            </TableRow>
          </TableHeader>
          <TableBody>
            {rules.map((r) => (
              <TableRow key={r.id}>
                <TableCell className="font-medium">{r.name}</TableCell>
                <TableCell><Badge variant="secondary">{r.triggerEvent}</Badge></TableCell>
                <TableCell><Badge variant="outline">{r.actionType}</Badge></TableCell>
                <TableCell className="text-muted-foreground text-xs">
                  {r.lastStatus ? `${r.lastStatus} · ${r.lastRunAt?.toString().slice(0, 16) ?? ''}` : '—'}
                </TableCell>
                <TableCell>
                  <Switch
                    checked={r.enabled}
                    disabled={!canMutate}
                    onCheckedChange={(v) => setEnabled(r, v)}
                  />
                </TableCell>
                <TableCell className="flex gap-1">
                  {canMutate ? (
                    <>
                      <Button size="sm" variant="ghost" onClick={() => setEditing(r)}>Edit</Button>
                      <Button size="sm" variant="ghost" onClick={() => remove(r)}>Delete</Button>
                    </>
                  ) : null}
                </TableCell>
              </TableRow>
            ))}
          </TableBody>
        </Table>
      )}
      {creating ? (
        <RuleForm
          mode="create"
          onClose={(saved) => {
            setCreating(false);
            if (saved) setRules((rs) => [saved, ...rs]);
          }}
        />
      ) : null}
      {editing ? (
        <RuleForm
          mode="edit"
          rule={editing}
          onClose={(saved) => {
            setEditing(null);
            if (saved) setRules((rs) => rs.map((r) => (r.id === saved.id ? { ...r, ...saved } : r)));
          }}
        />
      ) : null}
    </div>
  );
}
```

```tsx
    (initialCondition.operator as schema.AutomationOperator) ?? 'equals',
  );
  const [value, setValue] = useState(
    initialCondition.value != null ? String(initialCondition.value) : '',
  );
  const [actionType, setActionType] = useState<schema.AutomationActionType>(
    (initial?.actionType as schema.AutomationActionType) ?? 'notify',
  );
  const [actionConfig, setActionConfig] = useState(
    JSON.stringify(initial?.actionConfig ?? {}, null, 2),
  );
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function submit() {
    setError(null);
    setSaving(true);
    try {
      let parsedConfig: Record<string, unknown>;
      try {
        parsedConfig = JSON.parse(actionConfig || '{}');
      } catch {
        throw new Error('action config must be valid JSON');
      }
      const condition =
        property.length === 0
          ? {}
          : { property, operator, value: parseLiteral(value) };
      const body = {
        name,
        triggerEvent,
        condition,
        actionType,
        actionConfig: parsedConfig,
      };
      const url = props.mode === 'create' ? '/api/automation/rules' : `/api/automation/rules/${props.rule.id}`;
      const method = props.mode === 'create' ? 'POST' : 'PATCH';
      const res = await fetch(url, {
        method,
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify(body),
      });
      if (!res.ok) throw new Error(`${res.status}: ${(await res.json()).error ?? 'request failed'}`);
      const saved = (await res.json()) as schema.AutomationRule;
      props.onClose(saved);
    } catch (err) {
      setError(err instanceof Error ? err.message : String(err));
    } finally {
      setSaving(false);
    }
  }

  return (
    <Dialog open onOpenChange={(open) => !open && props.onClose(null)}>
      <DialogContent className="max-w-2xl">
        <DialogHeader>
          <DialogTitle>{props.mode === 'create' ? 'New rule' : `Edit ${initial?.name}`}</DialogTitle>
        </DialogHeader>
        <div className="space-y-4">
          <div className="space-y-1">
            <Label>Name</Label>
            <Input value={name} onChange={(e) => setName(e.target.value)} />
          </div>
          <div className="space-y-1">
            <Label>Trigger</Label>
            <Select value={triggerEvent} onValueChange={(v) => setTriggerEvent(v as (typeof TRIGGERS)[number])}>
              <SelectTrigger><SelectValue /></SelectTrigger>
              <SelectContent>
                {TRIGGERS.map((t) => <SelectItem key={t} value={t}>{t}</SelectItem>)}
              </SelectContent>
            </Select>
          </div>
          <fieldset className="space-y-2 rounded-md border p-3">
            <legend className="px-1 text-sm font-medium">Condition (optional)</legend>
            <div className="grid grid-cols-3 gap-2">
              <Input placeholder="property path (e.g. row.cells.status)" value={property} onChange={(e) => setProperty(e.target.value)} />
              <Select value={operator} onValueChange={(v) => setOperator(v as schema.AutomationOperator)}>
                <SelectTrigger><SelectValue /></SelectTrigger>
                <SelectContent>
                  {OPERATORS.map((o) => <SelectItem key={o} value={o}>{o}</SelectItem>)}
                </SelectContent>
              </Select>
              <Input placeholder="value" value={value} onChange={(e) => setValue(e.target.value)} />
            </div>
            <p className="text-muted-foreground text-xs">
              Leave the property blank to match every event. Numeric/boolean values are auto-coerced.
            </p>
          </fieldset>
          <div className="space-y-1">
            <Label>Action</Label>
            <Select value={actionType} onValueChange={(v) => setActionType(v as schema.AutomationActionType)}>
              <SelectTrigger><SelectValue /></SelectTrigger>
              <SelectContent>
                {ACTIONS.map((a) => <SelectItem key={a} value={a}>{a}</SelectItem>)}
              </SelectContent>
            </Select>
          </div>
          <div className="space-y-1">
            <Label>Action config (JSON)</Label>
            <textarea
              className="border-input w-full rounded-md border bg-background px-3 py-2 font-mono text-xs"
              rows={8}
              value={actionConfig}
              onChange={(e) => setActionConfig(e.target.value)}
            />
            <p className="text-muted-foreground text-xs">
              Shape depends on the action — see <code>src/lib/automation/actions/</code>.
            </p>
          </div>
          {error ? <p className="text-destructive text-sm">{error}</p> : null}
        </div>
        <DialogFooter>
          <Button variant="ghost" onClick={() => props.onClose(null)}>Cancel</Button>
          <Button onClick={submit} disabled={saving || !name}>{saving ? 'Saving…' : 'Save rule'}</Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}

function parseLiteral(s: string): unknown {
  if (s === '') return null;
  if (s === 'true') return true;
  if (s === 'false') return false;
  const n = Number(s);
  if (!Number.isNaN(n) && s.trim() !== '') return n;
  return s;
}
```
